# Cycles the player's cadence: 0 fast, 1 medium, 2 slow. Unset counts as medium.
# The location pulses read it through the cadence triggers.

mnt_wealth_cadence_cycle = {
	scope = country
	is_shown = { is_ai = no }
	is_valid = { is_ai = no }

	effect = {
		if = {
			limit = { NOT = { has_variable = mnt_wealth_cadence } }
			set_variable = { name = mnt_wealth_cadence value = 1 }
		}
		if = {
			limit = { var:mnt_wealth_cadence = 2 }
			set_variable = { name = mnt_wealth_cadence value = 0 }
		}
		else = {
			change_variable = { name = mnt_wealth_cadence add = 1 }
		}
	}
}
