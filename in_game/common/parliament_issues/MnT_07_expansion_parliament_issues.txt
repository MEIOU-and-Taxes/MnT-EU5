REPLACE:establish_common_food_stores = {
	estate = peasants_estate

	modifier_when_in_debate = {
		#MnT removed: global_monthly_food_modifier = 0.05
		global_provisions_output_modifier = 0.05
	}

	allow = {
		always = yes
	}

	chance = {
		add = 1
	}

	on_debate_passed = {
		add_country_modifier = {
			modifier = parl_common_food_stores_mod
			years = 10
			mode = add_and_extend
		}
	}

	on_debate_failed = {
		add_prestige = prestige_mild_penalty
		parliament_debate_failed_effect = yes
	}
}
