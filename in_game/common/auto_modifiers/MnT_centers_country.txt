# M&T Centers of Importance - country-wide culture centre effects
# Country keys M&T had inside location modifiers (where they were dropped), scaled by the highest culture
# centre tier held. Tiers do not stack; bases are the world values, the scales hold the ladder.

mnt_center_culture_capacity_country = {
	category = country
	potential_trigger = {
		has_variable = mnt_center_culture_tier
		var:mnt_center_culture_tier > 0
	}
	scales_with = center_culture_capacity_scale
	cultures_capacity = 1
}

mnt_center_culture_prestige_country = {
	category = country
	potential_trigger = {
		has_variable = mnt_center_culture_tier
		var:mnt_center_culture_tier > 0
	}
	scales_with = center_culture_prestige_scale
	monthly_prestige = 0.1
}
