### M&T overwriting entire Vanilla file to replace climate boni and modify the location_potential
# In M&T the Province needs to produce Camels from them to be buildable in any Location
# This to prevent full camel armies, and to signify that you are recruiting camels from where they thrive, not from where-ever

# M&T: camels fast in desert (no vegetation) and sparse vegetation, slowed down by jungle and forest
# Combat: desert and sparse boni

a_light_camelry_1 = {
	is_special = yes
	buildable = yes
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry
	upgrades_to = a_light_camelry_2

	morale_damage_done = 0.05
	initiative = 5
	movement_speed = 0.20

	location_potential = {
		unit_camelry_location_trigger = yes    #Changed by M&T to require camels production in Province
	}

	impact = {
		desert = -0.5    #M&T from -0.75
		sparse = -0.25    #M&T from arid climate
		forest = 0.15   #M&T from arctic climate
		jungle = 0.25
	}
	combat = {
		desert = 0.15
		sparse = 0.1     #M&T from arid climate
	}

	#root = location. Checks location_potential first so don't need that scripting in
	mercenaries_per_location = {
		pop_type = tribesmen
		multiply = 0.3
	}

	maintenance_demand = camel_cavalry_early_maintenance
	construction_demand = camel_cavalry_early_construction

	gfx_tags = { camel_tag medium_tag }
}
