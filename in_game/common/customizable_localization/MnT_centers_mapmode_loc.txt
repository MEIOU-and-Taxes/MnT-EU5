# Line the "Centers of Importance" map mode tooltip prints for one category.
# ROOT: location. One key per category, the map mode tooltip stacks all four.
# Each line reuses the tooltip of the tier the location currently holds, and a
# short line when it holds none.

mnt_centers_mapmode_line_trade = {
	type = location

	text = {
		trigger = {
			has_location_modifier = mnt_center_trade_world
		}
		localization_key = mapmode_center_trade_world_tooltip
	}

	text = {
		trigger = {
			has_location_modifier = mnt_center_trade_continental
		}
		localization_key = mapmode_center_trade_continental_tooltip
	}

	text = {
		trigger = {
			has_location_modifier = mnt_center_trade_regional
		}
		localization_key = mapmode_center_trade_regional_tooltip
	}

	text = {
		trigger = {
			has_location_modifier = mnt_center_trade_local
		}
		localization_key = mapmode_center_trade_local_tooltip
	}

	text = {
		localization_key = mapmode_center_trade_none_tooltip
		fallback = yes
	}
}
