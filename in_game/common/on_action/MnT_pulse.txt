on_game_start = {
	on_actions = {
		mnt_rgo_seed_eligibility
		mnt_rgo_setup_dump_on_start	# Reports nothing until the building setup generator runs with --setup-data-dump
		mnt_building_setup_removals_on_start
		on_game_start_MnT
		hide_buildings_on_start
		epbm_on_initialize
		aut_on_game_start
		delay = { days = 1 }
		on_game_has_started_MnT
	}
}

#Seed RGO eligibility cache before the setup dump and on_game_start_MnT: the RGO build triggers read the cache.
mnt_rgo_seed_eligibility = {
	effect = {
		set_global_variable = mnt_rgo_eligibility_initialized
		every_ownable_location = {
			mnt_rgo_refresh_eligibility = yes
		}
	}
}

# Reports the start state to debug.log for the building setup generator, before the mod's other
# game start changes and after the eligibility cache the build triggers read is seeded
mnt_rgo_setup_dump_on_start = {
	effect = {
		mnt_rgo_setup_dump = yes
	}
}

# Removes the starting building levels the building setup generator replaces, before the mod's other game start changes
mnt_building_setup_removals_on_start = {
	effect = {
		mnt_remove_starting_building_levels = yes
	}
}

		}
	}
}

# The below runs BEFORE country selection so is_ai and is_human checks will not work!
on_game_start_MnT = {
	effect = {
		# Remove Societies of Pops, they don't do anything yet cost performance
		fum_remove_sop_effect = yes
		# Initialize center of importance systems on game start
		random_country = { trigger_event_silently = { id = mnt_centers.1 } }
		location:al_hajr.market = { destroy_market = { } }
		location:urgench.market = { destroy_market = { } }
		every_market_in_world = {	#Helicopter-tools to make the world go vroom
			add_goods_supply = {
				goods = goods:tools
				amount = 250
			}
		}

		# Initialize update frequencies and trigger first update
		every_country = {
			trigger_event_silently = { id = rgo_removal.01 }
			trigger_event_silently = { id = mnt_goods_value.1 }
		}
		# Handle Pop startup
		every_country = {
			limit = {
				num_locations > 0
			}
			trigger_event_silently = { id = mnt_slaves.1 } # Has to be before mnt_tribes.2 because we're creating peasants
			trigger_event_silently = { id = mnt_tribes.2 }
			trigger_event_silently = { id = mnt_tribes.3 }
		}
		# Provisions demand exists from day one, after the pop startup above
		mnt_provisions_food_demand_month = yes
	}
}

# If you need things to fire on the start of the game AFTER the player country is defined use this one
# The delay of one day is real though, this will fire 2nd of April 1337
# Ergo, firing things for the player on literally the first day of the game is just not possible
on_game_has_started_MnT = {
	effect = {
		# Initialize the hidden buildings system
		every_country = {
			limit = {
				is_human = yes
			}
			# Show RGO buildings in the UI for the player by default
			set_variable = { name = mnt_show_rgo_buildings value = 1 }
			# M&T Startup welcome event	
			trigger_event_non_silently = { id = mnt_information.1 }
			# Income graph for the player
			if = {
				limit = { NOT = { has_variable = mnt_goods_value_update_freq } }
				set_variable = { name = mnt_goods_value_update_freq value = 0 }
			}
			if = {
				limit = { NOT = { has_variable = mnt_income_graph_update_freq } }
				set_variable = { name = mnt_income_graph_update_freq value = 0 }
			}
			mnt_update_income_graph_history = yes
		}
	}
}
