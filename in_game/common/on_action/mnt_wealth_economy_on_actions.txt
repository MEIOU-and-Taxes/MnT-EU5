# Each handler hands off to named effects in the relevant system file.

mnt_on_game_start_after_lobby = {
	effect = {
		mnt_wealth_seed_economy_constants = yes
		# Set the elapsed month now, because the owner-change and capture hooks
		# can fire before the first monthly pulse and need a month to read.
		mnt_wealth_begin_month = yes
		mnt_wealth_seed_owned_locations = yes
		# Countries come after the locations because the national totals sum them.
		every_landed_country = { mnt_wealth_initialize_country = yes }
		every_country = { mnt_wealth_apply_default_loot_stance = yes }
		random_country = { trigger_event_silently = { id = mnt_centers.1 } }
	}
}

# The national estate raw power map is cleared when a civil war resolves, so rebuild it from the owned locations.

mnt_on_civil_war_annexed = {
	effect = {
		scope:target ?= { mnt_wealth_clear_annexed_country = yes }
		mnt_wealth_rebuild_national_estate_raw_power = yes
	}
}

# Rebels win: the engine fires this first, root = the rebel country annexing the original, then on_civil_war_lost
# with the rebels as scope:winner, and only after every hook hands the survivor the original's variable maps. The
# pre-terms war hooks never fire for a civil war. The rebel side is settled against its own counter here while it
# can still be read.
mnt_on_civil_war_annex = {
	effect = {
		if = {
			limit = { is_rebel_country = yes }
			mnt_wealth_settle_rebel_side_before_swap = yes
		}
	}
}

mnt_on_civil_war_won = {
	effect = {
		scope:winner ?= { mnt_wealth_settle_rebel_side_before_swap = yes }
		mnt_wealth_rebuild_national_estate_raw_power = yes
		trigger_event_silently = { id = mnt_wealth_economy.31 days = 1 }
	}
}

mnt_on_civil_war_lost = {
	effect = {
		scope:winner ?= { mnt_wealth_settle_rebel_side_before_swap = yes }
		scope:winner ?= {
			mnt_wealth_rebuild_national_estate_raw_power = yes
			trigger_event_silently = { id = mnt_wealth_economy.31 days = 1 }
		}
	}
}

mnt_on_location_changed_owner = {
	effect = {
		mnt_wealth_transfer_location_between_owners = yes

		# Decolonization: no new owner, so remove this location from its service
		# bucket so the dispatch iteration never picks it up.
		if = {
			limit = { NOT = { exists = scope:winner } }
			mnt_wealth_unschedule_location = yes

			save_temporary_scope_as = mnt_wealth_decolonize_loc
			remove_from_global_variable_map = { name = mnt_wealth_location_last_service key = scope:mnt_wealth_decolonize_loc }
		}

		# The census runs here, after the transfer, on the tick: the effect block and the events list have no
		# defined order between them, and the national totals corrupt if the census runs before or later.
		if = {
			limit = { has_owner = yes }
			mnt_wealth_settle_owner_change = yes
		}
	}

	# The modifier reads wait a day: modifiers for a new country need a day to initialize, so the wage floor
	# and the cached rates read zero from the owner on its formation frame.
	events = {
		delay = { days = 1 }
		mnt_wealth_economy.32
	}
}

# Every location pulse checks for an owner: a balanced pulse lands up to a month after the dispatch that queued it,
# and a location can lose its owner in between.

# The cadence table. Slow is every AI country and any human who chose it; medium is the human default; fast runs
# everything monthly.

mnt_wealth_monthly_human_location_pulse = {
	trigger = { has_owner = yes  owner = { NOT = { mnt_wealth_cadence_is_slow = yes } } }
	effect = {
		mnt_wealth_location_income_and_expenses = yes
		if = {
			limit = { owner = { mnt_wealth_cadence_is_fast = yes } }
			mnt_wealth_reconcile_location = yes
			mnt_wealth_census_location = yes
		}
	}
}

mnt_wealth_yearly_location_pulse = {

# ROOT = the occupied location. controller = the occupier, owner = the original owner.
mnt_custom_on_location_occupied = {
	# on_location_occupied and on_siege_won only fire for locations directly sieged by an army
	# This costs about 11ms per month spread across the days in the month and handles all locations
	# when they change from owner == controller to owner != controller.
	# It does not fire when the owner re-occupies the location, and does not fire on location transfers
	# from civil wars or auto conquest wars.
	# This on_action and the system that triggers it can be removed when on_location_occupied is fixed to
	# work on locations auto-occupied from province capitals.
	effect = {
		controller = {
			save_temporary_scope_as = mnt_wealth_occupier
			mnt_wealth_read_occupier_loot_stance = yes
		}
		mnt_wealth_loot_occupied_location = yes
	}
}

mnt_occupation_on_game_start = {
	effect = {
		mnt_occupation_build_initial_state = yes
		mnt_occupation_schedule_daily_scans = yes
	}
}

	trigger = { is_ai = no exists = scope:war }
	effect = { mnt_wealth_report_war_loot = yes }
}
