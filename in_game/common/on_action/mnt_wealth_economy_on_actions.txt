# Each handler hands off to named effects in the relevant system file.

mnt_on_game_start_after_lobby = {
	effect = {
		mnt_wealth_seed_economy_constants = yes
		# Set the elapsed month now, because the owner-change and capture hooks
		# can fire before the first monthly pulse and need a month to read.
		mnt_wealth_begin_month = yes
		mnt_wealth_seed_owned_locations = yes
		# Countries come after the locations because the national totals sum them.
		every_landed_country = { mnt_wealth_initialize_country = yes }
		random_country = { trigger_event_silently = { id = mnt_centers.1 } }
	}
}

mnt_on_weather_monthly_pulse = {
	effect = {
		mnt_wealth_maintain_building_produced_goods_cache = yes
		mnt_wealth_begin_month = yes
		mnt_dispatch_location_and_market_on_actions = yes
		clear_global_variable_list = mnt_wealth_player_countries
	}
}

mnt_on_siege_won = {

	}

	}

mnt_on_annexed = {
	effect = {
		scope:target ?= { mnt_wealth_clear_annexed_country = yes }
	}
}

# The national estate raw power map is cleared when a civil war resolves, so rebuild it from the owned locations.

mnt_on_civil_war_annexed = {
	effect = {
		scope:target ?= { mnt_wealth_clear_annexed_country = yes }
		mnt_wealth_rebuild_national_estate_raw_power = yes
	}
}

mnt_on_civil_war_won = {
	effect = {
		mnt_wealth_rebuild_national_estate_raw_power = yes
		trigger_event_silently = { id = mnt_wealth_economy.31 days = 1 }
	}
}

mnt_on_civil_war_lost = {
	effect = {
		scope:winner ?= {
			mnt_wealth_rebuild_national_estate_raw_power = yes
			trigger_event_silently = { id = mnt_wealth_economy.31 days = 1 }
		}
	}
}

mnt_on_location_changed_owner = {
	effect = {
		mnt_wealth_transfer_location_between_owners = yes

		# Decolonization: no new owner, so remove this location from its service
		# bucket so the dispatch iteration never picks it up.
		if = {
			limit = { NOT = { exists = scope:winner } }
			mnt_wealth_unschedule_location = yes

			save_temporary_scope_as = mnt_wealth_decolonize_loc
			remove_from_global_variable_map = { name = mnt_wealth_location_last_service key = scope:mnt_wealth_decolonize_loc }
		}
	}

	# Delayed one day because new countries have no estate data on their formation frame.
	events = { mnt_wealth_economy.30 }
}

# Every location pulse checks for an owner: the balanced ones land up to a month after the dispatch that
# queued them, and a location can lose its owner in between.

mnt_monthly_instant_human_location_pulse = {
	trigger = { exists = owner }
	effect = {
		mnt_wealth_human_location_month = yes
		# The fast census frequency moves the census and review from the yearly service to here.
		if = {
			limit = { owner ?= { mnt_wealth_census_frequency_is_fast = yes } }
			mnt_wealth_monthly_human_census = yes
		}
	}
}
