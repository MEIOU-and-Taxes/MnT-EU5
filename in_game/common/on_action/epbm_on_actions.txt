@epbm_rebuild_version = 510

epbm_on_initialize = {
	effect = {
		epbm_initialize = yes
	}
}

epbm_on_global_monthly = {
	effect = {
		epbm_global_monthly = yes
	}
}

epbm_on_market_monthly = {
	effect = {
		location = {
			if = { limit = { has_variable_map = epbm_pm_cache } clear_variable_map = epbm_pm_cache }
			if = { limit = { has_variable_map = epbm_bt_costs } clear_variable_map = epbm_bt_costs }
		}
	}
}

epbm_on_location_yearly = {
	trigger = { has_owner = yes }
	effect = { epbm_price_location_buildings = yes }
}

epbm_on_country_monthly = {
	trigger = { has_global_variable = epbm_version any_active_estate = { exists = yes } }

	events = { epbm.1 }
}

epbm_on_country_yearly = {
	trigger = {
		has_global_variable = epbm_version
		global_var:epbm_version >= @epbm_rebuild_version
		any_active_estate = { exists = yes }
	}

	events = { epbm.2 }
}
