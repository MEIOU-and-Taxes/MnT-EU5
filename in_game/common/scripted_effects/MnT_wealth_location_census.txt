# The census: gross output from buildings and prices, estate pops and raw power, departed estates.

# The census: every slow-rolling figure a location carries from the location wealth system.

mnt_wealth_census_location = {
	mnt_wealth_update_local_gross_output = yes
	mnt_wealth_retake_estate_census = yes
	# Gross output moved, so the cached wage bill moves with it.
	mnt_wealth_set_location_wages = yes
	set_local_variable = { name = mnt_wealth_rebuild_location_estate_list value = yes }
}

mnt_wealth_update_local_gross_output = {
	save_scope_as = mnt_wealth_gross_output_location
	mnt_wealth_remove_local_gross_output_from_national_total = yes
	mnt_wealth_clear_found_goods = yes
	every_buildings_in_location         = { mnt_wealth_cache_building_produced_goods = yes }
	every_foreign_buildings_in_location = { mnt_wealth_cache_building_produced_goods = yes }
	mnt_wealth_calculate_local_gross_output = yes
	mnt_wealth_raise_gross_output_map_mode_ceilings = yes
	mnt_wealth_add_local_gross_output_to_national_total = yes
}

mnt_wealth_clear_found_goods = {
	clear_local_variable_map = mnt_wealth_found_goods_map
	clear_local_variable_list = mnt_wealth_found_goods_list
}

mnt_wealth_cache_building_produced_goods = {
	building_type ?= { save_temporary_scope_as = mnt_wealth_current_building_type }
	mnt_wealth_classify_building_type = yes
	mnt_wealth_collect_building_goods = yes
}

mnt_wealth_classify_building_type = {
	if = {
		limit = {
			NOT = {
				is_key_in_global_variable_map = {
					name = mnt_wealth_building_makes_goods
					target = scope:mnt_wealth_current_building_type
				}
			}
		}
		if = {
			limit = { any_production_method_of_building = { exists = produced_goods } }
			add_to_global_variable_map = {
				name = mnt_wealth_building_makes_goods
				key = scope:mnt_wealth_current_building_type
				value = 1
			}
		}
		else = {
			add_to_global_variable_map = {
				name = mnt_wealth_building_makes_goods
				key = scope:mnt_wealth_current_building_type
				value = 0
			}
		}
	}
}

mnt_wealth_collect_building_goods = {
	if = {
		limit = { "global_variable_map(mnt_wealth_building_makes_goods|scope:mnt_wealth_current_building_type)" > 0 }
		every_production_method_of_building = {
			limit = {
				exists = produced_goods
				NOT = { is_key_in_local_variable_map = { name = mnt_wealth_found_goods_map target = produced_goods } }
			}
			add_to_local_variable_map = {
				name = mnt_wealth_found_goods_map
				key = produced_goods
				value = 1
			}

			add_to_local_variable_list = { name = mnt_wealth_found_goods_list target = produced_goods }
		}
	}
}


mnt_wealth_calculate_local_gross_output = {
	set_variable = { name = mnt_wealth_local_gross_output value = 0 }
	set_variable = { name = mnt_wealth_local_rural_gross_output value = 0 }
	set_variable = { name = mnt_wealth_local_urban_gross_output value = 0 }
	clear_variable_map = mnt_wealth_goods_value

	every_in_local_list = {
		variable = mnt_wealth_found_goods_list
		save_temporary_scope_as = mnt_wealth_current_good
		set_local_variable = { name = mnt_wealth_current_price value = "price_in_market(scope:mnt_wealth_gross_output_location.market)" }

		scope:mnt_wealth_gross_output_location = {
			set_local_variable = {
				name = mnt_wealth_good_value
				value = { value = "goods_output(scope:mnt_wealth_current_good)" multiply = local_var:mnt_wealth_current_price }
			}

			add_to_variable_map = { name = mnt_wealth_goods_value key = scope:mnt_wealth_current_good value = local_var:mnt_wealth_good_value }
			change_variable = { name = mnt_wealth_local_gross_output add = local_var:mnt_wealth_good_value }

			if = {
				limit = { "global_variable_map(mnt_wealth_good_classification|scope:mnt_wealth_current_good)" = flag:urban }
				change_variable = { name = mnt_wealth_local_urban_gross_output add = local_var:mnt_wealth_good_value }
			}
			else = {
				change_variable = { name = mnt_wealth_local_rural_gross_output add = local_var:mnt_wealth_good_value }
			}
		}
	}
}

mnt_wealth_maintain_building_produced_goods_cache = {
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = mnt_wealth_building_table_age }
				global_var:mnt_wealth_building_table_age >= mnt_wealth_building_table_lifetime_months
			}
		}
		clear_global_variable_map = mnt_wealth_building_makes_goods
		set_global_variable = { name = mnt_wealth_building_table_age value = 0 }
	}
	else = {
		change_global_variable = { name = mnt_wealth_building_table_age add = 1 }
	}
}

mnt_wealth_retake_estate_census = {
	owner = { save_temporary_scope_as = mnt_wealth_raw_power_country }
	mnt_wealth_apply_national_estate_raw_power = { SIGN = -1 }
	mnt_wealth_census_estate_data = yes
	mnt_wealth_apply_national_estate_raw_power = { SIGN = 1 }
}

mnt_wealth_census_estate_data = {
	# Raw power is the measure of presence: an estate with pops here is present, one holding wealth with no pops left has departed.
	clear_variable_map = mnt_estate_raw_power
	set_variable = { name = mnt_wealth_local_raw_power_total value = 0 }
	mnt_wealth_count_estate_pops = yes
	mnt_wealth_refresh_relative_power = yes

	# Departed estates: strip modifiers, redistribute gold to remaining estates.
	set_local_variable = { name = mnt_wealth_freed_investment    value = 0 }
	set_local_variable = { name = mnt_wealth_freed_estate_wealth value = 0 }
	every_active_estate = {
		limit = {
			prev = {
				NOT = { is_key_in_variable_map = { name = mnt_estate_raw_power target = prev.estate_type } }
				OR = {
					is_key_in_variable_map = { name = mnt_wealth_estate_wealth   target = prev.estate_type }
					is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = prev.estate_type }
				}
			}
		}
		estate_type = { save_temporary_scope_as = mnt_wealth_stale_key }
		prev = { mnt_wealth_release_absent_estate = yes }
	}
	if = {
		limit = { OR = { local_var:mnt_wealth_freed_investment != 0 local_var:mnt_wealth_freed_estate_wealth != 0 } }
		set_local_variable = { name = mnt_wealth_rebuild_location_estate_list value = yes }
		mnt_wealth_for_each_cached_estate = { EFFECT = mnt_wealth_redistribute_to_estate }
	}
}

# Counts pops into estates and computes raw power (pops x power_per_pop).

mnt_wealth_count_estate_pops = {
	mnt_wealth_clear_pop_counts = yes

	mnt_wealth_count_pops_by_type = yes

	# Dhimmi can take any pop type, so its presence forces the full pop iteration.
	if = {
		limit = { "local_estate_power(estate_type:dhimmi_estate)" > 0 }
		mnt_wealth_clear_pop_counts = yes
		mnt_wealth_count_pops_full = yes
	}
	else = {
		# Slaves carry no power, so a dhimmi estate of only slaves reads as absent above.
		if = {
			limit = {
				local_var:mnt_wealth_pops_slaves > 0
				owner = { country_has_estate = estate_type:dhimmi_estate }
			}
			mnt_wealth_count_pops_slave_split = yes
		}

		if = {
			limit = { "local_estate_power(estate_type:cossacks_estate)" > 0 }
			mnt_wealth_count_pops_tribesmen_split = yes
		}
	}

	set_local_variable = { name = mnt_wealth_pops_raw_total value = 0 }

	mnt_wealth_store_estate_pops = { ESTATE = nobles }
	mnt_wealth_store_estate_pops = { ESTATE = clergy }
	mnt_wealth_store_estate_pops = { ESTATE = burghers }
	mnt_wealth_store_estate_pops = { ESTATE = peasants }
	mnt_wealth_store_estate_pops = { ESTATE = dhimmi }
	mnt_wealth_store_estate_pops = { ESTATE = tribes }
	mnt_wealth_store_estate_pops = { ESTATE = cossacks }

	set_variable = { name = mnt_wealth_local_raw_power_total value = local_var:mnt_wealth_pops_raw_total }
}

mnt_wealth_clear_pop_counts = {
	set_local_variable = { name = mnt_wealth_pops_nobles value = 0 }
	set_local_variable = { name = mnt_wealth_pops_clergy value = 0 }
	set_local_variable = { name = mnt_wealth_pops_burghers value = 0 }
	set_local_variable = { name = mnt_wealth_pops_peasants value = 0 }
	set_local_variable = { name = mnt_wealth_pops_dhimmi value = 0 }
	set_local_variable = { name = mnt_wealth_pops_tribes value = 0 }
	set_local_variable = { name = mnt_wealth_pops_cossacks value = 0 }
}

mnt_wealth_count_pops_by_type = {
	# One local per pop type, so food demand can multiply each by its consumption later.
	set_local_variable = { name = mnt_wealth_pop_type_nobles    value = num_pop_type:nobles }
	set_local_variable = { name = mnt_wealth_pop_type_clergy    value = num_pop_type:clergy }
	set_local_variable = { name = mnt_wealth_pop_type_burghers  value = num_pop_type:burghers }
	set_local_variable = { name = mnt_wealth_pop_type_laborers  value = num_pop_type:laborers }
	set_local_variable = { name = mnt_wealth_pop_type_soldiers  value = num_pop_type:soldiers }
	set_local_variable = { name = mnt_wealth_pop_type_peasants  value = num_pop_type:peasants }
	set_local_variable = { name = mnt_wealth_pop_type_tribesmen value = num_pop_type:tribesmen }
	set_local_variable = { name = mnt_wealth_pop_type_slaves    value = num_pop_type:slaves }

	# Estate buckets from the pop type counts.
	set_local_variable = { name = mnt_wealth_pops_nobles   value = local_var:mnt_wealth_pop_type_nobles }
	set_local_variable = { name = mnt_wealth_pops_clergy   value = local_var:mnt_wealth_pop_type_clergy }
	set_local_variable = { name = mnt_wealth_pops_burghers value = local_var:mnt_wealth_pop_type_burghers }
	set_local_variable = { name = mnt_wealth_pops_tribes   value = local_var:mnt_wealth_pop_type_tribesmen }
	set_local_variable = { name = mnt_wealth_pops_slaves   value = local_var:mnt_wealth_pop_type_slaves }
	set_local_variable = {
		name = mnt_wealth_pops_peasants
		value = {
			value = local_var:mnt_wealth_pop_type_laborers
			add = local_var:mnt_wealth_pop_type_soldiers
			add = local_var:mnt_wealth_pop_type_peasants
			add = local_var:mnt_wealth_pop_type_slaves
		}
	}
}

mnt_wealth_count_pops_slave_split = {
	every_pop = {
		limit = {
			pop_type = pop_type:slaves
			is_dhimmi = yes
		}
		change_local_variable = { name = mnt_wealth_pops_dhimmi   add = pop_size }
		change_local_variable = { name = mnt_wealth_pops_peasants subtract = pop_size }
	}
}

mnt_wealth_count_pops_tribesmen_split = {
	set_local_variable = { name = mnt_wealth_pops_tribes value = 0 }

	every_pop = {
		limit = { pop_type = pop_type:tribesmen }
		if = {
			limit = { is_cossacks = yes }
			change_local_variable = { name = mnt_wealth_pops_cossacks add = pop_size }
		}
		else = { change_local_variable = { name = mnt_wealth_pops_tribes add = pop_size } }
	}
}

mnt_wealth_count_pops_full = {
	every_pop = {
		if = {
			limit = { pop_type = pop_type:tribesmen }
			if = {
				limit = { is_cossacks = yes }
				change_local_variable = { name = mnt_wealth_pops_cossacks add = pop_size }
			}
			else = { change_local_variable = { name = mnt_wealth_pops_tribes add = pop_size } }
		}
		else_if = {
			limit = { is_dhimmi = yes }
			change_local_variable = { name = mnt_wealth_pops_dhimmi add = pop_size }
		}
		else_if = {
			limit = { pop_type = pop_type:nobles }
			change_local_variable = { name = mnt_wealth_pops_nobles add = pop_size }
		}
		else_if = {
			limit = { pop_type = pop_type:clergy }
			change_local_variable = { name = mnt_wealth_pops_clergy add = pop_size }
		}
		else_if = {
			limit = { pop_type = pop_type:burghers }
			change_local_variable = { name = mnt_wealth_pops_burghers add = pop_size }
		}
		else = { change_local_variable = { name = mnt_wealth_pops_peasants add = pop_size } }
	}
}

mnt_wealth_store_estate_pops = {
	if = {
		limit = {
			local_var:mnt_wealth_pops_$ESTATE$ > 0
		}
		set_local_variable = { name = mnt_wealth_pops_key value = estate_type:$ESTATE$_estate }
		set_local_variable = {
			name = mnt_wealth_pops_per_pop
			value = "global_variable_map(mnt_wealth_power_per_pop|local_var:mnt_wealth_pops_key)"
		}
		set_local_variable = {
			name = mnt_wealth_pops_raw
			value = {
				value = local_var:mnt_wealth_pops_$ESTATE$
				multiply = local_var:mnt_wealth_pops_per_pop
			}
		}

		add_to_variable_map = {
			name = mnt_estate_raw_power
			key = local_var:mnt_wealth_pops_key
			value = local_var:mnt_wealth_pops_raw
		}

		change_local_variable = { name = mnt_wealth_pops_raw_total add = local_var:mnt_wealth_pops_raw }
	}
}

mnt_wealth_store_estate_census_results = {
	set_local_variable = {
		name = mnt_wealth_estate_relative_power
		value = "local_relative_estate_power(local_var:mnt_wealth_estate_key)"
	}

	set_local_variable = { name = mnt_wealth_raw_power_for_one_estate value = 0 }
	if = {
		limit = {
			is_key_in_variable_map = { name = mnt_estate_raw_power target = local_var:mnt_wealth_estate_key }
		}
		set_local_variable = {
			name = mnt_wealth_raw_power_for_one_estate
			value = "variable_map(mnt_estate_raw_power|local_var:mnt_wealth_estate_key)"
		}
	}

	if = {
		limit = { OR = { local_var:mnt_wealth_estate_relative_power > 0 local_var:mnt_wealth_raw_power_for_one_estate > 0 } }

		add_to_variable_map = {
			name = mnt_wealth_cached_local_relative_estate_power
			key = local_var:mnt_wealth_estate_key
			value = local_var:mnt_wealth_estate_relative_power
		}
	}
}

# Strips a departed estate's modifiers and accumulates its wealth (including neglect debt) into the freed
# totals. Nothing is written off; redistribution runs after all departures.

mnt_wealth_release_absent_estate = {
	set_local_variable = { name = mnt_wealth_estate_key value = scope:mnt_wealth_stale_key }
	mnt_wealth_run_effect_with_estate_name = { EFFECT = mnt_wealth_remove_estate_investment_modifiers }
	mnt_wealth_add_progress_to_invested_wealth = yes

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = local_var:mnt_wealth_estate_key } }

		change_local_variable = {
			name = mnt_wealth_freed_investment
			add = "variable_map(mnt_wealth_invested_wealth|local_var:mnt_wealth_estate_key)"
		}

		remove_from_variable_map = { name = mnt_wealth_invested_wealth key = local_var:mnt_wealth_estate_key }
	}

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }

		change_local_variable = {
			name = mnt_wealth_freed_estate_wealth
			add = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
		}

		remove_from_variable_map = { name = mnt_wealth_estate_wealth key = local_var:mnt_wealth_estate_key }
	}

	remove_from_variable_map = { name = mnt_wealth_invested_since_review key = local_var:mnt_wealth_estate_key }
}

# Gives each remaining estate its power-weighted share of what the departed estates freed, debt included.

mnt_wealth_redistribute_to_estate = {
	# The cache was built by the census that is calling, so it is the live figure.
	set_local_variable = {
		name = mnt_wealth_relative_power
		value = "variable_map(mnt_wealth_cached_local_relative_estate_power|local_var:mnt_wealth_estate_key)"
	}

	if = {
		limit = { local_var:mnt_wealth_freed_investment != 0 }

		set_local_variable = {
			name = mnt_wealth_investment_share
			value = { value = local_var:mnt_wealth_freed_investment multiply = local_var:mnt_wealth_relative_power }
		}

		set_local_variable = { name = mnt_wealth_existing_wealth value = 0 }

		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = local_var:mnt_wealth_estate_key } }
			set_local_variable = {
				name = mnt_wealth_existing_wealth
				value = "variable_map(mnt_wealth_invested_wealth|local_var:mnt_wealth_estate_key)"
			}
		}

		set_local_variable = {
			name = mnt_wealth_new_invested_wealth
			value = { value = local_var:mnt_wealth_existing_wealth add = local_var:mnt_wealth_investment_share }
		}

		add_to_variable_map = {
			name = mnt_wealth_invested_wealth
			key = local_var:mnt_wealth_estate_key
			value = local_var:mnt_wealth_new_invested_wealth
		}
	}

	if = {
		limit = { local_var:mnt_wealth_freed_estate_wealth != 0 }
		set_local_variable = { name = mnt_wealth_existing_gold value = 0 }

		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
			set_local_variable = {
				name = mnt_wealth_existing_gold
				value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
			}
		}

		set_local_variable = {
			name = mnt_wealth_new_estate_gold
			value = {
				value = local_var:mnt_wealth_existing_gold
				add = { value = local_var:mnt_wealth_freed_estate_wealth multiply = local_var:mnt_wealth_relative_power }
			}
		}

		add_to_variable_map = {
			name = mnt_wealth_estate_wealth
			key = local_var:mnt_wealth_estate_key
			value = local_var:mnt_wealth_new_estate_gold
		}
	}
}
