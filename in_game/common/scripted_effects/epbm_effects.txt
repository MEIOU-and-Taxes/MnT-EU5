epbm_init_modifier_scale = {
	set_variable = { name = epbm_mod_$estate$ value = 0 }
}

epbm_snapshot_mod = {
	if = {
		limit = { has_variable = epbm_mod_$estate$ }
		set_variable = { name = epbm_show_mod_$estate$ value = var:epbm_mod_$estate$ }
	}
}

# Sets the auto_modifier scale for each estate's building_upkeep_multiplier reduction
epbm_compute_modifier_scale = {
	set_local_variable = { name = est value = estate_type:$estate$_estate }

	if = {
		limit = { country_has_estate = local_var:est }
		set_variable = {
			name = epbm_mod_$estate$
			value = {
				value = "estate_power(local_var:est)"
				multiply = epbm_cost_share
				multiply = epbm_inflation_factor
			}
		}
	}
	else = { set_variable = { name = epbm_mod_$estate$ value = 0 } }
}

# Variable map operations. All use every_estate with estate
# scope as map key. Scope: country.

epbm_split_by_estate_power_dynamic = {
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_est

		prev = {
			set_local_variable = {
				name = epbm_est_share
				value = { value = local_var:epbm_total_cost multiply = "estate_power(scope:epbm_est.estate_type)" }
			}

			add_to_variable_map = {
				name = epbm_pay
				key = scope:epbm_est
				value = local_var:epbm_est_share
			}
			add_to_variable_map = {
				name = epbm_pool
				key = scope:epbm_est
				value = local_var:epbm_est_share
			}
			# Fresh base each cycle; estates with building costs are
			# overwritten by epbm_add_estate_building_costs_dynamic.
			add_to_variable_map = {
				name = epbm_assigned
				key = scope:epbm_est
				value = 0
			}
		}
	}
}

epbm_zero_pay_dynamic = {
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_est

		prev = {
			add_to_variable_map = {
				name = epbm_pay
				key = scope:epbm_est
				value = 0
			}
			add_to_variable_map = {
				name = epbm_pool
				key = scope:epbm_est
				value = 0
			}
			add_to_variable_map = {
				name = epbm_assigned
				key = scope:epbm_est
				value = 0
			}
		}
	}
}

epbm_add_estate_building_costs_dynamic = {
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_est

		prev = {
			if = {
				limit = {
					has_variable_map = epbm_estate_costs
					is_key_in_variable_map = { name = epbm_estate_costs target = scope:epbm_est }
				}

				set_local_variable = { name = epbm_est_raw value = "variable_map(epbm_estate_costs|scope:epbm_est)" }

				if = {
					limit = { local_var:epbm_est_raw > 0 }

					set_local_variable = {
						name = epbm_est_scaled
						value = { value = local_var:epbm_est_raw multiply = var:epbm_charge_factor }
					}

					add_to_variable_map = {
						name = epbm_assigned
						key = scope:epbm_est
						value = local_var:epbm_est_scaled
					}

					if = {
						limit = { has_variable_map = epbm_pay is_key_in_variable_map = { name = epbm_pay target = scope:epbm_est } }
						set_local_variable = { name = epbm_existing_pay value = "variable_map(epbm_pay|scope:epbm_est)" }
					}
					else = { set_local_variable = { name = epbm_existing_pay value = 0 } }

					set_local_variable = {
						name = epbm_est_total_pay
						value = { value = local_var:epbm_existing_pay add = local_var:epbm_est_scaled }
					}

					add_to_variable_map = {
						name = epbm_pay
						key = scope:epbm_est
						value = local_var:epbm_est_total_pay
					}
				}
			}
		}
	}
}

epbm_snapshot_estates_dynamic = {
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_est

		prev = {
			if = {
				limit = { has_variable_map = epbm_show is_key_in_variable_map = { name = epbm_show target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_prev
					key = scope:epbm_est
					value = "variable_map(epbm_show|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_show_pool is_key_in_variable_map = { name = epbm_show_pool target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_prev_pool
					key = scope:epbm_est
					value = "variable_map(epbm_show_pool|scope:epbm_est)"
				}
			}

			if = {
				limit = {
					has_variable_map = epbm_show_assigned
					is_key_in_variable_map = { name = epbm_show_assigned target = scope:epbm_est }
				}
				add_to_variable_map = {
					name = epbm_prev_assigned
					key = scope:epbm_est
					value = "variable_map(epbm_show_assigned|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_pay is_key_in_variable_map = { name = epbm_pay target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_show
					key = scope:epbm_est
					value = "variable_map(epbm_pay|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_pool is_key_in_variable_map = { name = epbm_pool target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_show_pool
					key = scope:epbm_est
					value = "variable_map(epbm_pool|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_assigned is_key_in_variable_map = { name = epbm_assigned target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_show_assigned
					key = scope:epbm_est
					value = "variable_map(epbm_assigned|scope:epbm_est)"
				}
			}
		}
	}
}

@epbm_rebuild_version = 500

	epbm_init_blacklist = yes
	epbm_full_rebuild = yes

	set_global_variable = {
		name = epbm_version
		value = @epbm_rebuild_version
		days = 32
	}
}

epbm_global_monthly = {
	if = {
		limit = {
			OR = { NOT = { has_global_variable = epbm_version } global_var:epbm_version < @epbm_rebuild_version }
		}
		epbm_full_rebuild = yes
	}

	set_global_variable = {
		name = epbm_version
		value = @epbm_rebuild_version
		days = 32
	}

	# Market price caches are cleared by mnt_monthly_instant_market_pulse.
}
epbm_full_rebuild = {
	epbm_init_blacklist = yes
	every_country = { limit = { is_real_country = yes } epbm_for_each_estate = { effect = epbm_init_modifier_scale } }

	# Seed location-level costs for every AI-owned location so the
	# aggregation has data immediately instead of waiting a full sweep cycle.
	every_ownable_location = {
		limit = { exists = owner owner = { is_ai = yes } }
		epbm_price_location_buildings = yes
	}

	# AI countries: price estate and foreign buildings, aggregate, charge,
	# then copy mod scales to show so AI gets upkeep refund from game start.
	every_country = {
		limit = { is_real_country = yes is_ai = yes any_active_estate = { exists = yes } }
		epbm_calculate_estate_and_foreign = yes
		epbm_aggregate_maintenance = yes
		epbm_charge_estates = yes
		epbm_for_each_estate = { effect = epbm_snapshot_mod }
	}

	# Player countries: the full three-loop calculation and immediate
	# charge, then zero pay so the first monthly snapshot captures it.
	every_country = {
		limit = { is_real_country = yes is_ai = no any_active_estate = { exists = yes } }
		epbm_calculate_maintenance = yes
		epbm_charge_estates = yes
		epbm_zero_pay_dynamic = yes
	}
}
