# Variable map operations. Scope: country. Charge maps are keyed by the
# estate. The upkeep scale maps are keyed by estate_type so the auto_modifier
# script values can read them with a bare estate_type key.

epbm_split_by_estate_power_dynamic = {
	# Rebuilt every cycle so an estate the country no longer has drops out.
	if = { limit = { has_variable_map = epbm_upkeep_scale } clear_variable_map = epbm_upkeep_scale }

	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_est

		prev = {
			set_local_variable = {
				name = epbm_est_share
				value = { value = local_var:epbm_total_cost multiply = "estate_power(scope:epbm_est.estate_type)" }
			}

			# Scale for this estate's building_upkeep_multiplier refund.
			set_local_variable = {
				name = epbm_est_upkeep_scale
				value = { value = "estate_power(scope:epbm_est.estate_type)" multiply = var:epbm_charge_factor }
			}
			add_to_variable_map = {
				name = epbm_upkeep_scale
				key = scope:epbm_est.estate_type
				value = local_var:epbm_est_upkeep_scale
			}

			add_to_variable_map = {
				name = epbm_maintenance_charge
				key = scope:epbm_est
				value = local_var:epbm_est_share
			}
			add_to_variable_map = {
				name = epbm_shared_charge
				key = scope:epbm_est
				value = local_var:epbm_est_share
			}
			# Fresh base each cycle; estates with building costs are
			# overwritten by epbm_add_estate_building_costs_dynamic.
			add_to_variable_map = {
				name = epbm_estate_building_charge
				key = scope:epbm_est
				value = 0
			}
		}
	}
}

epbm_add_estate_building_costs_dynamic = {
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_est

		prev = {
			if = {
				limit = {
					has_variable_map = epbm_estate_costs
					is_key_in_variable_map = { name = epbm_estate_costs target = scope:epbm_est }
				}

				set_local_variable = { name = epbm_est_raw value = "variable_map(epbm_estate_costs|scope:epbm_est)" }

				if = {
					limit = { local_var:epbm_est_raw > 0 }

					set_local_variable = {
						name = epbm_est_scaled
						value = { value = local_var:epbm_est_raw multiply = var:epbm_charge_factor }
					}

					add_to_variable_map = {
						name = epbm_estate_building_charge
						key = scope:epbm_est
						value = local_var:epbm_est_scaled
					}

					if = {
						limit = { has_variable_map = epbm_maintenance_charge is_key_in_variable_map = { name = epbm_maintenance_charge target = scope:epbm_est } }
						set_local_variable = { name = epbm_existing_pay value = "variable_map(epbm_maintenance_charge|scope:epbm_est)" }
					}
					else = { set_local_variable = { name = epbm_existing_pay value = 0 } }

					set_local_variable = {
						name = epbm_est_total_pay
						value = { value = local_var:epbm_existing_pay add = local_var:epbm_est_scaled }
					}

					add_to_variable_map = {
						name = epbm_maintenance_charge
						key = scope:epbm_est
						value = local_var:epbm_est_total_pay
					}
				}
			}
		}
	}
}

epbm_snapshot_estates_dynamic = {
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_est

		prev = {
			if = {
				limit = { has_variable_map = epbm_maintenance_charged_last_month is_key_in_variable_map = { name = epbm_maintenance_charged_last_month target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_maintenance_charged_two_months_ago
					key = scope:epbm_est
					value = "variable_map(epbm_maintenance_charged_last_month|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_shared_charged_last_month is_key_in_variable_map = { name = epbm_shared_charged_last_month target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_shared_charged_two_months_ago
					key = scope:epbm_est
					value = "variable_map(epbm_shared_charged_last_month|scope:epbm_est)"
				}
			}

			if = {
				limit = {
					has_variable_map = epbm_estate_building_charged_last_month
					is_key_in_variable_map = { name = epbm_estate_building_charged_last_month target = scope:epbm_est }
				}
				add_to_variable_map = {
					name = epbm_estate_building_charged_two_months_ago
					key = scope:epbm_est
					value = "variable_map(epbm_estate_building_charged_last_month|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_maintenance_charge is_key_in_variable_map = { name = epbm_maintenance_charge target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_maintenance_charged_last_month
					key = scope:epbm_est
					value = "variable_map(epbm_maintenance_charge|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_shared_charge is_key_in_variable_map = { name = epbm_shared_charge target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_shared_charged_last_month
					key = scope:epbm_est
					value = "variable_map(epbm_shared_charge|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_estate_building_charge is_key_in_variable_map = { name = epbm_estate_building_charge target = scope:epbm_est } }
				add_to_variable_map = {
					name = epbm_estate_building_charged_last_month
					key = scope:epbm_est
					value = "variable_map(epbm_estate_building_charge|scope:epbm_est)"
				}
			}

			if = {
				limit = { has_variable_map = epbm_upkeep_scale is_key_in_variable_map = { name = epbm_upkeep_scale target = scope:epbm_est.estate_type } }
				add_to_variable_map = {
					name = epbm_upkeep_scale_last_month
					key = scope:epbm_est.estate_type
					value = "variable_map(epbm_upkeep_scale|scope:epbm_est.estate_type)"
				}
			}
		}
	}
}

@epbm_rebuild_version = 500
epbm_initialize = {
	epbm_init_blacklist = yes
	epbm_full_rebuild = yes

	# 32-day TTL: expires if the mod is removed, triggering rebuild on re-add
	set_global_variable = {
		name = epbm_version
		value = @epbm_rebuild_version
		days = 32
	}
}

epbm_global_monthly = {
	if = {
		limit = {
			OR = { NOT = { has_global_variable = epbm_version } global_var:epbm_version < @epbm_rebuild_version }
		}
		epbm_full_rebuild = yes
	}

	set_global_variable = {
		name = epbm_version
		value = @epbm_rebuild_version
		days = 32
	}

	# Market price caches are cleared by mnt_monthly_instant_market_pulse.
}
epbm_full_rebuild = {
	epbm_init_blacklist = yes

	# Seed location-level costs for every AI-owned location so the
	# aggregation has data immediately instead of waiting a full sweep cycle.
	every_ownable_location = {
		limit = { exists = owner owner = { is_ai = yes } }
		epbm_price_location_buildings = yes
	}

	# Compute only. No country is gated on owning land: building-based
	# countries hold foreign buildings. The first monthly settle charges.
	every_country = {
		limit = { is_ai = yes any_active_estate = { exists = yes } }
		epbm_calculate_estate_and_foreign = yes
		epbm_aggregate_maintenance = yes
	}

	every_country = {
		limit = { is_ai = no any_active_estate = { exists = yes } }
		epbm_calculate_maintenance = yes
	}
}
