# Scope: none. Fills every country's raw power map and gdp variable again from its own
# locations, then audits so the log shows the result.

mnt_wealth_recache_all_national_estate_raw_power = {
	every_landed_country = { mnt_wealth_rebuild_national_estate_raw_power = yes }
	mnt_wealth_verify_national_gross_output = yes
}

# Scope: none. Console effect. Compares maintained gross output totals against a fresh
# recount and audits national raw power per estate. Logs every diverging country.

mnt_wealth_verify_national_gross_output = {
	set_global_variable = { name = mnt_wealth_verify_over_count        value = 0 }
	set_global_variable = { name = mnt_wealth_verify_under_count       value = 0 }
	set_global_variable = { name = mnt_wealth_verify_rp_over_count     value = 0 }
	set_global_variable = { name = mnt_wealth_verify_rp_under_count    value = 0 }
	set_global_variable = { name = mnt_wealth_verify_seizure_neg_count value = 0 }

	every_landed_country = {
		limit = { has_variable = mnt_wealth_national_gross_output }
		set_local_variable = { name = mnt_wealth_maintained value = var:mnt_wealth_national_gross_output }
		set_local_variable = { name = mnt_wealth_expected   value = mnt_wealth_owned_local_gross_output_sum }
		set_local_variable = { name = mnt_wealth_diff       value = local_var:mnt_wealth_maintained }
		change_local_variable = { name = mnt_wealth_diff subtract = local_var:mnt_wealth_expected }
		# Small tolerance ignores floating-point drift.
		if = {
			limit = { local_var:mnt_wealth_diff > .05 }
			change_global_variable = { name = mnt_wealth_verify_over_count add = 1 }
			debug_log = "MNT_WEALTH-DIVERGE over gdp=[SCOPE.GetLocalVariable('mnt_wealth_maintained').GetValue|2] expected=[SCOPE.GetLocalVariable('mnt_wealth_expected').GetValue|2]"
		}
		else_if = {
			limit = { local_var:mnt_wealth_diff < -.05 }
			change_global_variable = { name = mnt_wealth_verify_under_count add = 1 }
			debug_log = "MNT_WEALTH-DIVERGE under gdp=[SCOPE.GetLocalVariable('mnt_wealth_maintained').GetValue|2] expected=[SCOPE.GetLocalVariable('mnt_wealth_expected').GetValue|2]"
		}

		# Raw power audit: each estate's maintained national raw power against a
		# fresh recount across this country's locations. Any divergence is logged
		# the month it appears, with the country's permanent id.
		save_temporary_scope_as = mnt_wealth_audit_country
		set_local_variable = { name = mnt_wealth_rp_country_id value = var:mnt_wealth_country_id }

		every_key_in_variable_map = {
			variable = mnt_wealth_national_estate_raw_power
			save_temporary_scope_as = mnt_wealth_audit_estate

			scope:mnt_wealth_audit_country = {
				set_local_variable = {
					name = mnt_wealth_rp_maintained
					value = "variable_map(mnt_wealth_national_estate_raw_power|scope:mnt_wealth_audit_estate)"
				}
				set_local_variable = { name = mnt_wealth_rp_recount value = mnt_wealth_owned_raw_power_sum }
				set_local_variable = { name = mnt_wealth_rp_diff    value = local_var:mnt_wealth_rp_maintained }
				change_local_variable = { name = mnt_wealth_rp_diff subtract = local_var:mnt_wealth_rp_recount }

				if = {
					limit = { local_var:mnt_wealth_rp_diff > 1 }
					change_global_variable = { name = mnt_wealth_verify_rp_over_count add = 1 }
					debug_log = "MNT_WEALTH-RAWPOWER-DIVERGE over id=[SCOPE.GetLocalVariable('mnt_wealth_rp_country_id').GetValue|0] maintained=[SCOPE.GetLocalVariable('mnt_wealth_rp_maintained').GetValue|2] recount=[SCOPE.GetLocalVariable('mnt_wealth_rp_recount').GetValue|2]"
				}
				else_if = {
					limit = { local_var:mnt_wealth_rp_diff < -1 }
					change_global_variable = { name = mnt_wealth_verify_rp_under_count add = 1 }
					debug_log = "MNT_WEALTH-RAWPOWER-DIVERGE under id=[SCOPE.GetLocalVariable('mnt_wealth_rp_country_id').GetValue|0] maintained=[SCOPE.GetLocalVariable('mnt_wealth_rp_maintained').GetValue|2] recount=[SCOPE.GetLocalVariable('mnt_wealth_rp_recount').GetValue|2]"
				}
			}
		}

		# Seizure demand only accumulates and decays, so any negative value
		# means the decay underflowed or a subtraction went wrong.
		scope:mnt_wealth_audit_country = {
			set_local_variable = { name = mnt_wealth_seizure_negative_count value = 0 }
			every_key_in_variable_map = {
				variable = mnt_wealth_estate_seizure_demand
				save_temporary_scope_as = mnt_wealth_seizure_audit_key

				scope:mnt_wealth_audit_country = {
					set_local_variable = {
						name = mnt_wealth_seizure_val
						value = "variable_map(mnt_wealth_estate_seizure_demand|scope:mnt_wealth_seizure_audit_key)"
					}
					if = {
						limit = { local_var:mnt_wealth_seizure_val < 0 }
						change_local_variable  = { name = mnt_wealth_seizure_negative_count   add = 1 }
						change_global_variable = { name = mnt_wealth_verify_seizure_neg_count add = 1 }
						error_log = "MNT_WEALTH-INVARIANT negative seizure demand id=[SCOPE.GetLocalVariable('mnt_wealth_rp_country_id').GetValue|0] value=[SCOPE.GetLocalVariable('mnt_wealth_seizure_val').GetValue|4]"
					}
				}
			}
		}
	}

	# A log line cannot read a global directly, so copy the tallies into locals.
	set_local_variable = { name = mnt_wealth_over  value = global_var:mnt_wealth_verify_over_count }
	set_local_variable = { name = mnt_wealth_under value = global_var:mnt_wealth_verify_under_count }
	debug_log = "MNT_WEALTH-ACCOUNTING diverging over=[SCOPE.GetLocalVariable('mnt_wealth_over').GetValue|0] under=[SCOPE.GetLocalVariable('mnt_wealth_under').GetValue|0]"
	set_local_variable = { name = mnt_wealth_rp_over  value = global_var:mnt_wealth_verify_rp_over_count }
	set_local_variable = { name = mnt_wealth_rp_under value = global_var:mnt_wealth_verify_rp_under_count }
	debug_log = "MNT_WEALTH-RAWPOWER diverging over=[SCOPE.GetLocalVariable('mnt_wealth_rp_over').GetValue|0] under=[SCOPE.GetLocalVariable('mnt_wealth_rp_under').GetValue|0]"
	set_local_variable = { name = mnt_wealth_seizure_neg value = global_var:mnt_wealth_verify_seizure_neg_count }
	debug_log = "MNT_WEALTH-MAPAUDIT negative_seizure=[SCOPE.GetLocalVariable('mnt_wealth_seizure_neg').GetValue|0]"
}

# Scope: none. Console effect. Checks each owned location's internal
# consistency: pool sign, wealth share totals, desired wealth totals, and
# yearly-pass coverage. Each check tallies failures in a global counter,
# logged once at the end.

mnt_wealth_verify_locations = {
	set_global_variable = { name = mnt_wealth_locaudit_pool_fail    value = 0 }
	set_global_variable = { name = mnt_wealth_locaudit_map_diff     value = 0 }
	set_global_variable = { name = mnt_wealth_locaudit_stale_fail   value = 0 }

	every_ownable_location = {
		limit = { exists = owner }
		save_temporary_scope_as = mnt_wealth_locaudit_location
		set_local_variable = { name = mnt_wealth_locaudit_loc_id value = var:mnt_wealth_location_id }

		# Pool must never go negative.
		if = {
			limit = { var:mnt_wealth_pool < -0.01 }
			change_global_variable = { name = mnt_wealth_locaudit_pool_fail add = 1 }
			set_local_variable = { name = mnt_wealth_locaudit_pool_val value = var:mnt_wealth_pool }
			error_log = "MNT_WEALTH-INVARIANT negative wealth pool loc_id=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_loc_id').GetValue|0] pool=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_pool_val').GetValue|2]"
		}

		# Invariant: pool must equal the sum of mnt_wealth_estate_wealth.
		set_local_variable = { name = mnt_wealth_locaudit_map_sum value = 0 }

		every_key_in_variable_map = {
			variable = mnt_wealth_estate_wealth
			save_temporary_scope_as = mnt_wealth_locaudit_map_key

			scope:mnt_wealth_locaudit_location = {
				change_local_variable = {
					name = mnt_wealth_locaudit_map_sum
					add = "variable_map(mnt_wealth_estate_wealth|scope:mnt_wealth_locaudit_map_key)"
				}
			}
		}

		set_local_variable = { name = mnt_wealth_locaudit_pool_stored value = var:mnt_wealth_pool }
		set_local_variable = {
			name = mnt_wealth_locaudit_map_delta
			value = { value = local_var:mnt_wealth_locaudit_map_sum subtract = local_var:mnt_wealth_locaudit_pool_stored }
		}

		if = {
			limit = { local_var:mnt_wealth_locaudit_map_delta > 0.01 }
			change_global_variable = { name = mnt_wealth_locaudit_map_diff add = 1 }
			error_log = "MNT_WEALTH-INVARIANT pool != map sum loc_id=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_loc_id').GetValue|0] map_sum=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_map_sum').GetValue|4] pool=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_pool_stored').GetValue|4]"
		}
		else_if = {
			limit = { local_var:mnt_wealth_locaudit_map_delta < -0.01 }
			change_global_variable = { name = mnt_wealth_locaudit_map_diff add = 1 }
			error_log = "MNT_WEALTH-INVARIANT pool != map sum loc_id=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_loc_id').GetValue|0] map_sum=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_map_sum').GetValue|4] pool=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_pool_stored').GetValue|4]"
		}

		# Yearly-pass coverage.
		if = {
			limit = { owner ?= { is_ai = yes } }
			set_local_variable = { name = mnt_wealth_locaudit_now    value = mnt_wealth_months_since_campaign_start }
			set_local_variable = { name = mnt_wealth_locaudit_review value = var:mnt_wealth_review_month }
			set_local_variable = {
				name = mnt_wealth_locaudit_age
				value = { value = local_var:mnt_wealth_locaudit_now subtract = local_var:mnt_wealth_locaudit_review }
			}

			if = {
				limit = { local_var:mnt_wealth_locaudit_age > 13 }
				change_global_variable = { name = mnt_wealth_locaudit_stale_fail add = 1 }
				error_log = "MNT_WEALTH-INVARIANT stale location loc_id=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_loc_id').GetValue|0] months_since_review=[SCOPE.GetLocalVariable('mnt_wealth_locaudit_age').GetValue|0]"
			}
		}
	}

	set_local_variable = { name = mnt_wealth_la_pool    value = global_var:mnt_wealth_locaudit_pool_fail }
	set_local_variable = { name = mnt_wealth_la_map_d   value = global_var:mnt_wealth_locaudit_map_diff }
	set_local_variable = { name = mnt_wealth_la_stale   value = global_var:mnt_wealth_locaudit_stale_fail }
	debug_log = "MNT_WEALTH-LOCAUDIT pool_fail=[SCOPE.GetLocalVariable('mnt_wealth_la_pool').GetValue|0] map_diff=[SCOPE.GetLocalVariable('mnt_wealth_la_map_d').GetValue|0] stale_fail=[SCOPE.GetLocalVariable('mnt_wealth_la_stale').GetValue|0]"
}
