print_regional_prices = {
	every_goods = {
		save_scope_as = good_curr
		every_region = {
			set_local_variable = {
				name = avg_cost
				value = {
					every_location_in_region = { limit = { is_land = yes }
						add = "market.market_price(scope:good_curr)"
					}
					divide = location_counter
				}

			}
			if = { limit = { location_counter > 0 }
				error_log = "::GP::[GetCurrentYear]:[PREV.GetGoods.GetNameWithNoTooltip]:[THIS.GetRegion.GetNameWithNoTooltip]:[SLV('avg_cost')]"
			}
		}
	}
	remove_local_variable = avg_cost
}

}

print_regional_roads = {
	every_road_type = {
		if = { limit = {
				OR = {
					this = road_type:gravel_road
					AND = {
						this = road_type:paved_road
						current_age_or_later = { age = age_3_discovery }
					}
					AND = {
						this = road_type:modern_road
						current_age_or_later = { age = age_5_absolutism }
					}
					AND = {
						this = road_type:railroad
						current_age_or_later = { age = age_6_revolutions }
					}
				}
			}

			set_local_variable = { name = road_curr value = this }
			every_region = {
				set_local_variable = { name = no_locations value = 0 }
				set_local_variable = { name = no_locations_with_road value = 0 }
				every_ownable_location_in_region = {
					change_local_variable = { name = no_locations add = 1 }

					if = { limit = { has_local_variable = loc_has_road }
						remove_local_variable = loc_has_road
					}

					every_neighbor_location = {
						if = { limit = {
								has_road_of_type_to = {
									target = prev
									type = local_var:road_curr
								}
							}
							set_local_variable = loc_has_road
						}
					}
					if = { limit = { has_local_variable = loc_has_road }
						change_local_variable = { name = no_locations_with_road add = 1 }
					}
				}
				if = { limit = { local_var:no_locations_with_road > 0 }
					error_log = "::RT::[GetCurrentYear]:[THIS.GetRegion.GetNameWithNoTooltip]:[LocalVar('road_curr').GetRoadType.GetNameWithNoTooltip]:[SLV('no_locations')]:[SLV('no_locations_with_road')]"
				}
			}
		}
	}

	remove_local_variable = road_curr
	remove_local_variable = no_locations
	remove_local_variable = no_locations_with_road
}
