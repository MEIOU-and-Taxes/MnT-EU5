# Load balancer: splits a DATA_SET index across CYCLE months and worker
# countries. Callers name only a DATA_SET; the balancer builds and refreshes
# the index on demand. A new index needs five names:
#
#   mnt_wealth_<DATA_SET>_index            - global variable map (key:int, value:scope)
#   mnt_wealth_<DATA_SET>_count            - entry count
#   mnt_wealth_<DATA_SET>_index_fresh      - expiring freshness flag
#   mnt_wealth_<DATA_SET>_index_lifetime   - macro: days the freshness flag survives
#   mnt_wealth_<DATA_SET>_index_accessor   - scripted effect: scopes to the cursor's entry
#   mnt_wealth_create_<DATA_SET>_index     - scripted effect: rebuilds the whole index

# ── MAIN DISPATCHER ──
# DATA_SET selects which index to sweep. Legal values:
#   location         - the global location index (mnt_wealth_location_index)
#   human_location   - the human-owned location index (mnt_wealth_human_location_index)
#   market           - the global market index (mnt_wealth_market_index)
# CYCLE is how many months a full sweep takes; up to @mnt_wealth_worker_cap
# countries work each pass. EFFECT runs once per entry in the worker's slice.
#
# EFFECT must never rewrite the index it is sweeping.

@mnt_wealth_worker_cap = 512

mnt_wealth_load_balance_this_effect = {
	mnt_wealth_load_balance_ensure_indices = { DATA_SET = $DATA_SET$ }

	if = {
		limit = { var:mnt_wealth_country_index < @mnt_wealth_worker_cap }
		mnt_wealth_load_balance_resolve_slice = { DATA_SET = $DATA_SET$ CYCLE = $CYCLE$ }

		if = {
			limit = { local_var:mnt_wealth_load_balance_remaining > 0 }
			while = { # Stepping by CYCLE keeps the cursor on this month's remainder.
				count = local_var:mnt_wealth_load_balance_remaining
				mnt_wealth_$DATA_SET$_index_accessor = { EFFECT = $EFFECT$ }
				change_local_variable = { name = mnt_wealth_load_balance_cursor add = $CYCLE$ }
			}
		}
	}
}

# ── SLICE RESOLUTION ──
# Remainder-based dispatch: work is done on entry i when i % CYCLE == month % CYCLE.
# Cycles that are multiples of one another can be resolved by the same load balancer

# Works out this worker's slice of this month's entries, saving values to
# local_var:mnt_wealth_load_balance_remaining and local_var:mnt_wealth_load_balance_cursor 
# so the main effect can properly split the work.

mnt_wealth_load_balance_resolve_slice = {
	set_local_variable = { name = mnt_wealth_index_size value = global_var:mnt_wealth_$DATA_SET$_count }

	set_local_variable = {
		name = mnt_wealth_load_balance_offset
		value = { value = global_var:mnt_wealth_months_elapsed modulo = $CYCLE$ }
	}

	# How many entries this month handles
	set_local_variable = {
		name = mnt_wealth_load_balance_monthly_count
		value = {
			value = local_var:mnt_wealth_index_size
			subtract = local_var:mnt_wealth_load_balance_offset
			divide = $CYCLE$
			ceiling = yes
			min = 0
		}
	}

	# Use at most the smaller of the worker cap and the country count.
	set_local_variable = {
		name = mnt_wealth_load_balance_worker_count
		value = { value = @mnt_wealth_worker_cap max = global_var:mnt_wealth_country_count }
	}

	# Split this month's entries evenly across workers
	set_local_variable = {
		name = mnt_wealth_locations_per_tick
		value = {
			value = local_var:mnt_wealth_load_balance_monthly_count
			divide = local_var:mnt_wealth_load_balance_worker_count
			ceiling = yes
		}
	}

	# Where this worker starts among this month's entries
	set_local_variable = {
		name = mnt_wealth_load_balance_seq_start
		value = {
			value = var:mnt_wealth_country_index
			multiply = local_var:mnt_wealth_locations_per_tick
		}
	}

	# How many entries the worker has left; clamps the tail safely.
	set_local_variable = {
		name = mnt_wealth_load_balance_remaining
		value = {
			value = local_var:mnt_wealth_load_balance_monthly_count
			subtract = local_var:mnt_wealth_load_balance_seq_start
			max = local_var:mnt_wealth_locations_per_tick
			min = 0
		}
	}

	set_local_variable = {
		name = mnt_wealth_load_balance_cursor
		value = {
			value = local_var:mnt_wealth_load_balance_seq_start
			multiply = $CYCLE$
			add = local_var:mnt_wealth_load_balance_offset
		}
	}
}

# ── FRESHNESS ──
# Builds whatever this sweep needs that is missing: the current month variable, 
# the country indices that pick the workers, and the index being used.

mnt_wealth_load_balance_ensure_indices = {
	if = {
		limit = { NOT = { has_global_variable = mnt_wealth_months_elapsed } }
		set_global_variable = {
			name = mnt_wealth_months_elapsed
			value = mnt_wealth_months_since_campaign_start
			days = mnt_wealth_month_lifetime
		}
	}
	if = {
		limit = { NOT = { has_global_variable = mnt_wealth_country_index_fresh } }
		mnt_wealth_build_index = { DATA_SET = country }
	}
	if = {
		limit = { NOT = { has_global_variable = mnt_wealth_$DATA_SET$_index_fresh } }
		mnt_wealth_build_index = { DATA_SET = $DATA_SET$ }
	}
}

# Builds one index and flags it for its own lifetime. Callers check the flag first.
mnt_wealth_build_index = {
	mnt_wealth_create_$DATA_SET$_index = yes
	set_global_variable = {
		name = mnt_wealth_$DATA_SET$_index_fresh
		value = 1
		days = mnt_wealth_$DATA_SET$_index_lifetime
	}
}

# ── INDEX CONSTRUCTION ──

# Rebuilt monthly; annexations and formations invalidate the previous index.
mnt_wealth_create_country_index = {
	set_global_variable = { name = mnt_wealth_country_count value = 0 }

	every_country = {
		set_variable = { name = mnt_wealth_country_index value = global_var:mnt_wealth_country_count }
		change_global_variable = { name = mnt_wealth_country_count add = 1 }
	}
}

# Built once, then maintained in place. A rebuild reshuffles slots, so clear ids first.
mnt_wealth_create_location_index = {
	clear_global_variable_map = mnt_wealth_location_index
	set_global_variable = { name = mnt_wealth_location_count value = 0 }

	every_ownable_location = {
		remove_variable = mnt_wealth_location_id
		if = { limit = { exists = owner } mnt_wealth_index_location = yes }
	}
}

# Rebuilt monthly to track country swaps and multiplayer joins.
mnt_wealth_create_human_location_index = {
	clear_global_variable_map = mnt_wealth_human_location_index
	set_global_variable = { name = mnt_wealth_human_location_count value = 0 }

	every_country = {
		limit = { is_human = yes }
		every_owned_location = {
			add_to_global_variable_map = {
				name = mnt_wealth_human_location_index
				key = global_var:mnt_wealth_human_location_count
				value = this
			}
			change_global_variable = { name = mnt_wealth_human_location_count add = 1 }
		}
	}
}

# Small set; cheaper to rebuild monthly than maintain.
mnt_wealth_create_market_index = {
	clear_global_variable_map = mnt_wealth_market_index
	set_global_variable = { name = mnt_wealth_market_count value = 0 }

	every_market_in_world = {
		add_to_global_variable_map = {
			name = mnt_wealth_market_index
			key = global_var:mnt_wealth_market_count
			value = this
		}
		change_global_variable = { name = mnt_wealth_market_count add = 1 }
	}
}

# ── INDEX ACCESSORS ──
# Each of these runs EFFECT on the entry the cursor is currently sitting on.

mnt_wealth_location_index_accessor = {
	"global_variable_map(mnt_wealth_location_index|local_var:mnt_wealth_load_balance_cursor)" = { $EFFECT$ = yes }
}

mnt_wealth_human_location_index_accessor = {
	"global_variable_map(mnt_wealth_human_location_index|local_var:mnt_wealth_load_balance_cursor)" = { $EFFECT$ = yes }
}

mnt_wealth_market_index_accessor = {
	"global_variable_map(mnt_wealth_market_index|local_var:mnt_wealth_load_balance_cursor)" = { $EFFECT$ = yes }
}

# ── LOCATION INDEX MAINTENANCE ──
# Add a location to the end of the index. Callers check it is not already in.

mnt_wealth_index_location = {
	if = {
		limit = { NOT = { has_global_variable = mnt_wealth_location_count } }
		set_global_variable = { name = mnt_wealth_location_count value = 0 }
	}

	set_variable = { name = mnt_wealth_location_id value = global_var:mnt_wealth_location_count }

	add_to_global_variable_map = {
		name = mnt_wealth_location_index
		key = global_var:mnt_wealth_location_count
		value = this
	}

	change_global_variable = { name = mnt_wealth_location_count add = 1 }
}

# Scope: location. Removes this location from the index by swapping
# the last entry into its slot, keeping the index dense.
# Callers check it is in the index.

mnt_wealth_unindex_location = {
	if = {
		limit = { has_variable = mnt_wealth_location_id }
		set_local_variable = { name = mnt_wealth_freed_slot value = var:mnt_wealth_location_id }
		remove_variable = mnt_wealth_location_id
		change_global_variable = { name = mnt_wealth_location_count subtract = 1 }

		if = {
			limit = { local_var:mnt_wealth_freed_slot < global_var:mnt_wealth_location_count }
			"global_variable_map(mnt_wealth_location_index|global_var:mnt_wealth_location_count)" = {
				set_variable = { name = mnt_wealth_location_id value = local_var:mnt_wealth_freed_slot }
				add_to_global_variable_map = {
					name = mnt_wealth_location_index
					key = local_var:mnt_wealth_freed_slot
					value = this
				}
			}
			# Stale-location tracer: A location that gets moved into the middle of the index can miss its turn, this can log when that happens
			# set_local_variable = { name = mnt_wealth_log_move_from  value = global_var:mnt_wealth_location_count }
			# set_local_variable = { name = mnt_wealth_log_move_month value = mnt_wealth_months_since_campaign_start }
			# debug_log = "MNT_WEALTH-INDEXMOVE from=[SCOPE.GetLocalVariable('mnt_wealth_log_move_from').GetValue|0] to=[SCOPE.GetLocalVariable('mnt_wealth_freed_slot').GetValue|0] month=[SCOPE.GetLocalVariable('mnt_wealth_log_move_month').GetValue|0]"
		}
	}
}
