# Load balancer: sweeps a DATA_SET's batch once a month.
# The batch is rebuilt at the head of each sweep by the scheduler in
# mnt_wealth_schedule.txt and holds only the entries that are ready, so every
# entry handed out is worked. A new DATA_SET needs four names:
#
#   mnt_wealth_<DATA_SET>_batch           - global variable map (key:int, value:scope)
#   mnt_wealth_<DATA_SET>_batch_count     - entry count
#   mnt_wealth_<DATA_SET>_batch_accessor  - scripted effect: scopes to the cursor's entry
#   mnt_wealth_build_<DATA_SET>_batch     - scripted effect: rebuilds the batch

# ── MAIN DISPATCHER ──
# Scope: none. DATA_SET selects which batch to sweep. Legal values:
#   location  - locations that are ready (mnt_wealth_location_batch)
#   market    - markets that are ready (mnt_wealth_market_batch)
# EFFECT runs once per entry.
#
# EFFECT must never rewrite the batch it is sweeping.

mnt_wealth_load_balance_this_effect = {
	mnt_wealth_build_$DATA_SET$_batch = yes

	set_local_variable = { name = mnt_wealth_load_balance_cursor value = 0 }
	set_local_variable = {
		name = mnt_wealth_load_balance_count
		value = global_var:mnt_wealth_$DATA_SET$_batch_count
	}

	if = {
		limit = { local_var:mnt_wealth_load_balance_count > 0 }
		while = {
			count = local_var:mnt_wealth_load_balance_count
			mnt_wealth_$DATA_SET$_batch_accessor = { EFFECT = $EFFECT$ }
			change_local_variable = { name = mnt_wealth_load_balance_cursor add = 1 }
		}
	}
}

# ── LANE ROUTING ──
# Every batch entry fires both lanes. What each lane does is decided by the
# effects registered on its on_action, not here.
#
# Share of the spread lane's fires that land on the month tick instead of a day
# tick. Raising it moves work off the days and onto the month without touching
# what either lane runs. At 1 the spread lane behaves like the instant one.

@mnt_wealth_month_tick_share = 0.1

# Scope: location. Run once per batch entry. The batch is ordered longest wait
# first, so the share that skips the delay is the most overdue. Player-owned
# locations also skip it, since their monthly income cannot land in the wrong
# month.

mnt_wealth_route_location = {
	trigger_event_silently = { on_action = mnt_instant_location_pulse }

	set_local_variable = {
		name = mnt_wealth_route_month_tick_count
		value = {
			value = global_var:mnt_wealth_location_batch_count
			multiply = @mnt_wealth_month_tick_share
			ceiling = yes
		}
	}

	set_local_variable = { name = mnt_wealth_route_day value = 0 }
	if = {
		limit = {
			local_var:mnt_wealth_load_balance_cursor >= local_var:mnt_wealth_route_month_tick_count
			NOT = { owner ?= { is_human = yes } }
		}
		set_local_variable = {
			name = mnt_wealth_route_day
			value = {
				value = var:mnt_wealth_location_id
				modulo = global_var:mnt_wealth_current_days_in_month
			}
		}
	}

	trigger_event_silently = {
		on_action = mnt_spread_location_pulse
		days = local_var:mnt_wealth_route_day
	}
}

# ── MONTH SETUP ──
# Scope: none. Runs at the head of the month, before any sweep reads the month,
# the day count or the country index.

mnt_wealth_begin_month = {
	set_global_variable = {
		name = mnt_wealth_months_elapsed
		value = mnt_wealth_months_since_campaign_start
	}
	mnt_wealth_set_days_in_month = yes
	mnt_wealth_create_country_index = yes
}

# Scope: none. Resolves this calendar month's length once, so the routing reads
# a plain global rather than the map. Thirty covers a month the map has no entry
# for, which keeps the delay bounded rather than dividing by nothing.

mnt_wealth_set_days_in_month = {
	set_local_variable = { name = mnt_wealth_calendar_month value = current_month }
	set_local_variable = { name = mnt_wealth_month_days value = 30 }

	if = {
		limit = {
			is_key_in_global_variable_map = {
				name = mnt_wealth_days_in_month
				target = local_var:mnt_wealth_calendar_month
			}
		}
		set_local_variable = {
			name = mnt_wealth_month_days
			value = "global_variable_map(mnt_wealth_days_in_month|local_var:mnt_wealth_calendar_month)"
		}
	}

	set_global_variable = {
		name = mnt_wealth_current_days_in_month
		value = local_var:mnt_wealth_month_days
	}
}

# ── COUNTRY INDEX ──
# Numbers every country so the food rotation and the pulse marker can stagger
# themselves. The sweeps do not read it.

mnt_wealth_create_country_index = {
	set_global_variable = { name = mnt_wealth_country_count value = 0 }

	every_country = {
		set_variable = { name = mnt_wealth_country_index value = global_var:mnt_wealth_country_count }
		change_global_variable = { name = mnt_wealth_country_count add = 1 }
	}
}

# ── BATCH ACCESSORS ──
# Each of these runs EFFECT on the entry the cursor is currently sitting on.

mnt_wealth_location_batch_accessor = {
	"global_variable_map(mnt_wealth_location_batch|local_var:mnt_wealth_load_balance_cursor)" = { $EFFECT$ = yes }
}

mnt_wealth_market_batch_accessor = {
	"global_variable_map(mnt_wealth_market_batch|local_var:mnt_wealth_load_balance_cursor)" = { $EFFECT$ = yes }
}
