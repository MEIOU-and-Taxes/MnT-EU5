# Scheduling and dispatch: which locations get serviced on which day.

mnt_wealth_begin_month = {
	set_global_variable = {
		name = mnt_wealth_months_elapsed
		value = mnt_wealth_months_since_campaign_start
	}
	mnt_wealth_set_days_in_month = yes
}

mnt_wealth_set_days_in_month = {
	set_local_variable = { name = mnt_wealth_calendar_month value = current_month }
	set_local_variable = { name = mnt_wealth_month_days value = 30 }

	if = {
		limit = {
			is_key_in_global_variable_map = {
				name = mnt_wealth_days_in_month
				target = local_var:mnt_wealth_calendar_month
			}
		}
		set_local_variable = {
			name = mnt_wealth_month_days
			value = "global_variable_map(mnt_wealth_days_in_month|local_var:mnt_wealth_calendar_month)"
		}
	}

	set_global_variable = {
		name = mnt_wealth_current_days_in_month
		value = local_var:mnt_wealth_month_days
	}
}

# Locations in the same bucket spread across the month's days by id.

mnt_wealth_dispatch_location = {
	set_local_variable = {
		name = mnt_wealth_dispatch_day
		value = {
			value = var:mnt_wealth_location_id
			divide = 12
			floor = yes
			modulo = global_var:mnt_wealth_current_days_in_month
		}
	}

	trigger_event_silently = {
		on_action = mnt_yearly_balanced_location_pulse
		days = local_var:mnt_wealth_dispatch_day
	}
}

mnt_wealth_ensure_location_id = {
	if = {
		limit = { NOT = { has_variable = mnt_wealth_location_id } }
		set_variable = { name = mnt_wealth_location_id value = global_var:mnt_wealth_location_serial }
		change_global_variable = { name = mnt_wealth_location_serial add = 1 }
	}
}

mnt_wealth_schedule_next_service = {

		value = { value = local_var:mnt_wealth_next_service modulo = 12 add = 1 }
	}

	# Owner-change requeue: pull the old entry when the bucket changes.
	if = {
		limit = {
			has_variable = mnt_wealth_service_bucket
			NOT = { var:mnt_wealth_service_bucket = local_var:mnt_wealth_service_bucket_index }
		}
		mnt_wealth_unschedule_location = yes
	}
	set_variable = { name = mnt_wealth_service_bucket value = local_var:mnt_wealth_service_bucket_index }

	save_temporary_scope_as = mnt_wealth_bucket_entity

	if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 1 }

mnt_wealth_load_last_service = {
	set_local_variable = { name = mnt_wealth_last_service value = 0 }
	set_local_variable = { name = mnt_wealth_first_service value = 1 }
	if = {
		limit = { is_key_in_global_variable_map = { name = mnt_wealth_location_last_service target = this } }
		set_local_variable = {
			name = mnt_wealth_last_service
			value = "global_variable_map(mnt_wealth_location_last_service|this)"
		}
		set_local_variable = { name = mnt_wealth_first_service value = 0 }
	}
}

# Game start: the last service is backdated by the location id, so the first year's services spread across twelve months.

mnt_wealth_seed_location_schedule = {
	mnt_wealth_ensure_location_id = yes
	set_local_variable = {
		name = mnt_wealth_last_service
		value = {
			value = var:mnt_wealth_location_id
			modulo = 12
			multiply = -1
		}
	}
	mnt_wealth_write_location_schedule = yes
}

# Records local mnt_wealth_last_service as the location's last service, picks its census cohort by id, and queues the next service.

mnt_wealth_write_location_schedule = {
	mnt_wealth_ensure_location_id = yes
	save_temporary_scope_as = mnt_wealth_seed_location
	add_to_global_variable_map = {
		name = mnt_wealth_location_last_service
		key = scope:mnt_wealth_seed_location
		value = local_var:mnt_wealth_last_service
	}

	# Spread the review and gross output census across cohorts.
	set_local_variable = {
		name = mnt_wealth_seed_cycle
		value = { value = var:mnt_wealth_location_id modulo = 10 }
	}
	set_variable = { name = mnt_wealth_service_cycle value = local_var:mnt_wealth_seed_cycle }

	mnt_wealth_write_location_next_service = yes
}

mnt_wealth_service_location = {
	mnt_wealth_ensure_location_id = yes

	# The owner's census frequency only matters for players: 0 fast, 1 default, 2 slow. Default when unset.
	set_local_variable = { name = mnt_wealth_is_player        value = 0 }
	set_local_variable = { name = mnt_wealth_census_frequency value = 1 }
	owner ?= {
		if = { limit = { is_human = yes } set_local_variable = { name = mnt_wealth_is_player value = 1 } }
		if = {
			limit = { has_variable = mnt_wealth_census_frequency }
			set_local_variable = { name = mnt_wealth_census_frequency value = var:mnt_wealth_census_frequency }
		}
	}

	mnt_wealth_load_last_service = yes
	set_local_variable = {
		name = mnt_wealth_service_age
		value = {
			value = global_var:mnt_wealth_months_elapsed
			subtract = local_var:mnt_wealth_last_service
		}
	}

	set_local_variable = { name = mnt_wealth_censused_full value = 0 }
	set_local_variable = { name = mnt_wealth_censused_yearly value = 0 }

	# The periodic ladder: only when a full year has passed. An owner-change
	# requeue arrives early, skips it, and keeps its old rhythm.
	if = {
		limit = { local_var:mnt_wealth_service_age >= 12 }

		change_variable = { name = mnt_wealth_service_cycle add = 1 }

		if = {
			limit = { NOT = { has_variable = mnt_wealth_review_backlog } }
			set_variable = { name = mnt_wealth_review_backlog value = 0 }
		}
		change_variable = { name = mnt_wealth_review_backlog add = local_var:mnt_wealth_service_age }

		# Which of the gross output census and the review run this service.
		# AI, and players on slow: the census every fifth service, the review every second.
		# Players on default: both every service. Players on fast: both every month from the human pulse, so neither runs here.
		# The first service runs both, except on fast, where the monthly pulse has the same month covered.
		set_local_variable = { name = mnt_wealth_census_phase value = { value = var:mnt_wealth_service_cycle modulo = 5 } }
		set_local_variable = { name = mnt_wealth_review_phase   value = { value = var:mnt_wealth_service_cycle modulo = 2 } }
		if = {
			limit = { OR = { local_var:mnt_wealth_is_player = 0 local_var:mnt_wealth_census_frequency = 2 } }
			if = { limit = { local_var:mnt_wealth_census_phase = 0 } set_local_variable = { name = mnt_wealth_censused_full   value = 1 } }
			if = { limit = { local_var:mnt_wealth_review_phase = 0 }   set_local_variable = { name = mnt_wealth_censused_yearly value = 1 } }
		}
		else_if = {
			limit = { local_var:mnt_wealth_census_frequency = 1 }
			set_local_variable = { name = mnt_wealth_censused_full   value = 1 }
			set_local_variable = { name = mnt_wealth_censused_yearly value = 1 }
		}
		if = {
			limit = {
				local_var:mnt_wealth_first_service = 1
				NOT = { local_var:mnt_wealth_census_frequency = 0 }
			}
			set_local_variable = { name = mnt_wealth_censused_full   value = 1 }
			set_local_variable = { name = mnt_wealth_censused_yearly value = 1 }
		}

		if = {
			limit = { local_var:mnt_wealth_censused_full = 1 }
			trigger_event_silently = { on_action = mnt_five_yearly_balanced_location_pulse }
		}
		if = {
			limit = { local_var:mnt_wealth_censused_yearly = 1 }
			set_variable = {
				name = mnt_wealth_service_months
				value = { value = var:mnt_wealth_review_backlog min = 1 }
			}
			set_variable = { name = mnt_wealth_review_backlog value = 0 }
			trigger_event_silently = { on_action = mnt_biyearly_balanced_location_pulse }
		}

		set_local_variable = { name = mnt_wealth_last_service value = global_var:mnt_wealth_months_elapsed }
		save_temporary_scope_as = mnt_wealth_service_entity
		add_to_global_variable_map = {
			name = mnt_wealth_location_last_service
			key = scope:mnt_wealth_service_entity
			value = local_var:mnt_wealth_last_service
		}
	}

	if = {
		limit = { local_var:mnt_wealth_censused_full = 1 }
		mnt_wealth_retake_estate_census = yes
	}
	else_if = {
		limit = { local_var:mnt_wealth_censused_yearly = 1 }
		mnt_wealth_refresh_relative_power = yes
	}

	if = {
		limit = { OR = { local_var:mnt_wealth_censused_full = 1 local_var:mnt_wealth_censused_yearly = 1 } }
		set_local_variable = { name = mnt_wealth_rebuild_location_estate_list value = yes }
	}

	if = {
		limit = { local_var:mnt_wealth_service_age >= 12 }
		calc_location_center_scores = yes
		epbm_price_location_buildings = yes
	}

	mnt_wealth_write_location_next_service = yes
}

# Fast census frequency: the owner wants the census and the review every month, so the human monthly pulse
# runs here what the yearly service would otherwise run, in the same order. Runs after the monthly pass, so the
# month's state is applied before the census changes the data the next month reads.

mnt_wealth_monthly_human_census = {
	set_variable = { name = mnt_wealth_service_months value = 1 }
	set_variable = { name = mnt_wealth_review_backlog value = 0 }
	trigger_event_silently = { on_action = mnt_five_yearly_balanced_location_pulse }
	trigger_event_silently = { on_action = mnt_biyearly_balanced_location_pulse }
	mnt_wealth_retake_estate_census = yes
}
