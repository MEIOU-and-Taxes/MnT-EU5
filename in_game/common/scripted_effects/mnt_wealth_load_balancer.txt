# Monthly dispatch: walk the current month's service bucket and fire the balanced
# service for every location whose service month has arrived. AI locations spread across the
# month's days by their id; player locations run on the month tick so income
# lands in the right month. Markets are few and run every month on the tick.

# Scope: location, scheduled for this month. Fires both lanes; each lane's
# effect keeps only the locations it owns.

mnt_wealth_dispatch_location = {
	# The spread day. Every location in this month's bucket shares an id mod
	# twelve, so their ids step by twelve. Dividing that step back out turns
	# them into consecutive numbers, which spread evenly over a month of any
	# length; the raw id would only ever reach five days of a thirty-day month.
	set_local_variable = {
		name = mnt_wealth_dispatch_day
		value = {
			value = var:mnt_wealth_location_id
			divide = 12
			floor = yes
			modulo = global_var:mnt_wealth_current_days_in_month
		}
	}

	trigger_event_silently = { on_action = mnt_monthly_instant_location_pulse }
	trigger_event_silently = {
		on_action = mnt_balanced_location_pulse
		days = local_var:mnt_wealth_dispatch_day
	}
}

# ── MONTH SETUP ──
# Scope: none. Runs at the head of the month, before any sweep reads the month
# or the day count.

mnt_wealth_begin_month = {
	set_global_variable = {
		name = mnt_wealth_months_elapsed
		value = mnt_wealth_months_since_campaign_start
	}
	mnt_wealth_set_days_in_month = yes
}

# Scope: none. Resolves this calendar month's length once, so the routing reads
# a plain global rather than the map. Thirty covers a month the map has no entry
# for, which keeps the delay bounded rather than dividing by nothing.

mnt_wealth_set_days_in_month = {
	set_local_variable = { name = mnt_wealth_calendar_month value = current_month }
	set_local_variable = { name = mnt_wealth_month_days value = 30 }

	if = {
		limit = {
			is_key_in_global_variable_map = {
				name = mnt_wealth_days_in_month
				target = local_var:mnt_wealth_calendar_month
			}
		}
		set_local_variable = {
			name = mnt_wealth_month_days
			value = "global_variable_map(mnt_wealth_days_in_month|local_var:mnt_wealth_calendar_month)"
		}
	}

	set_global_variable = {
		name = mnt_wealth_current_days_in_month
		value = local_var:mnt_wealth_month_days
	}
}
