# Three loops: the engine exposes estate, domestic, and foreign buildings
# through separate iterators that don't overlap. Estate buildings route
# costs to their owning estate via variable map; all other non-blacklisted
# buildings go into a shared pool split by estate power.
#
# The player path uses this effect directly every month. The AI path splits
# the work: location pricing runs on the balanced sweep, estate and foreign
# buildings are priced yearly, and the results are aggregated monthly by
# epbm_aggregate_maintenance before charging.

epbm_calculate_maintenance = {
	set_local_variable = { name = epbm_total_cost value = 0 }

	if = { limit = { has_variable_map = epbm_estate_costs } clear_variable_map = epbm_estate_costs }

	# Estate buildings: iterate per estate, accumulate cost per estate into variable map.
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_cur_estate
		set_local_variable = { name = epbm_cur_estate_cost value = 0 }

		every_building_owned_by_estate = {
			save_temporary_scope_as = epbm_bldg
			building_type ?= { save_temporary_scope_as = epbm_bt }
			location ?= { market ?= { save_temporary_scope_as = epbm_mkt } }

			epbm_price_building = yes

			if = {
				limit = { local_var:epbm_bldg_cost > 0 }
				change_local_variable = { name = epbm_cur_estate_cost add = local_var:epbm_bldg_cost }
			}
		}

		if = {
			limit = { local_var:epbm_cur_estate_cost > 0 }
			prev = {
				add_to_variable_map = {
					name = epbm_estate_costs
					key = scope:epbm_cur_estate
					value = local_var:epbm_cur_estate_cost
				}
			}
		}
	}

	# Non-estate buildings: all go to shared pool.
	every_owned_building = {
		limit = {
			is_opened = yes
			NOT = {
				building_type = { is_key_in_global_variable_map = { name = epbm_blacklist target = this } }
			}
			NOT = { building_category = estate_category }
		}

		save_temporary_scope_as = epbm_bldg
		building_type ?= { save_temporary_scope_as = epbm_bt }
		location ?= { market ?= { save_temporary_scope_as = epbm_mkt } }

		epbm_price_building = yes

		if = { limit = { local_var:epbm_bldg_cost > 0 } change_local_variable = { name = epbm_total_cost add = local_var:epbm_bldg_cost } }
	}

	every_owned_foreign_building = {
		limit = {
			is_opened = yes
			NOT = {
				building_type = { is_key_in_global_variable_map = { name = epbm_blacklist target = this } }
			}
			NOT = { building_category = estate_category }
		}

		save_temporary_scope_as = epbm_bldg
		building_type ?= { save_temporary_scope_as = epbm_bt }
		location ?= { market ?= { save_temporary_scope_as = epbm_mkt } }

		epbm_price_building = yes

		if = { limit = { local_var:epbm_bldg_cost > 0 } change_local_variable = { name = epbm_total_cost add = local_var:epbm_bldg_cost } }
	}

	epbm_split_and_scale_costs = yes
}

# Scope: country. The post-iteration phase shared by epbm_calculate_maintenance
# and epbm_aggregate_maintenance. Expects local_var:epbm_total_cost and a
# populated epbm_estate_costs variable map.

epbm_split_and_scale_costs = {
	set_variable = { name = epbm_charge_factor value = { value = epbm_cost_share multiply = epbm_inflation_factor } }

	change_local_variable = { name = epbm_total_cost multiply = var:epbm_charge_factor }

	epbm_split_by_estate_power_dynamic = yes
	epbm_add_estate_building_costs_dynamic = yes

	set_variable = {
		name = epbm_crown_pool_cost
		value = { value = local_var:epbm_total_cost multiply = "estate_power(estate_type:crown_estate)" }
	}
}

# Scope: location (AI-owned). Prices every non-estate, non-blacklisted,
# owner-domestic building in this location and stores the total as a
# persistent location variable. The balanced sweep calls this once per
# location per year; game-start seeding calls it once for every AI location.

epbm_price_location_buildings = {
	if = {
		limit = { owner = { is_ai = yes } }

		set_variable = { name = epbm_loc_cost value = 0 }

		every_buildings_in_location = {
			limit = {
				is_opened = yes
				is_building_owned_by = location.owner
				NOT = {
					building_type = { is_key_in_global_variable_map = { name = epbm_blacklist target = this } }
				}
				NOT = { building_category = estate_category }
			}

			save_temporary_scope_as = epbm_bldg
			building_type ?= { save_temporary_scope_as = epbm_bt }
			location ?= { market ?= { save_temporary_scope_as = epbm_mkt } }

			epbm_price_building = yes

			if = {
				limit = { local_var:epbm_bldg_cost > 0 }
				location = { change_variable = { name = epbm_loc_cost add = local_var:epbm_bldg_cost } }
			}
		}
	}
}

# Scope: country (AI). Monthly aggregation: sums the location-level costs
# written by the balanced sweep, adds the yearly foreign costs, then runs
# the shared post-iteration phase. The epbm_estate_costs map persists from
# the yearly pass and is read directly by the tail.

epbm_aggregate_maintenance = {
	set_local_variable = { name = epbm_total_cost value = 0 }

	every_owned_location = {
		if = {
			limit = { has_variable = epbm_loc_cost }
			change_local_variable = { name = epbm_total_cost add = var:epbm_loc_cost }
		}
	}

	if = {
		limit = { has_variable = epbm_foreign_cost }
		change_local_variable = { name = epbm_total_cost add = var:epbm_foreign_cost }
	}

	epbm_split_and_scale_costs = yes
}
epbm_price_building = {
	set_local_variable = { name = epbm_bldg_cost value = 0 }

	random_production_method_of_building = {
		limit = {
			any_required_goods = { count >= 1 }
			NOT = { exists = produced_goods }
			is_available_for = scope:epbm_bldg.owner
			is_allowed_for = scope:epbm_bldg.owner
			is_visible_for = scope:epbm_bldg.owner
		}

		save_temporary_scope_as = epbm_pm

		prev = {
			set_local_variable = { name = epbm_per_level value = 0 }

			# Check cache on market center
			scope:epbm_mkt = {
				location = {
					if = {
						limit = { has_variable_map = epbm_pm_cache is_key_in_variable_map = { name = epbm_pm_cache target = scope:epbm_pm } }
						set_local_variable = { name = epbm_per_level value = "variable_map(epbm_pm_cache|scope:epbm_pm)" }
					}
				}
			}

			# Cache miss: compute from building_goods_input
			if = {
				limit = { local_var:epbm_per_level <= 0 }

				scope:epbm_pm = {
					every_required_goods = {
						save_temporary_scope_as = epbm_cur_good
						set_local_variable = { name = epbm_price value = "price_in_market(scope:epbm_mkt)" }
						scope:epbm_bldg = { set_local_variable = { name = epbm_qty value = "building_goods_input(scope:epbm_cur_good)" } }
						change_local_variable = { name = epbm_bldg_cost add = { value = local_var:epbm_price multiply = local_var:epbm_qty } }
					}
				}

				# building_goods_input returns market_access * efficiency scaled values;
				# divide both out to recover the base per-level cost for caching
				set_local_variable = { name = epbm_loc_access value = "location.market_access" }
				set_local_variable = { name = epbm_bldg_eff value = "location.building_efficiency(scope:epbm_bt)" }

				if = {
					limit = { local_var:epbm_loc_access > 0 }
					change_local_variable = { name = epbm_bldg_cost divide = local_var:epbm_loc_access }
				}

				if = {
					limit = { local_var:epbm_bldg_eff > 0 }
					change_local_variable = { name = epbm_bldg_cost divide = local_var:epbm_bldg_eff }
				}

				set_local_variable = { name = epbm_per_level value = { value = local_var:epbm_bldg_cost divide = building_level } }

				# Cache the per-level base cost
				scope:epbm_mkt = {
					location = {
						add_to_variable_map = {
							name = epbm_pm_cache
							key = scope:epbm_pm
							value = local_var:epbm_per_level
						}
						add_to_variable_map = {
							name = epbm_bt_costs
							key = scope:epbm_bt
							value = local_var:epbm_per_level
						}
						add_to_variable_map = {
							name = epbm_bt_display
							key = scope:epbm_bt
							value = local_var:epbm_per_level
						}
					}
				}
			}

			# Apply level and efficiency to get actual cost. A building in another
			# country's location has no efficiency there (reads 0), so foreign
			# buildings are priced at level cost without it.
			set_local_variable = { name = epbm_bldg_eff value = "location.building_efficiency(scope:epbm_bt)" }
			if = {
				limit = { NOT = { is_building_owned_by = location.owner } }
				set_local_variable = { name = epbm_bldg_eff value = 1 }
			}
			set_local_variable = {
				name = epbm_bldg_cost
				value = {
					value = local_var:epbm_per_level
					multiply = building_level
					multiply = local_var:epbm_bldg_eff
				}
			}
		}
	}
}
