# Investment level: at each review, merges investment progress, recomputes each estate's level, and sizes its modifiers.

mnt_wealth_update_investment_levels = {
	mnt_wealth_for_each_cached_estate = { EFFECT = mnt_wealth_update_estate_investment_level }
}

mnt_wealth_update_estate_investment_level = {
	mnt_wealth_run_effect_with_estate_name = { EFFECT = mnt_wealth_update_estate_investment_level_by_name }
}

mnt_wealth_update_estate_investment_level_by_name = {
	mnt_wealth_add_progress_to_invested_wealth = yes

	# Only recompute if invested wealth changed since last review.
	if = {
		limit = { local_var:mnt_wealth_current_since_review != 0 }
		mnt_wealth_recompute_estate_investment_level = { ESTATE = $ESTATE$ }
	}
}

# Each level of investment(I) costs cost_base * I^2

mnt_wealth_recompute_estate_investment_level = {
	# pow = 0.5 is expensive; only recalculate if the level changes.
	set_local_variable = { name = mnt_wealth_investment_level_now value = 0 }
	if = {
		limit = { has_variable = mnt_wealth_applied_mnt_wealth_investment_$ESTATE$ }
		set_local_variable = {
			name = mnt_wealth_investment_level_now
			value = {
				value = var:mnt_wealth_applied_mnt_wealth_investment_$ESTATE$
				multiply = mnt_wealth_investment_rounding_factor
			}
		}
	}
	else_if = {
		limit = { has_variable = mnt_wealth_applied_mnt_wealth_neglect_$ESTATE$ }
		set_local_variable = {
			name = mnt_wealth_investment_level_now
			value = {
				value = var:mnt_wealth_applied_mnt_wealth_neglect_$ESTATE$
				multiply = mnt_wealth_investment_rounding_factor
				multiply = -1
			}
		}
	}

	# Negative invested wealth = neglect. Level uses absolute value; sign is restored at the end.
	set_local_variable = { name = mnt_wealth_level_cost_base value = mnt_wealth_investment_level_cost_base }
	set_local_variable = { name = mnt_wealth_investment_pool_value      value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_investment_pool_value
			value = "variable_map(mnt_wealth_invested_wealth|local_var:mnt_wealth_estate_key)"
		}
	}
	set_local_variable = { name = mnt_wealth_investment_level_sign value = 1 }
	set_local_variable = { name = mnt_wealth_investment_pool_magnitude        value = local_var:mnt_wealth_investment_pool_value }
	if = {
		limit = { local_var:mnt_wealth_investment_pool_value < 0 }
		set_local_variable = { name = mnt_wealth_investment_level_sign value = -1 }
		set_local_variable = {
			name = mnt_wealth_investment_pool_magnitude
			value = { value = local_var:mnt_wealth_investment_pool_value multiply = -1 }
		}
	}

	set_local_variable = {
		name = mnt_wealth_whole_investment_level
		value = {
			value = local_var:mnt_wealth_investment_level_now
			multiply = local_var:mnt_wealth_investment_level_sign
			floor = yes
			min = 0
		}
	}
	set_local_variable = {
		name = mnt_wealth_reached_cost
		value = {
			value = local_var:mnt_wealth_whole_investment_level
			multiply = local_var:mnt_wealth_whole_investment_level
			multiply = local_var:mnt_wealth_level_cost_base
		}
	}
	set_local_variable = {
		name = mnt_wealth_next_investment_level_cost
		value = {
			value = local_var:mnt_wealth_whole_investment_level
			add = 1
			save_temporary_value_as = mnt_wealth_next_whole_investment_level
			multiply = scope:mnt_wealth_next_whole_investment_level
			multiply = local_var:mnt_wealth_level_cost_base
		}
	}

	if = {
		limit = {
			OR = {
				local_var:mnt_wealth_investment_pool_magnitude < local_var:mnt_wealth_reached_cost
				local_var:mnt_wealth_investment_pool_magnitude >= local_var:mnt_wealth_next_investment_level_cost
			}
		}
		set_local_variable = {
			name = mnt_wealth_whole_investment_level
			value = {
				value = local_var:mnt_wealth_investment_pool_magnitude
				divide = local_var:mnt_wealth_level_cost_base
				pow = 0.5
				floor = yes
			}
		}
		set_local_variable = {
			name = mnt_wealth_reached_cost
			value = {
				value = local_var:mnt_wealth_whole_investment_level
				multiply = local_var:mnt_wealth_whole_investment_level
				multiply = local_var:mnt_wealth_level_cost_base
			}
		}
		set_local_variable = {
			name = mnt_wealth_next_investment_level_cost
			value = {
				value = local_var:mnt_wealth_whole_investment_level
				add = 1
				save_temporary_value_as = mnt_wealth_next_whole_investment_level
				multiply = scope:mnt_wealth_next_whole_investment_level
				multiply = local_var:mnt_wealth_level_cost_base
			}
		}
	}

	# Whole level plus the linear fraction of the way to the next one, signed.
	set_local_variable = {
		name = mnt_wealth_current_level
		value = {
			value = local_var:mnt_wealth_next_investment_level_cost
			subtract = local_var:mnt_wealth_reached_cost
			save_temporary_value_as = mnt_wealth_this_investment_level_cost
			value = local_var:mnt_wealth_investment_pool_magnitude
			subtract = local_var:mnt_wealth_reached_cost
			divide = scope:mnt_wealth_this_investment_level_cost
			add = local_var:mnt_wealth_whole_investment_level
			multiply = local_var:mnt_wealth_investment_level_sign
		}
	}

	if = {
		limit = { local_var:mnt_wealth_current_level > 0 }
		mnt_wealth_remove_sized_modifier = { MODIFIER = mnt_wealth_neglect_$ESTATE$ }
		mnt_wealth_apply_sized_modifier = {
			MODIFIER = mnt_wealth_investment_$ESTATE$
			SIZE = local_var:mnt_wealth_current_level
			FACTOR = mnt_wealth_investment_rounding_factor
		}
	}
	else_if = {
		limit = { local_var:mnt_wealth_current_level < 0 }
		mnt_wealth_remove_sized_modifier = { MODIFIER = mnt_wealth_investment_$ESTATE$ }
		set_local_variable = {
			name = mnt_wealth_neglect_size
			value = { value = local_var:mnt_wealth_current_level multiply = -1 }
		}
		mnt_wealth_apply_sized_modifier = {
			MODIFIER = mnt_wealth_neglect_$ESTATE$
			SIZE = local_var:mnt_wealth_neglect_size
			FACTOR = mnt_wealth_investment_rounding_factor
		}
	}
	else = {
		mnt_wealth_remove_estate_investment_modifiers = { ESTATE = $ESTATE$ }
	}
}

# Runs EFFECT with ESTATE set to the short estate name, for effects that name modifiers or variables per estate.

mnt_wealth_run_effect_with_estate_name = {
	switch = {
		trigger = local_var:mnt_wealth_estate_key
		estate_type:nobles_estate   = { $EFFECT$ = { ESTATE = nobles } }
		estate_type:clergy_estate   = { $EFFECT$ = { ESTATE = clergy } }
		estate_type:burghers_estate = { $EFFECT$ = { ESTATE = burghers } }
		estate_type:peasants_estate = { $EFFECT$ = { ESTATE = peasants } }
		estate_type:dhimmi_estate   = { $EFFECT$ = { ESTATE = dhimmi } }
		estate_type:tribes_estate   = { $EFFECT$ = { ESTATE = tribes } }
		estate_type:cossacks_estate = { $EFFECT$ = { ESTATE = cossacks } }
	}
}

mnt_wealth_add_progress_to_invested_wealth = {
	set_local_variable = { name = mnt_wealth_current_invested value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_current_invested
			value = "variable_map(mnt_wealth_invested_wealth|local_var:mnt_wealth_estate_key)"
		}
	}

	set_local_variable = { name = mnt_wealth_current_since_review value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_since_review target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_current_since_review
			value = "variable_map(mnt_wealth_invested_since_review|local_var:mnt_wealth_estate_key)"
		}
	}

	set_local_variable = {
		name = mnt_wealth_merged_invested
		value = { value = local_var:mnt_wealth_current_invested add = local_var:mnt_wealth_current_since_review }
	}

	add_to_variable_map = {
		name = mnt_wealth_invested_wealth
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_merged_invested
	}

	add_to_variable_map = {
		name = mnt_wealth_invested_since_review
		key = local_var:mnt_wealth_estate_key
		value = 0
	}
}

mnt_wealth_remove_estate_investment_modifiers = {
	mnt_wealth_remove_sized_modifier = { MODIFIER = mnt_wealth_investment_$ESTATE$ }
	mnt_wealth_remove_sized_modifier = { MODIFIER = mnt_wealth_neglect_$ESTATE$ }
}
