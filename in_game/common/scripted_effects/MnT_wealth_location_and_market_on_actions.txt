# The scheduler: which locations get serviced on which day, and the service bookkeeping. It fires on_actions and does no other work.
# The entire purpose of this system is to spread calculations out over day ticks and to let any systems that need to do an every_market
# or every_location loop do it in the same loop. It also stops stutters from large nations evaluating every_owned_location on one day when
# that can be at or above 10% of the total owned locations, Yuan/Golden Horde/etc.

# One service bucket per month of the year; twelve maps because map names cannot be computed.

mnt_dispatch_location_and_market_on_actions = {
	# This month's bucket: months mod twelve. Target locations are collected first into a list, so the bucket map is
	# never written while it is iterated.
	set_local_variable = {
		name = mnt_wealth_this_months_locations
		value = { value = global_var:mnt_wealth_months_elapsed modulo = 12 add = 1 }
	}
	clear_local_variable_list = mnt_wealth_target_locations
	switch = {
		trigger = local_var:mnt_wealth_this_months_locations
		1  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_1  mnt_wealth_collect_target_location = yes } }
		2  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_2  mnt_wealth_collect_target_location = yes } }
		3  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_3  mnt_wealth_collect_target_location = yes } }
		4  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_4  mnt_wealth_collect_target_location = yes } }
		5  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_5  mnt_wealth_collect_target_location = yes } }
		6  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_6  mnt_wealth_collect_target_location = yes } }
		7  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_7  mnt_wealth_collect_target_location = yes } }
		8  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_8  mnt_wealth_collect_target_location = yes } }
		9  = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_9  mnt_wealth_collect_target_location = yes } }
		10 = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_10 mnt_wealth_collect_target_location = yes } }
		11 = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_11 mnt_wealth_collect_target_location = yes } }
		12 = { every_key_in_global_variable_map = { variable = mnt_wealth_service_bucket_12 mnt_wealth_collect_target_location = yes } }
	}
	every_in_local_list = {
		variable = mnt_wealth_target_locations
		mnt_wealth_dispatch_location = yes
	}

	# Every human-owned location gets the monthly pulses: one now, one on its spread day.
	every_country = {
		limit = { is_human = yes }
		every_owned_location = {
			mnt_wealth_set_dispatch_day = yes
			trigger_event_silently = { on_action = monthly_human_location_instant_pulse_MnT }
			trigger_event_silently = { on_action = monthly_human_location_balanced_pulse_MnT days = local_var:mnt_wealth_dispatch_day }
		}
	}

	every_market_in_world = {
		trigger_event_silently = { on_action = monthly_market_pulse_MnT }
	}
}

# A bucket entry is the target when its next service month has arrived. Rescheduled entries sit in the same bucket a year out.

mnt_wealth_collect_target_location = {
	if = {
		limit = {
			has_owner = yes
			var:mnt_wealth_next_service_month <= global_var:mnt_wealth_months_elapsed
		}
		add_to_local_variable_list = { name = mnt_wealth_target_locations target = this }
	}
}

# One location's yearly service: count it, fire the pulses for this month (instant now, balanced on some day during 
# the month, evenly spread), record it, and queue the next one twelve months out. What runs on the pulses is not this file's business.

mnt_wealth_dispatch_location = {
	mnt_wealth_ensure_location_id = yes
	mnt_wealth_set_dispatch_day = yes

	# One counter per location: every second service is also biyearly, every fourth also four-yearly.
	change_variable = { name = mnt_wealth_service_counter add = 1 }
	set_local_variable = { name = mnt_wealth_biyearly_phase    value = { value = var:mnt_wealth_service_counter modulo = 2 } }
	set_local_variable = { name = mnt_wealth_four_yearly_phase value = { value = var:mnt_wealth_service_counter modulo = 4 } }

	# Fire yearly pulses
	trigger_event_silently = { on_action = yearly_location_instant_pulse_MnT }
	trigger_event_silently = { on_action = yearly_location_balanced_pulse_MnT days = local_var:mnt_wealth_dispatch_day }
	if = {
		limit = { local_var:mnt_wealth_biyearly_phase = 0 }
		# Fire bi-yearly pulses
		trigger_event_silently = { on_action = biyearly_location_instant_pulse_MnT }
		trigger_event_silently = { on_action = biyearly_location_balanced_pulse_MnT days = local_var:mnt_wealth_dispatch_day }
	}
	if = {
		# Fire 4-yearly pulses
		limit = { local_var:mnt_wealth_four_yearly_phase = 0 }
		trigger_event_silently = { on_action = four_yearly_location_instant_pulse_MnT }
		trigger_event_silently = { on_action = four_yearly_location_balanced_pulse_MnT days = local_var:mnt_wealth_dispatch_day }
	}

	# Track when this location was last updated
	set_local_variable = { name = mnt_wealth_last_service value = global_var:mnt_wealth_months_elapsed }
	save_temporary_scope_as = mnt_wealth_service_entity
	add_to_global_variable_map = {
		name = mnt_wealth_location_last_service
		key = scope:mnt_wealth_service_entity
		value = local_var:mnt_wealth_last_service
	}
	mnt_wealth_write_location_next_service = yes
}

# Locations in the same bucket spread across the month's days by id.

mnt_wealth_set_dispatch_day = {
	set_local_variable = {
		name = mnt_wealth_dispatch_day
		value = {
			value = var:mnt_wealth_location_id
			divide = 12
			floor = yes
			modulo = global_var:mnt_wealth_current_days_in_month
		}
	}
}

mnt_wealth_begin_month = {
	set_global_variable = {
		name = mnt_wealth_months_elapsed
		value = mnt_wealth_months_since_campaign_start
	}
	mnt_wealth_set_days_in_month = yes
}

mnt_wealth_set_days_in_month = {
	set_local_variable = { name = mnt_wealth_calendar_month value = current_month }
	set_local_variable = { name = mnt_wealth_month_days value = 30 }

	if = {
		limit = {
			is_key_in_global_variable_map = {
				name = mnt_wealth_days_in_month
				target = local_var:mnt_wealth_calendar_month
			}
		}
		set_local_variable = {
			name = mnt_wealth_month_days
			value = "global_variable_map(mnt_wealth_days_in_month|local_var:mnt_wealth_calendar_month)"
		}
	}

	set_global_variable = {
		name = mnt_wealth_current_days_in_month
		value = local_var:mnt_wealth_month_days
	}
}

mnt_wealth_ensure_location_id = {
	if = {
		limit = { NOT = { has_variable = mnt_wealth_location_id } }
		set_variable = { name = mnt_wealth_location_id value = global_var:mnt_wealth_location_serial }
		change_global_variable = { name = mnt_wealth_location_serial add = 1 }
	}
}

# Game start: the last service is backdated by the location id, so the first year's services spread across twelve months.

mnt_wealth_seed_location_schedule = {
	mnt_wealth_ensure_location_id = yes
	set_local_variable = {
		name = mnt_wealth_last_service
		value = {
			value = var:mnt_wealth_location_id
			modulo = 12
			multiply = -1
		}
	}
	mnt_wealth_write_location_schedule = yes
}

# Records last service, seeds the service counter by id, and queues the next service.

mnt_wealth_write_location_schedule = {
	mnt_wealth_ensure_location_id = yes
	save_temporary_scope_as = mnt_wealth_seed_location
	add_to_global_variable_map = {
		name = mnt_wealth_location_last_service
		key = scope:mnt_wealth_seed_location
		value = local_var:mnt_wealth_last_service
	}

	# Spread the biyearly and four-yearly pulses across cohorts.
	set_local_variable = {
		name = mnt_wealth_seed_counter
		value = { value = var:mnt_wealth_location_id modulo = 4 }
	}
	set_variable = { name = mnt_wealth_service_counter value = local_var:mnt_wealth_seed_counter }

	mnt_wealth_write_location_next_service = yes
}

mnt_wealth_write_location_next_service = {
	set_local_variable = {
		name = mnt_wealth_next_service
		value = { value = local_var:mnt_wealth_last_service add = 12 }
	}

	mnt_wealth_schedule_next_service = yes
}

mnt_wealth_schedule_next_service = {
	set_local_variable = {
		name = mnt_wealth_service_bucket_index
		value = { value = local_var:mnt_wealth_next_service modulo = 12 add = 1 }
	}

	# Pull the old entry when the bucket changes.
	if = {
		limit = {
			has_variable = mnt_wealth_service_bucket
			NOT = { var:mnt_wealth_service_bucket = local_var:mnt_wealth_service_bucket_index }
		}
		mnt_wealth_unschedule_location = yes
	}
	set_variable = { name = mnt_wealth_service_bucket     value = local_var:mnt_wealth_service_bucket_index }
	set_variable = { name = mnt_wealth_next_service_month value = local_var:mnt_wealth_next_service }

	save_temporary_scope_as = mnt_wealth_bucket_entity
	switch = {
		trigger = local_var:mnt_wealth_service_bucket_index
		1  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_1  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		2  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_2  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		3  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_3  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		4  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_4  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		5  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_5  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		6  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_6  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		7  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_7  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		8  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_8  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		9  = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_9  key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		10 = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_10 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		11 = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_11 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
		12 = { add_to_global_variable_map = { name = mnt_wealth_service_bucket_12 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service } }
	}
}

mnt_wealth_unschedule_location = {
	if = {
		limit = { has_variable = mnt_wealth_service_bucket }
		save_temporary_scope_as = mnt_wealth_bucket_entity
		switch = {
			trigger = var:mnt_wealth_service_bucket
			1  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_1  key = scope:mnt_wealth_bucket_entity } }
			2  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_2  key = scope:mnt_wealth_bucket_entity } }
			3  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_3  key = scope:mnt_wealth_bucket_entity } }
			4  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_4  key = scope:mnt_wealth_bucket_entity } }
			5  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_5  key = scope:mnt_wealth_bucket_entity } }
			6  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_6  key = scope:mnt_wealth_bucket_entity } }
			7  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_7  key = scope:mnt_wealth_bucket_entity } }
			8  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_8  key = scope:mnt_wealth_bucket_entity } }
			9  = { remove_from_global_variable_map = { name = mnt_wealth_service_bucket_9  key = scope:mnt_wealth_bucket_entity } }

	}
