# GDP and Wages: GDP survey pipeline.
#
# Prices every good a location produces to compute its GDP,
# and computes each estate's raw power and desired wealth. The survey
# runs on a multi-month rotation via the load balancer so not every
# location is surveyed every month.

# ── GOODS DISCOVERY ──

mnt_wealth_clear_found_goods = {
	clear_local_variable_map = mnt_wealth_found_goods_map
	clear_local_variable_list = mnt_wealth_found_goods_list
}

mnt_wealth_add_raw_material_good = {
	if = {
		limit = { exists = raw_material }

		add_to_local_variable_map = {
			name = mnt_wealth_found_goods_map
			key = raw_material
			value = 1
		}

		add_to_local_variable_list = { name = mnt_wealth_found_goods_list target = raw_material }
	}
}

mnt_wealth_set_goods_list = {
	clear_variable_list = mnt_wealth_goods_list

	every_in_local_list = {
		variable = mnt_wealth_found_goods_list
		save_temporary_scope_as = mnt_wealth_current_good

		scope:mnt_wealth_gdp_location ?= {
			add_to_variable_list = { name = mnt_wealth_goods_list target = scope:mnt_wealth_current_good }
		}
	}
}

# ── LOCAL GDP ──

# Scope: location. Calculates this location's GDP: the market value of
# every good it produces, summed up.

mnt_wealth_calculate_local_gdp = {
	set_variable = { name = mnt_wealth_local_gdp value = 0 }
	set_variable = { name = mnt_wealth_local_rural_gdp value = 0 }
	set_variable = { name = mnt_wealth_local_urban_gdp value = 0 }

	every_in_list = {
		variable = mnt_wealth_goods_list
		save_temporary_scope_as = mnt_wealth_current_good
		set_local_variable = { name = mnt_wealth_current_price value = "price_in_market(scope:mnt_wealth_gdp_location.market)" }

		scope:mnt_wealth_gdp_location = {
			set_local_variable = {
				name = mnt_wealth_good_value
				value = { value = "goods_output(scope:mnt_wealth_current_good)" multiply = local_var:mnt_wealth_current_price }
			}

			change_variable = { name = mnt_wealth_local_gdp add = local_var:mnt_wealth_good_value }

			if = {
				limit = { "global_variable_map(mnt_wealth_good_classification|scope:mnt_wealth_current_good)" = flag:urban }
				change_variable = { name = mnt_wealth_local_urban_gdp add = local_var:mnt_wealth_good_value }
			}
			else = {
				change_variable = { name = mnt_wealth_local_rural_gdp add = local_var:mnt_wealth_good_value }
			}
		}
	}
}

# Scope: location. Sums each building's profit to get the urban net
# domestic product.

mnt_wealth_calculate_local_urban_ndp = {
	set_variable = { name = mnt_wealth_local_urban_ndp value = 0 }
	every_buildings_in_location = {
		scope:mnt_wealth_gdp_location = {
			change_variable = {
				name = mnt_wealth_local_urban_ndp
				add = prev.building_profit
			}
		}
	}
}

# ── BUILDING CLASSIFICATION ──

# Scope: none (fired from weather_monthly_pulse).
# Rebuilds the building-type lookup table when it expires. The table
# caches which building types produce goods so the GDP survey does
# not have to check every building every time.

mnt_wealth_maintain_building_produced_goods_cache = {
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = mnt_wealth_building_table_age }
				global_var:mnt_wealth_building_table_age >= mnt_wealth_building_table_lifetime_months
			}
		}
		clear_global_variable_map = mnt_wealth_building_makes_goods
		set_global_variable = { name = mnt_wealth_building_table_age value = 0 }
	}
	else = {
		change_global_variable = { name = mnt_wealth_building_table_age add = 1 }
	}
}

# Scope: building. Caches whether this building type produces goods
# in a global map so we only check each type once per table lifetime.

mnt_wealth_classify_building_type = {
	if = {
		limit = {
			NOT = {
				is_key_in_global_variable_map = {
					name = mnt_wealth_building_makes_goods
					target = scope:mnt_wealth_current_building_type
				}
			}
		}
		if = {
			limit = { any_production_method_of_building = { exists = produced_goods } }
			add_to_global_variable_map = {
				name = mnt_wealth_building_makes_goods
				key = scope:mnt_wealth_current_building_type
				value = 1
			}
		}
		else = {
			add_to_global_variable_map = {
				name = mnt_wealth_building_makes_goods
				key = scope:mnt_wealth_current_building_type
				value = 0
			}
		}
	}
}

mnt_wealth_collect_building_goods = {
	if = {
		limit = { "global_variable_map(mnt_wealth_building_makes_goods|scope:mnt_wealth_current_building_type)" > 0 }
		every_production_method_of_building = {
			limit = {
				exists = produced_goods
				NOT = { is_key_in_local_variable_map = { name = mnt_wealth_found_goods_map target = produced_goods } }
			}
			add_to_local_variable_map = {
				name = mnt_wealth_found_goods_map
				key = produced_goods
				value = 1
			}

			add_to_local_variable_list = { name = mnt_wealth_found_goods_list target = produced_goods }
		}
	}
}

# ── ESTATE POWER ──

# Scope: location. Counts every pop into the estate that actually holds it and
# files the totals in mnt_estate_pops, with pops x power per pop in
# mnt_estate_raw_power. Membership is per pop, not per
# pop type: dhimmi and cossacks divert pops out of their usual estate.

mnt_wealth_count_estate_pops = {
	mnt_wealth_clear_pop_counts = yes

	# Count by type first. It needs no iteration and costs about a microsecond,
	# and the totals give the check below something to test the engine against.
	mnt_wealth_count_pops_by_type = yes

	set_local_variable = { name = mnt_wealth_needs_pop_walk value = 0 }

	# Total what was filed and compare it against the location's own population.
	# Counting by type only knows the pop types listed above, so a shortfall means
	# a pop reached no estate. The walk routes whatever it finds, so it cannot
	# miss one the same way.
	set_local_variable = { name = mnt_wealth_pops_total value = 0 }
	change_local_variable = { name = mnt_wealth_pops_total add = local_var:mnt_wealth_pops_nobles }
	change_local_variable = { name = mnt_wealth_pops_total add = local_var:mnt_wealth_pops_clergy }
	change_local_variable = { name = mnt_wealth_pops_total add = local_var:mnt_wealth_pops_burghers }
	change_local_variable = { name = mnt_wealth_pops_total add = local_var:mnt_wealth_pops_peasants }
	change_local_variable = { name = mnt_wealth_pops_total add = local_var:mnt_wealth_pops_dhimmi }
	change_local_variable = { name = mnt_wealth_pops_total add = local_var:mnt_wealth_pops_tribes }
	change_local_variable = { name = mnt_wealth_pops_total add = local_var:mnt_wealth_pops_cossacks }

	set_local_variable = { name = mnt_wealth_pops_population value = population }
	set_local_variable = { name = mnt_wealth_pops_unfiled value = local_var:mnt_wealth_pops_population }
	change_local_variable = { name = mnt_wealth_pops_unfiled subtract = local_var:mnt_wealth_pops_total }

	# Held for the log, since the walk below overwrites the live counters.
	set_local_variable = { name = mnt_wealth_pops_bytype  value = local_var:mnt_wealth_pops_total }
	set_local_variable = { name = mnt_wealth_pops_shortfall value = local_var:mnt_wealth_pops_unfiled }
	# 0 by type, 1 unused (shortfall check disabled), 2 walked for dhimmi power,
	# 3 tribesmen split, 4 slave split, 5 both splits.
	set_local_variable = { name = mnt_wealth_pops_path value = 0 }

	# DISABLED. A shortfall against the location's own population would mean a pop
	# reached no estate, and the walk would pick it up. It never once fired across
	# 335,337 surveys in testing, so it is off for a small runtime saving. Turn it
	# back on if common/pop_types gains a type that mnt_wealth_count_pops_by_type
	# does not list, which is the case it guards.
	#
	# It does NOT guard a dhimmi estate made only of slaves: counting by type puts
	# those slaves in peasants, so the total still matches and nothing fires.
	# mnt_wealth_count_pops_slave_split covers that, and does get hit.
	#
	# The population read and the total above are kept only because the log line
	# reports them; drop them too if that logging goes.
	#
	#if = {
	#	limit = { local_var:mnt_wealth_pops_unfiled > 0.001 }
	#	set_local_variable = { name = mnt_wealth_needs_pop_walk value = 1 }
	#	set_local_variable = { name = mnt_wealth_pops_path value = 1 }
	#}

	# Dhimmi can take a pop of any type, so its presence forces the full walk.
	if = {
		limit = { "local_estate_power(estate_type:dhimmi_estate)" > 0 }
		set_local_variable = { name = mnt_wealth_needs_pop_walk value = 1 }
		set_local_variable = { name = mnt_wealth_pops_path value = 2 }
	}

	if = {
		limit = { local_var:mnt_wealth_needs_pop_walk = 1 }
		# The by-type totals filed pops into estates that may not hold them.
		mnt_wealth_clear_pop_counts = yes
		mnt_wealth_count_pops_full = yes
	}
	else = {
		# Slaves carry no power, so a dhimmi estate made only of them reads as
		# absent above. Zero dhimmi power means no pop that carries power is
		# dhimmi, which leaves the slaves as the only ones that can be hidden.
		if = {
			limit = {
				local_var:mnt_wealth_pops_slaves > 0
				owner = { country_has_estate = estate_type:dhimmi_estate }
			}
			mnt_wealth_count_pops_slave_split = yes
			set_local_variable = { name = mnt_wealth_pops_path value = 4 }
		}

		if = {
			limit = { "local_estate_power(estate_type:cossacks_estate)" > 0 }
			mnt_wealth_count_pops_tribesmen_split = yes
			if = {
				limit = { local_var:mnt_wealth_pops_path = 4 }
				set_local_variable = { name = mnt_wealth_pops_path value = 5 }
			}
			else = { set_local_variable = { name = mnt_wealth_pops_path value = 3 } }
		}
	}

	mnt_wealth_store_estate_pops = { ESTATE = nobles }
	mnt_wealth_store_estate_pops = { ESTATE = clergy }
	mnt_wealth_store_estate_pops = { ESTATE = burghers }
	mnt_wealth_store_estate_pops = { ESTATE = peasants }
	mnt_wealth_store_estate_pops = { ESTATE = dhimmi }
	mnt_wealth_store_estate_pops = { ESTATE = tribes }
	mnt_wealth_store_estate_pops = { ESTATE = cossacks }
}

# Scope: location. Resets the per-estate pop counters.

mnt_wealth_clear_pop_counts = {
	set_local_variable = { name = mnt_wealth_pops_nobles value = 0 }
	set_local_variable = { name = mnt_wealth_pops_clergy value = 0 }
	set_local_variable = { name = mnt_wealth_pops_burghers value = 0 }
	set_local_variable = { name = mnt_wealth_pops_peasants value = 0 }
	set_local_variable = { name = mnt_wealth_pops_dhimmi value = 0 }
	set_local_variable = { name = mnt_wealth_pops_tribes value = 0 }
	set_local_variable = { name = mnt_wealth_pops_cossacks value = 0 }
}

# Scope: location. Files one estate's pops and raw power. Skips an estate with
# no pops here, or one the owner does not hold, so the maps list only estates
# that are both present in the location and part of the country.

mnt_wealth_store_estate_pops = {
	if = {
		limit = {
			local_var:mnt_wealth_pops_$ESTATE$ > 0
			owner = { country_has_estate = estate_type:$ESTATE$_estate }
		}
		set_local_variable = { name = mnt_wealth_pops_key value = estate_type:$ESTATE$_estate }
		set_local_variable = {
			name = mnt_wealth_pops_per_pop
			value = "global_variable_map(mnt_wealth_power_per_pop|local_var:mnt_wealth_pops_key)"
		}
		# add_to_variable_map takes a plain target, so the product is worked out first.
		set_local_variable = {
			name = mnt_wealth_pops_raw
			value = {
				value = local_var:mnt_wealth_pops_$ESTATE$
				multiply = local_var:mnt_wealth_pops_per_pop
			}
		}

		add_to_variable_map = {
			name = mnt_estate_pops
			key = local_var:mnt_wealth_pops_key
			value = local_var:mnt_wealth_pops_$ESTATE$
		}
		add_to_variable_map = {
			name = mnt_estate_raw_power
			key = local_var:mnt_wealth_pops_key
			value = local_var:mnt_wealth_pops_raw
		}
	}
}

# Scope: location. No estate can divert a pop, so pop type is estate.

mnt_wealth_count_pops_by_type = {
	set_local_variable = { name = mnt_wealth_pops_nobles   value = num_pop_type:nobles }
	set_local_variable = { name = mnt_wealth_pops_clergy   value = num_pop_type:clergy }
	set_local_variable = { name = mnt_wealth_pops_burghers value = num_pop_type:burghers }
	set_local_variable = { name = mnt_wealth_pops_tribes   value = num_pop_type:tribesmen }
	set_local_variable = { name = mnt_wealth_pops_slaves   value = num_pop_type:slaves }
	set_local_variable = { name = mnt_wealth_pops_peasants value = num_pop_type:laborers }
	change_local_variable = { name = mnt_wealth_pops_peasants add = num_pop_type:soldiers }
	change_local_variable = { name = mnt_wealth_pops_peasants add = num_pop_type:peasants }
	change_local_variable = { name = mnt_wealth_pops_peasants add = local_var:mnt_wealth_pops_slaves }
}

# Scope: location. Moves the dhimmi slaves out of peasants, where counting by
# type put them. Only slaves are walked because only they can be hidden here.

mnt_wealth_count_pops_slave_split = {
	every_pop = {
		limit = {
			pop_type = pop_type:slaves
			is_dhimmi = yes
		}
		change_local_variable = { name = mnt_wealth_pops_dhimmi   add = pop_size }
		change_local_variable = { name = mnt_wealth_pops_peasants subtract = pop_size }
	}
}

# Scope: location. Cossacks only take tribesmen, so only tribesmen are walked.

mnt_wealth_count_pops_tribesmen_split = {
	set_local_variable = { name = mnt_wealth_pops_tribes value = 0 }

	every_pop = {
		limit = { pop_type = pop_type:tribesmen }
		if = {
			limit = { is_cossacks = yes }
			change_local_variable = { name = mnt_wealth_pops_cossacks add = pop_size }
		}
		else = { change_local_variable = { name = mnt_wealth_pops_tribes add = pop_size } }
	}
}

# Scope: location. Full walk, in the order common/pop_types resolves: tribesmen
# never reach dhimmi, and dhimmi outranks a pop's own estate for everything else.

mnt_wealth_count_pops_full = {
	every_pop = {
		if = {
			limit = { pop_type = pop_type:tribesmen }
			if = {
				limit = { is_cossacks = yes }
				change_local_variable = { name = mnt_wealth_pops_cossacks add = pop_size }
			}
			else = { change_local_variable = { name = mnt_wealth_pops_tribes add = pop_size } }
		}
		else_if = {
			limit = { is_dhimmi = yes }
			change_local_variable = { name = mnt_wealth_pops_dhimmi add = pop_size }
		}
		else_if = {
			limit = { pop_type = pop_type:nobles }
			change_local_variable = { name = mnt_wealth_pops_nobles add = pop_size }
		}
		else_if = {
			limit = { pop_type = pop_type:clergy }
			change_local_variable = { name = mnt_wealth_pops_clergy add = pop_size }
		}
		else_if = {
			limit = { pop_type = pop_type:burghers }
			change_local_variable = { name = mnt_wealth_pops_burghers add = pop_size }
		}
		else = { change_local_variable = { name = mnt_wealth_pops_peasants add = pop_size } }
	}
}

# Scope: location. Clears and rebuilds the full estate survey: raw power,
# cached relative power, expected wealth, base desired wealth, and desired
# total. Also seeds investment level for estates arriving at this location
# for the first time.
#
# ITERATOR selects the estate iterator: mnt_wealth_for_all_estate_types at game start
# (because the cache does not exist yet), mnt_wealth_for_each_national_estate at runtime.

mnt_wealth_survey_estate_data = {
	save_temporary_scope_as = mnt_wealth_survey_location
	clear_variable_map = mnt_estate_pops
	clear_variable_map = mnt_estate_raw_power
	clear_variable_map = mnt_wealth_cached_local_relative_estate_power
	clear_variable_map = mnt_wealth_expected_wealth
	set_variable = { name = mnt_wealth_desired_total value = 0 }

	if = {
		limit = { exists = owner }
		mnt_wealth_count_estate_pops = yes

		$ITERATOR$ = { EFFECT = mnt_wealth_store_estate_survey_results }
		# Estates holding a wealth share or pops can be missing from the owner's
		# list. Pick up any the iterator did not reach.
		every_key_in_variable_map = {
			variable = mnt_wealth_local_wealth_share
			save_temporary_scope_as = mnt_wealth_leftover_key

			scope:mnt_wealth_survey_location = {
				if = {
					limit = {
						NOT = { mnt_wealth_estate_is_present = { KEY = scope:mnt_wealth_leftover_key } }
					}
					set_local_variable = { name = mnt_wealth_estate_key value = scope:mnt_wealth_leftover_key }
					mnt_wealth_store_estate_survey_results = yes
				}
			}
		}

		every_key_in_variable_map = {
			variable = mnt_estate_pops
			save_temporary_scope_as = mnt_wealth_leftover_key

			scope:mnt_wealth_survey_location = {
				if = {
					limit = {
						NOT = { mnt_wealth_estate_is_present = { KEY = scope:mnt_wealth_leftover_key } }
					}
					set_local_variable = { name = mnt_wealth_estate_key value = scope:mnt_wealth_leftover_key }
					mnt_wealth_store_estate_survey_results = yes
				}
			}
		}
	}
}

# Scope: location. Writes one estate's survey results: cached relative power,
# investment level (if new), base desired wealth, and desired wealth. Raw power
# is already in mnt_estate_raw_power; this only lifts it into the local that
# mnt_wealth_set_base_desired_wealth reads.

mnt_wealth_store_estate_survey_results = {
	set_local_variable = {
		name = mnt_wealth_estate_relative_power
		value = "local_relative_estate_power(local_var:mnt_wealth_estate_key)"
	}

	set_local_variable = { name = mnt_wealth_raw_power_for_one_estate value = 0 }
	if = {
		limit = {
			is_key_in_variable_map = { name = mnt_estate_raw_power target = local_var:mnt_wealth_estate_key }
		}
		set_local_variable = {
			name = mnt_wealth_raw_power_for_one_estate
			value = "variable_map(mnt_estate_raw_power|local_var:mnt_wealth_estate_key)"
		}
	}

	# Raw power and relative power can each exist without the other: an estate
	# made entirely of slaves has raw power and no relative power. Either one
	# makes the estate part of the survey.
	if = {
		limit = { OR = { local_var:mnt_wealth_estate_relative_power > 0 local_var:mnt_wealth_raw_power_for_one_estate > 0 } }

		add_to_variable_map = {
			name = mnt_wealth_cached_local_relative_estate_power
			key = local_var:mnt_wealth_estate_key
			value = local_var:mnt_wealth_estate_relative_power
		}

		# An estate arriving at this location for the first time starts unlevelled.
		if = {
			limit = {
				NOT = { is_key_in_variable_map = { name = mnt_wealth_investment_level target = local_var:mnt_wealth_estate_key } }
			}
			add_to_variable_map = {
				name = mnt_wealth_investment_level
				key = local_var:mnt_wealth_estate_key
				value = 0
			}
		}

		mnt_wealth_set_base_desired_wealth = yes
		# Maintains mnt_wealth_desired_total itself.
		mnt_wealth_set_desired_wealth = yes
	}
}

# ── SURVEY ENTRY POINTS ──


# Scope: location. Fills in survey data only if it is missing. Used
# at game start so locations that already have data are not resurveyed.

mnt_wealth_ensure_survey_data = {
	if = { limit = { var:mnt_wealth_local_gdp = 0 } mnt_wealth_update_local_gdp = yes }

	if = {
		limit = {
			NOT = {
				has_variable_map = mnt_wealth_cached_local_relative_estate_power
				has_variable_map = mnt_estate_raw_power
				has_variable_map = mnt_wealth_expected_wealth
			}
		}
		mnt_wealth_survey_estate_data = { ITERATOR = mnt_wealth_for_all_estate_types }
	}
}

# Scope: location. Recomputes this location's GDP from the goods it
# produces, and keeps the owner's national GDP total correct by removing
# the old value before the recompute and adding the new value after.

mnt_wealth_update_local_gdp = {
	save_scope_as = mnt_wealth_gdp_location
	mnt_wealth_remove_local_gdp_from_national_total = yes
	mnt_wealth_clear_found_goods = yes
	every_buildings_in_location         = { mnt_wealth_check_building = yes }
	every_foreign_buildings_in_location = { mnt_wealth_check_building = yes }
	mnt_wealth_set_goods_list = yes
	mnt_wealth_calculate_local_gdp = yes
	mnt_wealth_calculate_local_urban_ndp = yes
	mnt_wealth_add_local_gdp_to_national_total = yes
}

# ── ESTATE POWER RESURVEY ──

# Scope: location. Rebuilds all survey data for this location and keeps the
# owner's national raw power map correct: subtract the old power, run a
# full survey, then add the new power back.

mnt_wealth_resurvey_estate_data = {
	if = {
		limit = { exists = owner }
		owner = { save_temporary_scope_as = mnt_wealth_raw_power_country }
		mnt_wealth_apply_national_estate_raw_power = { SIGN = -1 }
	}

	mnt_wealth_survey_estate_data = { ITERATOR = mnt_wealth_for_each_national_estate }

	if = {
		limit = { exists = owner }
		owner = { save_temporary_scope_as = mnt_wealth_raw_power_country }
		mnt_wealth_apply_national_estate_raw_power = { SIGN = 1 }
	}
}

mnt_wealth_check_building = {
	building_type ?= { save_temporary_scope_as = mnt_wealth_current_building_type }
	mnt_wealth_classify_building_type = yes
	mnt_wealth_collect_building_goods = yes
}
