# GDP and Wages: the gold an estate spends on investment.
#
# Each month an estate takes its share of a location's wealth pool, pays the
# upkeep that holds its current level, and banks the difference as progress.
# AI locations skip the monthly loop and settle a whole year in one pass.
# The level that progress buys is worked out in mnt_wealth_investment_level.

# Scope: location. Adds local_var:mnt_wealth_investment_change to this estate's
# since-review progress.

mnt_wealth_add_investment_progress = {
	set_local_variable = { name = mnt_wealth_updated_since_review value = local_var:mnt_wealth_investment_change }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_since_review target = local_var:mnt_wealth_estate_key } }
		change_local_variable = {
			name = mnt_wealth_updated_since_review
			add = "variable_map(mnt_wealth_invested_since_review|local_var:mnt_wealth_estate_key)"
		}
	}

	add_to_variable_map = {
		name = mnt_wealth_invested_since_review
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_updated_since_review
	}
}

# Scope: location. Banks the gold one estate paid as progress:
# progress = paid x investment efficiency - upkeep x upkeep efficiency.
# The monthly and yearly callers fill mnt_wealth_estate_paid and mnt_wealth_estate_upkeep
# with one month's or one year's worth.

mnt_wealth_set_investment_progress_change = {
	set_local_variable = {
		name = mnt_wealth_investment_change
		value = {
			value = local_var:mnt_wealth_estate_paid
			multiply = mnt_wealth_investment_efficiency_multiplier
			subtract = { value = local_var:mnt_wealth_estate_upkeep multiply = mnt_wealth_upkeep_efficiency_multiplier }
		}
	}

	mnt_wealth_add_investment_progress = yes
}

# ── MONTHLY INVESTMENT ──

# Scope: location. Called by mnt_wealth_location_month through
# mnt_wealth_for_each_cached_estate.
# Takes a fraction of this estate's wealth as investment, subtracts
# upkeep, and banks the net as progress.

mnt_wealth_invest_estate_wealth = {
	set_local_variable = { name = mnt_wealth_estate_paid value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_estate_paid
			value = {
				value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
				multiply = local_var:mnt_wealth_cached_investment_rate
			}
		}
	}

	change_local_variable = { name = mnt_wealth_invested_running_total add = local_var:mnt_wealth_estate_paid }

	# Subtract the spent amount from this estate's wealth map entry.
	if = {
		limit = { local_var:mnt_wealth_estate_paid > 0 }
		set_local_variable = { name = mnt_wealth_invest_current value = 0 }

		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
			set_local_variable = {
				name = mnt_wealth_invest_current
				value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
			}
		}

		set_local_variable = {
			name = mnt_wealth_invest_after
			value = { value = local_var:mnt_wealth_invest_current subtract = local_var:mnt_wealth_estate_paid }
		}

		add_to_variable_map = {
			name = mnt_wealth_estate_wealth
			key = local_var:mnt_wealth_estate_key
			value = local_var:mnt_wealth_invest_after
		}
	}

	mnt_wealth_compute_estate_upkeep = {
		ESTATE_KEY = local_var:mnt_wealth_estate_key
		MONTHS = 1
	}

	mnt_wealth_set_investment_progress_change = yes
}

# Scope: location. Called by mnt_wealth_invest_estate_wealth and
# mnt_wealth_set_estate_investment_progress.
# One estate's upkeep for the period: its invested wealth decayed at the
# live per-estate rate. Self-gates on a positive pool.
# Saves local_var:mnt_wealth_estate_upkeep.

mnt_wealth_compute_estate_upkeep = {
	set_local_variable = { name = mnt_wealth_upkeep_key value = $ESTATE_KEY$ }
	set_local_variable = { name = mnt_wealth_estate_upkeep value = 0 }

	# Nothing invested, nothing to decay. A non-positive pool owes no upkeep.
	if = {
		limit = {
			is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = local_var:mnt_wealth_upkeep_key }
			"variable_map(mnt_wealth_invested_wealth|local_var:mnt_wealth_upkeep_key)" > 0
		}
		set_local_variable = { name = mnt_wealth_upkeep_invested value = "variable_map(mnt_wealth_invested_wealth|local_var:mnt_wealth_upkeep_key)" }

		# The decay rate is a live per-estate modifier, read by name each time so
		# it responds the moment it changes rather than at the next survey.
		set_local_variable = { name = mnt_wealth_upkeep_decay_rate value = 0 }
		mnt_wealth_run_effect_with_estate_name = { EFFECT = mnt_wealth_read_estate_decay_rate }

		# Drain the pool by rate * months, never below what is in it.
		set_local_variable = {
			name = mnt_wealth_estate_upkeep
			value = {
				value = local_var:mnt_wealth_upkeep_invested
				multiply = local_var:mnt_wealth_upkeep_decay_rate
				multiply = $MONTHS$
				max = local_var:mnt_wealth_upkeep_invested
			}
		}
	}
}

# Scope: location. Reads one estate's live decay rate (local plus national)
# into mnt_wealth_upkeep_decay_rate.

mnt_wealth_read_estate_decay_rate = {
	set_local_variable = {
		name = mnt_wealth_upkeep_decay_rate
		value = {
			value = modifier:mnt_wealth_local_$ESTATE$_investment_decay
			add = owner.modifier:mnt_wealth_global_$ESTATE$_investment_decay
		}
	}
}

# ── YEARLY SETTLEMENT ──

# Scope: location. Called yearly by mnt_wealth_location_year.
# Applies accumulated investment totals into per-estate progress, then
# sends national payouts. Only runs when there are totals to settle.

mnt_wealth_settle_investment_accumulators = {
	if = {
		limit = {
			has_variable = mnt_wealth_investment_paid_total
			OR = { var:mnt_wealth_investment_paid_total != 0 var:mnt_wealth_investment_rate_total != 0 }
		}
		# The two halves write different maps, so one walk covers both.
		set_local_variable = { name = mnt_wealth_arriving_running_total value = 0 }
		mnt_wealth_for_each_cached_estate = { EFFECT = mnt_wealth_settle_estate_progress_and_flow }
		change_variable = { name = mnt_wealth_pool add = local_var:mnt_wealth_arriving_running_total }
		set_variable = { name = mnt_wealth_investment_paid_total value = 0 }
		set_variable = { name = mnt_wealth_investment_rate_total value = 0 }
		set_variable = { name = mnt_wealth_investment_months     value = 0 }
	}
}

# Scope: location. One estate's yearly settlement: progress and national flow.

mnt_wealth_settle_estate_progress_and_flow = {
	mnt_wealth_set_estate_investment_progress = yes
	mnt_wealth_apply_national_flow_to_location = yes
}

# Scope: location. Called by mnt_wealth_settle_estate_progress_and_flow.
# Yearly equivalent of mnt_wealth_invest_estate_wealth: distributes
# the year's paid total to one estate by its wealth fraction.

mnt_wealth_set_estate_investment_progress = {
	mnt_wealth_compute_estate_upkeep = {
		ESTATE_KEY = local_var:mnt_wealth_estate_key
		MONTHS = var:mnt_wealth_investment_months
	}

	set_local_variable = { name = mnt_wealth_estate_paid value = 0 }

	if = {
		limit = {
			var:mnt_wealth_pool > 0
			is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key }
		}
		set_local_variable = {
			name = mnt_wealth_settle_fraction
			value = {
				value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
				divide = var:mnt_wealth_pool
			}
		}

		set_local_variable = {
			name = mnt_wealth_estate_paid
			value = {
				value = var:mnt_wealth_investment_paid_total
				multiply = local_var:mnt_wealth_settle_fraction
			}
		}
	}

	mnt_wealth_set_investment_progress_change = yes
}

# Scope: location (AI-owned). Called by mnt_wealth_location_year.
# Solves the elapsed months of income-then-invest as a geometric series
# instead of running individual monthly ticks.

mnt_wealth_set_ai_yearly_investment = {
	set_local_variable = { name = mnt_wealth_cached_investment_rate value = mnt_wealth_investment_rate }
	set_local_variable = { name = mnt_wealth_monthly_income         value = var:mnt_wealth_kept_income_total }
	set_local_variable = { name = mnt_wealth_spend_rate             value = local_var:mnt_wealth_cached_investment_rate }

	# Save pool before the series so we can scale the map afterward.
	set_local_variable = { name = mnt_wealth_ai_old_pool value = var:mnt_wealth_pool }

	if = {
		limit = { local_var:mnt_wealth_spend_rate > 0 var:mnt_wealth_pool > 0 }
		set_local_variable = { name = mnt_wealth_decay         value = { value = 1 subtract = local_var:mnt_wealth_spend_rate } }
		set_local_variable = { name = mnt_wealth_decay_twelfth value = mnt_wealth_decay_twelfth_power }
		# decay^age: for a 24-month cycle, square the twelfth power.
		set_local_variable = { name = mnt_wealth_decay_age value = local_var:mnt_wealth_decay_twelfth }
		if = {
			limit = { var:mnt_wealth_service_months > 12 }
			set_local_variable = {
				name = mnt_wealth_decay_age
				value = { value = local_var:mnt_wealth_decay_twelfth multiply = local_var:mnt_wealth_decay_twelfth }
			}
		}

		set_local_variable = {
			name = mnt_wealth_effective_months
			value = {
				value = 1
				subtract = local_var:mnt_wealth_decay_age
				divide = local_var:mnt_wealth_spend_rate
			}
		}

		set_local_variable = {
			name = mnt_wealth_pool_plus_income
			value = { value = var:mnt_wealth_pool add = local_var:mnt_wealth_monthly_income }
		}

		set_variable = {
			name = mnt_wealth_investment_paid_total
			value = {
				value = local_var:mnt_wealth_pool_plus_income
				multiply = local_var:mnt_wealth_effective_months

				add = {
					value = local_var:mnt_wealth_monthly_income
					multiply = { value = var:mnt_wealth_service_months subtract = local_var:mnt_wealth_effective_months }
					divide = local_var:mnt_wealth_spend_rate
				}

				multiply = local_var:mnt_wealth_cached_investment_rate
			}
		}

		set_variable = {
			name = mnt_wealth_pool
			value = {
				value = local_var:mnt_wealth_pool_plus_income
				multiply = local_var:mnt_wealth_decay_age

				add = {
					value = local_var:mnt_wealth_monthly_income
					multiply = { value = local_var:mnt_wealth_decay subtract = local_var:mnt_wealth_decay_age }
					divide = local_var:mnt_wealth_spend_rate
				}
			}
		}
	}
	else = {
		change_variable = { name = mnt_wealth_pool add = { value = local_var:mnt_wealth_monthly_income multiply = var:mnt_wealth_service_months } }
		set_variable = { name = mnt_wealth_investment_paid_total value = 0 }
	}

	# Bring the map to the new pool. Income and spending were pool-level
	# and proportional, so each estate keeps its fraction of the total.
	if = {
		limit = { local_var:mnt_wealth_ai_old_pool > 0 }
		set_local_variable = {
			name = mnt_wealth_ai_scale_factor
			value = { value = var:mnt_wealth_pool divide = local_var:mnt_wealth_ai_old_pool }
		}

		save_temporary_scope_as = mnt_wealth_ai_scale_loc

		every_key_in_variable_map = {
			variable = mnt_wealth_estate_wealth
			save_temporary_scope_as = mnt_wealth_ai_scale_key

			scope:mnt_wealth_ai_scale_loc = {
				set_local_variable = {
					name = mnt_wealth_ai_scaled_value
					value = {
						value = "variable_map(mnt_wealth_estate_wealth|scope:mnt_wealth_ai_scale_key)"
						multiply = local_var:mnt_wealth_ai_scale_factor
					}
				}

				add_to_variable_map = {
					name = mnt_wealth_estate_wealth
					key = scope:mnt_wealth_ai_scale_key
					value = local_var:mnt_wealth_ai_scaled_value
				}
			}
		}
	}
	else_if = {
		limit = { var:mnt_wealth_pool > 0 }
		# Income grew an empty pool: distribute it by relative power.
		set_local_variable = { name = mnt_wealth_starting_pool value = var:mnt_wealth_pool }
		set_local_variable = { name = mnt_wealth_seed_running_total value = 0 }
		mnt_wealth_for_each_estate_in_power_cache = { EFFECT = mnt_wealth_seed_estate_wealth }
		set_variable = { name = mnt_wealth_pool value = local_var:mnt_wealth_seed_running_total }
	}

	set_variable = {
		name = mnt_wealth_investment_rate_total
		value = { value = local_var:mnt_wealth_cached_investment_rate multiply = var:mnt_wealth_service_months }
	}
	set_variable = { name = mnt_wealth_investment_months value = var:mnt_wealth_service_months }
}
