# Gross Output and Wages: the gold an estate spends on investment.

# Scope: location. Adds local_var:mnt_wealth_investment_change to this estate's
# since-review progress.

mnt_wealth_add_investment_progress = {
	set_local_variable = { name = mnt_wealth_updated_since_review value = local_var:mnt_wealth_investment_change }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_since_review target = local_var:mnt_wealth_estate_key } }
		change_local_variable = {
			name = mnt_wealth_updated_since_review
			add = "variable_map(mnt_wealth_invested_since_review|local_var:mnt_wealth_estate_key)"
		}
	}

	add_to_variable_map = {
		name = mnt_wealth_invested_since_review
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_updated_since_review
	}
}

# Scope: location. Banks one estate's progress: paid times investment
# efficiency, less decay times decay resistance, less its maintenance share.
# Reads local_var:mnt_wealth_estate_paid, local_var:mnt_wealth_estate_decay
# and local_var:mnt_wealth_estate_maintenance.

mnt_wealth_bank_estate_investment_progress = {
	set_local_variable = {
		name = mnt_wealth_investment_change
		value = {
			value = local_var:mnt_wealth_estate_paid
			multiply = mnt_wealth_investment_efficiency_multiplier
			subtract = { value = local_var:mnt_wealth_estate_decay multiply = mnt_wealth_decay_resistance_multiplier }
			subtract = local_var:mnt_wealth_estate_maintenance
		}
	}

	mnt_wealth_add_investment_progress = yes
}

# Scope: location. One estate's share of the location's Local Investment
# Maintenance: the flat monthly cost split by the estate's share of the
# location's raw power. Saves local_var:mnt_wealth_estate_maintenance.

mnt_wealth_set_estate_maintenance = {
	set_local_variable = { name = mnt_wealth_estate_maintenance_share value = 0 }

	if = {
		limit = {
			var:mnt_wealth_local_raw_power_total > 0
			is_key_in_variable_map = { name = mnt_estate_raw_power target = local_var:mnt_wealth_estate_key }
		}
		set_local_variable = {
			name = mnt_wealth_estate_maintenance_share
			value = {
				value = "variable_map(mnt_estate_raw_power|local_var:mnt_wealth_estate_key)"
				divide = var:mnt_wealth_local_raw_power_total
			}
		}
	}

	set_local_variable = {
		name = mnt_wealth_estate_maintenance
		value = {
			value = modifier:mnt_wealth_local_investment_maintenance
			multiply = local_var:mnt_wealth_estate_maintenance_share
		}
	}
}

# ── MONTHLY INVESTMENT ──

# Scope: location. One estate's monthly investment: spends the rate from
# its wealth, computes its decay and maintenance share, and banks the progress.

mnt_wealth_invest_estate_wealth = {
	mnt_wealth_set_estate_maintenance = yes
	set_local_variable = { name = mnt_wealth_estate_paid value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_estate_paid
			value = {
				value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
				multiply = local_var:mnt_wealth_cached_investment_rate
			}
		}
	}

	change_local_variable = { name = mnt_wealth_invested_running_total add = local_var:mnt_wealth_estate_paid }

	# Subtract the spent amount from this estate's wealth map entry.
	if = {
		limit = { local_var:mnt_wealth_estate_paid > 0 }
		set_local_variable = { name = mnt_wealth_invest_current value = 0 }

		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
			set_local_variable = {
				name = mnt_wealth_invest_current
				value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
			}
		}

		set_local_variable = {
			name = mnt_wealth_invest_after
			value = { value = local_var:mnt_wealth_invest_current subtract = local_var:mnt_wealth_estate_paid }
		}

		add_to_variable_map = {
			name = mnt_wealth_estate_wealth
			key = local_var:mnt_wealth_estate_key
			value = local_var:mnt_wealth_invest_after
		}
	}

	mnt_wealth_compute_estate_decay = {
		ESTATE_KEY = local_var:mnt_wealth_estate_key
		MONTHS = 1
	}

	mnt_wealth_bank_estate_investment_progress = yes
}

# Scope: location. One estate's decay for the period: the total invested
# stock (invested_wealth + invested_since_review) decayed at the live
# per-estate rate. Saves local_var:mnt_wealth_estate_decay.

mnt_wealth_compute_estate_decay = {
	set_local_variable = { name = mnt_wealth_decay_key value = $ESTATE_KEY$ }
	set_local_variable = { name = mnt_wealth_estate_decay value = 0 }

	# Total invested stock: lifetime pool plus progress banked since the review.
	set_local_variable = { name = mnt_wealth_decay_invested value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = local_var:mnt_wealth_decay_key } }
		change_local_variable = { name = mnt_wealth_decay_invested add = "variable_map(mnt_wealth_invested_wealth|local_var:mnt_wealth_decay_key)" }
	}
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_since_review target = local_var:mnt_wealth_decay_key } }
		change_local_variable = { name = mnt_wealth_decay_invested add = "variable_map(mnt_wealth_invested_since_review|local_var:mnt_wealth_decay_key)" }
	}

	# Decay is always toward zero: a neglected stock pays negative decay and
	# recovers at the same rate.
	if = {
		limit = { NOT = { local_var:mnt_wealth_decay_invested = 0 } }

		set_local_variable = { name = mnt_wealth_decay_rate value = 0 }
		mnt_wealth_run_effect_with_estate_name = { EFFECT = mnt_wealth_read_estate_decay_rate }

		set_local_variable = {
			name = mnt_wealth_estate_decay
			value = {
				value = local_var:mnt_wealth_decay_invested
				multiply = local_var:mnt_wealth_decay_rate
				multiply = $MONTHS$
			}
		}
	}
}

# Scope: location. Reads one estate's live decay rate (local plus national)
# into mnt_wealth_decay_rate.

mnt_wealth_read_estate_decay_rate = {
	set_local_variable = {
		name = mnt_wealth_decay_rate
		value = {
			value = modifier:mnt_wealth_local_$ESTATE$_investment_decay
			add = owner.modifier:mnt_wealth_global_$ESTATE$_investment_decay
		}
	}
}

# ── CACHED EXPONENTIATION ──

# Scope: location. Raises local_var:$BASE$ to the power of
# local_var:mnt_wealth_advance_months and writes the result into
# local_var:$RESULT$, using a global cache keyed on the base and exponent.

mnt_wealth_compute_retention_over_period = {
	set_local_variable = {
		name = mnt_wealth_retention_cache_key
		value = {
			value = local_var:mnt_wealth_advance_months
			multiply = 10000000
			add = { value = local_var:$BASE$ multiply = 1000000 }
			floor = yes
		}
	}
	if = {
		limit = { is_key_in_global_variable_map = { name = mnt_wealth_power_cache target = local_var:mnt_wealth_retention_cache_key } }
		set_local_variable = {
			name = $RESULT$
			value = "global_variable_map(mnt_wealth_power_cache|local_var:mnt_wealth_retention_cache_key)"
		}
	}
	else = {
		# Square the base five times: 2, 4, 8, 16.
		set_local_variable = { name = mnt_wealth_power_base_to_2 value = { value = local_var:$BASE$ multiply = local_var:$BASE$ } }
		set_local_variable = { name = mnt_wealth_power_base_to_4 value = { value = local_var:mnt_wealth_power_base_to_2 multiply = local_var:mnt_wealth_power_base_to_2 } }
		set_local_variable = { name = mnt_wealth_power_base_to_8 value = { value = local_var:mnt_wealth_power_base_to_4 multiply = local_var:mnt_wealth_power_base_to_4 } }
		set_local_variable = { name = mnt_wealth_power_base_to_16 value = { value = local_var:mnt_wealth_power_base_to_8 multiply = local_var:mnt_wealth_power_base_to_8 } }

		# Read the exponent in binary, multiplying in the matching square for
		# each set bit. Halving must floor since variables hold fractions.
		set_local_variable = { name = $RESULT$ value = 1 }
		set_local_variable = { name = mnt_wealth_power_exponent_halved value = local_var:mnt_wealth_advance_months }
		set_local_variable = { name = mnt_wealth_power_binary_digit value = { value = local_var:mnt_wealth_power_exponent_halved modulo = 2 } }
		if = {
			limit = { local_var:mnt_wealth_power_binary_digit = 1 }
			change_local_variable = { name = $RESULT$ multiply = local_var:$BASE$ }
		}
		set_local_variable = { name = mnt_wealth_power_exponent_halved value = { value = local_var:mnt_wealth_power_exponent_halved divide = 2 floor = yes } }
		set_local_variable = { name = mnt_wealth_power_binary_digit value = { value = local_var:mnt_wealth_power_exponent_halved modulo = 2 } }
		if = {
			limit = { local_var:mnt_wealth_power_binary_digit = 1 }
			change_local_variable = { name = $RESULT$ multiply = local_var:mnt_wealth_power_base_to_2 }
		}
		set_local_variable = { name = mnt_wealth_power_exponent_halved value = { value = local_var:mnt_wealth_power_exponent_halved divide = 2 floor = yes } }
		set_local_variable = { name = mnt_wealth_power_binary_digit value = { value = local_var:mnt_wealth_power_exponent_halved modulo = 2 } }
		if = {
			limit = { local_var:mnt_wealth_power_binary_digit = 1 }
			change_local_variable = { name = $RESULT$ multiply = local_var:mnt_wealth_power_base_to_4 }
		}
		set_local_variable = { name = mnt_wealth_power_exponent_halved value = { value = local_var:mnt_wealth_power_exponent_halved divide = 2 floor = yes } }
		set_local_variable = { name = mnt_wealth_power_binary_digit value = { value = local_var:mnt_wealth_power_exponent_halved modulo = 2 } }
		if = {
			limit = { local_var:mnt_wealth_power_binary_digit = 1 }
			change_local_variable = { name = $RESULT$ multiply = local_var:mnt_wealth_power_base_to_8 }
		}
		set_local_variable = { name = mnt_wealth_power_exponent_halved value = { value = local_var:mnt_wealth_power_exponent_halved divide = 2 floor = yes } }
		set_local_variable = { name = mnt_wealth_power_binary_digit value = { value = local_var:mnt_wealth_power_exponent_halved modulo = 2 } }
		if = {
			limit = { local_var:mnt_wealth_power_binary_digit = 1 }
			change_local_variable = { name = $RESULT$ multiply = local_var:mnt_wealth_power_base_to_16 }
		}

		add_to_global_variable_map = { name = mnt_wealth_power_cache key = local_var:mnt_wealth_retention_cache_key value = local_var:$RESULT$ }
	}
}

# ── ADVANCE LOCATION TO NOW ──

# Scope: location. Advances the stored state from the review month to now in
# closed form. Per estate: invests the rate from its wealth, then income lands,
# while invested wealth decays and pays its maintenance share. Moves the
# review month forward.

mnt_wealth_advance_location_to_now = {
	set_local_variable = {
		name = mnt_wealth_advance_months
		value = {
			value = global_var:mnt_wealth_months_elapsed
			subtract = var:mnt_wealth_review_month
		}
	}
	clamp_local_variable = { name = mnt_wealth_advance_months min = 0 max = 31 }

	if = {
		limit = { local_var:mnt_wealth_advance_months > 0 }

		# A negative rate spends nothing.
		set_local_variable = { name = mnt_wealth_advance_investment_rate value = { value = mnt_wealth_investment_rate min = 0 } }
		set_local_variable = { name = mnt_wealth_advance_kept_wages      value = var:mnt_wealth_kept_income_total }
		set_local_variable = { name = mnt_wealth_advance_pool_at_review  value = var:mnt_wealth_pool }

		# Each month invests from the wealth on hand, then income lands. The
		# share of wealth left after a month's investment, and after the whole
		# period. Steady income over the period adds up to income times
		# income_over_period once the investment taken from it is counted.
		if = {
			limit = { local_var:mnt_wealth_advance_investment_rate > 0 }
			set_local_variable = {
				name = mnt_wealth_monthly_retention
				value = { value = 1 subtract = local_var:mnt_wealth_advance_investment_rate }
			}

			mnt_wealth_compute_retention_over_period = {
				BASE = mnt_wealth_monthly_retention
				RESULT = mnt_wealth_period_retention
			}

			set_local_variable = {
				name = mnt_wealth_advance_income_over_period
				value = {
					value = 1
					subtract = local_var:mnt_wealth_period_retention
					divide = local_var:mnt_wealth_advance_investment_rate
				}
			}
		}
		else = {
			set_local_variable = { name = mnt_wealth_monthly_retention             value = 1 }
			set_local_variable = { name = mnt_wealth_period_retention value = 1 }
			set_local_variable = { name = mnt_wealth_advance_income_over_period         value = local_var:mnt_wealth_advance_months }
		}

		# Each estate writes its own wealth and progress; the pool is their sum.
		set_local_variable = { name = mnt_wealth_advance_pool_now value = 0 }
		mnt_wealth_for_each_cached_estate = { EFFECT = mnt_wealth_advance_estate }
		set_variable = { name = mnt_wealth_pool value = local_var:mnt_wealth_advance_pool_now }

		mnt_wealth_set_review_month = yes
	}
}

# Scope: location. One estate's advance: its wealth and its invested wealth
# after the elapsed months, written back, with the difference in invested
# wealth banked as progress.

mnt_wealth_advance_estate = {
	set_local_variable = { name = mnt_wealth_advance_wealth_at_review value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_advance_wealth_at_review
			value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
		}
	}

	# The estate's share of the kept wages: its fraction of the pool, or its
	# relative power when the pool is empty.
	set_local_variable = { name = mnt_wealth_advance_estate_share value = 0 }
	if = {
		limit = { local_var:mnt_wealth_advance_pool_at_review > 0 }
		set_local_variable = {
			name = mnt_wealth_advance_estate_share
			value = {
				value = local_var:mnt_wealth_advance_wealth_at_review
				divide = local_var:mnt_wealth_advance_pool_at_review
			}
		}
	}
	else_if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_cached_local_relative_estate_power target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_advance_estate_share
			value = "variable_map(mnt_wealth_cached_local_relative_estate_power|local_var:mnt_wealth_estate_key)"
		}
	}

	# This location's share of the estate's national flow, split by how much
	# of the estate's raw power sits here. Negative when the estate is seizing.
	set_local_variable = { name = mnt_wealth_advance_flow_share value = 0 }
	set_local_variable = { name = mnt_wealth_advance_raw_power value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = mnt_estate_raw_power target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_advance_raw_power
			value = "variable_map(mnt_estate_raw_power|local_var:mnt_wealth_estate_key)"
		}
	}
	owner ?= {
		if = {
			limit = {
				is_key_in_variable_map = { name = mnt_wealth_estate_flow               target = local_var:mnt_wealth_estate_key }
				is_key_in_variable_map = { name = mnt_wealth_national_estate_raw_power target = local_var:mnt_wealth_estate_key }
				"variable_map(mnt_wealth_national_estate_raw_power|local_var:mnt_wealth_estate_key)" > 0
			}
			set_local_variable = {
				name = mnt_wealth_advance_national_raw_power
				value = "variable_map(mnt_wealth_national_estate_raw_power|local_var:mnt_wealth_estate_key)"
			}
			set_local_variable = {
				name = mnt_wealth_advance_flow_share
				value = {
					value = "variable_map(mnt_wealth_estate_flow|local_var:mnt_wealth_estate_key)"
					multiply = local_var:mnt_wealth_advance_raw_power
					divide = local_var:mnt_wealth_advance_national_raw_power
				}
			}
		}
	}

	# The estate's maintenance share, from its raw power fraction.
	mnt_wealth_set_estate_maintenance = yes

	# Monthly income: the wage share plus the flow share.
	set_local_variable = {
		name = mnt_wealth_advance_estate_income
		value = {
			value = local_var:mnt_wealth_advance_estate_share
			multiply = local_var:mnt_wealth_advance_kept_wages
			add = local_var:mnt_wealth_advance_flow_share
		}
	}
	# Wealth now: what was there decays by the investment taken from it,
	# and the period's income adds up on top.
	set_local_variable = {
		name = mnt_wealth_advance_wealth_now
		value = {
			value = local_var:mnt_wealth_advance_wealth_at_review
			multiply = local_var:mnt_wealth_period_retention
			add = {
				value = local_var:mnt_wealth_advance_estate_income
				multiply = local_var:mnt_wealth_advance_income_over_period
			}
		}
	}

	# Invested wealth at the review, lifetime plus banked progress.
	set_local_variable = { name = mnt_wealth_advance_invested_pool_at_review value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = local_var:mnt_wealth_estate_key } }
		change_local_variable = { name = mnt_wealth_advance_invested_pool_at_review add = "variable_map(mnt_wealth_invested_wealth|local_var:mnt_wealth_estate_key)" }
	}
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_since_review target = local_var:mnt_wealth_estate_key } }
		change_local_variable = { name = mnt_wealth_advance_invested_pool_at_review add = "variable_map(mnt_wealth_invested_since_review|local_var:mnt_wealth_estate_key)" }
	}

	# The share of invested wealth left after a month's decay, and after the
	# whole period. Decay is toward zero for either sign.
	set_local_variable = { name = mnt_wealth_decay_rate value = 0 }
	switch = {
		trigger = local_var:mnt_wealth_estate_key
		estate_type:nobles_estate = { set_local_variable = { name = mnt_wealth_decay_rate value = { value = modifier:mnt_wealth_local_nobles_investment_decay add = owner.modifier:mnt_wealth_global_nobles_investment_decay } } }
		estate_type:clergy_estate = { set_local_variable = { name = mnt_wealth_decay_rate value = { value = modifier:mnt_wealth_local_clergy_investment_decay add = owner.modifier:mnt_wealth_global_clergy_investment_decay } } }
		estate_type:burghers_estate = { set_local_variable = { name = mnt_wealth_decay_rate value = { value = modifier:mnt_wealth_local_burghers_investment_decay add = owner.modifier:mnt_wealth_global_burghers_investment_decay } } }
		estate_type:peasants_estate = { set_local_variable = { name = mnt_wealth_decay_rate value = { value = modifier:mnt_wealth_local_peasants_investment_decay add = owner.modifier:mnt_wealth_global_peasants_investment_decay } } }
		estate_type:dhimmi_estate = { set_local_variable = { name = mnt_wealth_decay_rate value = { value = modifier:mnt_wealth_local_dhimmi_investment_decay add = owner.modifier:mnt_wealth_global_dhimmi_investment_decay } } }
		estate_type:tribes_estate = { set_local_variable = { name = mnt_wealth_decay_rate value = { value = modifier:mnt_wealth_local_tribes_investment_decay add = owner.modifier:mnt_wealth_global_tribes_investment_decay } } }
		estate_type:cossacks_estate = { set_local_variable = { name = mnt_wealth_decay_rate value = { value = modifier:mnt_wealth_local_cossacks_investment_decay add = owner.modifier:mnt_wealth_global_cossacks_investment_decay } } }
	}
	set_local_variable = {
		name = mnt_wealth_invested_monthly_retention
		value = {
			value = 1
			subtract = {
				value = local_var:mnt_wealth_decay_rate
				multiply = mnt_wealth_decay_resistance_multiplier
			}
		}
	}
	mnt_wealth_compute_retention_over_period = {
		BASE = mnt_wealth_invested_monthly_retention
		RESULT = mnt_wealth_invested_period_retention
	}

	# Each month invests the rate times the wealth on hand: a flat part, the
	# income, plus a part that slopes with the wealth as it is spent down.
	# Both are summed over the period with the decay each month's investment
	# has suffered since. Maintenance is a flat monthly charge summed the same
	# way as the flat part.
	set_local_variable = { name = mnt_wealth_advance_flat_inflow_sum value = local_var:mnt_wealth_advance_months }
	set_local_variable = {
		name = mnt_wealth_advance_invested_pool_change
		value = { value = 1 subtract = local_var:mnt_wealth_invested_monthly_retention }
	}
	if = {
		limit = { NOT = { local_var:mnt_wealth_advance_invested_pool_change = 0 } }
		set_local_variable = {
			name = mnt_wealth_advance_flat_inflow_sum
			value = {
				value = 1
				subtract = local_var:mnt_wealth_invested_period_retention
				divide = local_var:mnt_wealth_advance_invested_pool_change
			}
		}
	}

	set_local_variable = {
		name = mnt_wealth_advance_remaining_gap
		value = { value = local_var:mnt_wealth_invested_monthly_retention subtract = local_var:mnt_wealth_monthly_retention }
	}
	if = {
		limit = { NOT = { local_var:mnt_wealth_advance_remaining_gap = 0 } }
		set_local_variable = {
			name = mnt_wealth_advance_sloped_inflow_sum
			value = {
				value = local_var:mnt_wealth_invested_period_retention
				subtract = local_var:mnt_wealth_period_retention
				divide = local_var:mnt_wealth_advance_remaining_gap
			}
		}
	}
	else = {
		# Both shares equal: the sum is months times the share over the period.
		set_local_variable = {
			name = mnt_wealth_advance_sloped_inflow_sum
			value = {
				value = local_var:mnt_wealth_advance_months
				multiply = local_var:mnt_wealth_period_retention
				divide = local_var:mnt_wealth_monthly_retention
			}
		}
	}

	set_local_variable = {
		name = mnt_wealth_advance_inflow_slope
		value = {
			value = local_var:mnt_wealth_advance_investment_rate
			multiply = local_var:mnt_wealth_advance_wealth_at_review
			subtract = local_var:mnt_wealth_advance_estate_income
		}
	}

	set_local_variable = {
		name = mnt_wealth_advance_invested_pool_now
		value = {
			value = local_var:mnt_wealth_advance_invested_pool_at_review
			multiply = local_var:mnt_wealth_invested_period_retention
			add = {
				value = local_var:mnt_wealth_advance_estate_income
				multiply = local_var:mnt_wealth_advance_flat_inflow_sum
				add = {
					value = local_var:mnt_wealth_advance_inflow_slope
					multiply = local_var:mnt_wealth_advance_sloped_inflow_sum
				}
				multiply = mnt_wealth_investment_efficiency_multiplier
			}
			subtract = {
				value = local_var:mnt_wealth_estate_maintenance
				multiply = local_var:mnt_wealth_advance_flat_inflow_sum
			}
		}
	}

	# Progress banked since the review: the change in invested wealth, plus
	# any seizure past the estate's wealth, which comes out of invested wealth.
	set_local_variable = {
		name = mnt_wealth_advance_progress
		value = { value = local_var:mnt_wealth_advance_invested_pool_now subtract = local_var:mnt_wealth_advance_invested_pool_at_review }
	}
	if = {
		limit = { local_var:mnt_wealth_advance_wealth_now < 0 }
		change_local_variable = { name = mnt_wealth_advance_progress add = local_var:mnt_wealth_advance_wealth_now }
		set_local_variable = { name = mnt_wealth_advance_wealth_now value = 0 }
	}
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_invested_since_review target = local_var:mnt_wealth_estate_key } }
		change_local_variable = { name = mnt_wealth_advance_progress add = "variable_map(mnt_wealth_invested_since_review|local_var:mnt_wealth_estate_key)" }
	}
	add_to_variable_map = {
		name = mnt_wealth_invested_since_review
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_advance_progress
	}

	add_to_variable_map = {
		name = mnt_wealth_estate_wealth
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_advance_wealth_now
	}
	change_local_variable = { name = mnt_wealth_advance_pool_now add = local_var:mnt_wealth_advance_wealth_now }
}
