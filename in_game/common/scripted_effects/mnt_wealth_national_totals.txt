# GDP and Wages: national totals maintenance and owner-change transfer.
#
# A country carries two running sums: GDP as a single number, and raw power as
# a map with one entry per estate. Every location adds its contribution exactly
# once and takes it back off exactly once. The owner-change workers move both
# totals between countries in the same tick.

# ── NATIONAL GDP ──

# Scope: location. Takes this location's old GDP back off its owner's national
# total, so the recompute that follows adds the new value to a clean total.

mnt_wealth_remove_local_gdp_from_national_total = {
	save_temporary_scope_as = mnt_wealth_gdp_total_location

	owner ?= {
		if = {
			limit = { has_variable = mnt_wealth_national_gdp_total }
			change_variable = {
				name = mnt_wealth_national_gdp_total
				subtract = scope:mnt_wealth_gdp_total_location.var:mnt_wealth_local_gdp
			}
			change_variable = {
				name = mnt_wealth_national_rural_gdp
				subtract = scope:mnt_wealth_gdp_total_location.var:mnt_wealth_local_rural_gdp
			}
			change_variable = {
				name = mnt_wealth_national_urban_gdp
				subtract = scope:mnt_wealth_gdp_total_location.var:mnt_wealth_local_urban_gdp
			}
			change_variable = {
				name = mnt_wealth_national_urban_ndp
				subtract = scope:mnt_wealth_gdp_total_location.var:mnt_wealth_local_urban_ndp
			}
		}
	}
}

# Scope: location. Adds this location's fresh GDP to its owner's national total.

mnt_wealth_add_local_gdp_to_national_total = {
	save_temporary_scope_as = mnt_wealth_gdp_total_location

	owner ?= {
		if = {
			limit = { NOT = { has_variable = mnt_wealth_national_gdp_total } }
			set_variable = { name = mnt_wealth_national_gdp_total value = 0 }
			set_variable = { name = mnt_wealth_national_rural_gdp value = 0 }
			set_variable = { name = mnt_wealth_national_urban_gdp value = 0 }
			set_variable = { name = mnt_wealth_national_urban_ndp value = 0 }
		}
		change_variable = {
			name = mnt_wealth_national_gdp_total
			add = scope:mnt_wealth_gdp_total_location.var:mnt_wealth_local_gdp
		}
		change_variable = {
			name = mnt_wealth_national_rural_gdp
			add = scope:mnt_wealth_gdp_total_location.var:mnt_wealth_local_rural_gdp
		}
		change_variable = {
			name = mnt_wealth_national_urban_gdp
			add = scope:mnt_wealth_gdp_total_location.var:mnt_wealth_local_urban_gdp
		}
		change_variable = {
			name = mnt_wealth_national_urban_ndp
			add = scope:mnt_wealth_gdp_total_location.var:mnt_wealth_local_urban_ndp
		}
	}
}

# ── NATIONAL ESTATE RAW POWER ──
#
# mnt_wealth_national_estate_raw_power is a country map: per estate, the sum of that
# estate's raw power across every owned location. National estate investment is
# split across an estate's locations by each location's share of this total.

# Scope: location. scope:mnt_wealth_raw_power_country is the country whose map to
# change. Moves this location's per-estate raw power into that map: SIGN = 1
# adds it, SIGN = -1 takes it back out.
#
# Every scope this reads is saved and named. prev is the scope this effect was
# called from, which for anything reached through the load balancer is an
# unrelated worker country, so it can never be used to get back to the location.

mnt_wealth_apply_national_estate_raw_power = {
	save_temporary_scope_as = mnt_wealth_raw_power_location

	every_key_in_variable_map = {
		variable = mnt_estate_raw_power
		save_temporary_scope_as = mnt_wealth_raw_power_key

		scope:mnt_wealth_raw_power_country = {
			set_local_variable = {
				name = mnt_wealth_raw_power_delta
				value = {
					value = "scope:mnt_wealth_raw_power_location.variable_map(mnt_estate_raw_power|scope:mnt_wealth_raw_power_key)"
					multiply = $SIGN$
				}
			}

			# A country that has never held this estate has no entry yet.
			set_local_variable = { name = mnt_wealth_raw_power_running value = 0 }

			if = {
				limit = {
					is_key_in_variable_map = { name = mnt_wealth_national_estate_raw_power target = scope:mnt_wealth_raw_power_key }
				}
				set_local_variable = {
					name = mnt_wealth_raw_power_running
					value = "variable_map(mnt_wealth_national_estate_raw_power|scope:mnt_wealth_raw_power_key)"
				}
			}

			change_local_variable = { name = mnt_wealth_raw_power_running add = local_var:mnt_wealth_raw_power_delta }

			add_to_variable_map = {
				name = mnt_wealth_national_estate_raw_power
				key = scope:mnt_wealth_raw_power_key
				value = local_var:mnt_wealth_raw_power_running
			}
		}
	}
}

# Scope: country. Fills the map from scratch: one entry per estate, holding that
# estate's raw power summed across every owned location.

mnt_wealth_rebuild_national_estate_raw_power = {
	clear_variable_map = mnt_wealth_national_estate_raw_power
	save_temporary_scope_as = mnt_wealth_raw_power_country
	every_owned_location = { mnt_wealth_apply_national_estate_raw_power = { SIGN = 1 } }
}

# ── OWNER CHANGE ──
#
# The national totals are running sums, so a transfer takes this location off the
# old owner and puts it on the new one in the same tick. The location's own
# figures do not change in between, so both sides move the same numbers and the
# totals stay exact. A new owner that forbids one of the estates keeps the old
# owner's figures until its next survey, which takes them off and puts its own on.

# Scope: location. Moves this location's GDP and estate raw power off the old
# owner and onto the new one. New land is zeroed first.

mnt_wealth_transfer_location_between_owners = {
	# Unowned land is never visited by the yearly pass, so land gaining its
	# first owner gets its full variable set here.
	if = {
		limit = { NOT = { exists = scope:loser } }
		mnt_wealth_zero_location = yes
	}

	mnt_wealth_move_location_gdp_between_owners = yes
	mnt_wealth_move_location_estate_power_between_owners = yes
}

# Scope: location. Subtracts the location's GDP from the old owner and adds it
# to the new owner.
#
# When a country is formed by this very transfer, the engine gives it locations
# before mnt_wealth_initialize_country runs, so neither side has a total yet.
# Both sides start at zero here; the full recount a moment later replaces them.

mnt_wealth_move_location_gdp_between_owners = {
	save_temporary_scope_as = mnt_wealth_moving_location

	scope:loser ?= {
		if = {
			limit = { has_variable = mnt_wealth_national_gdp_total }
			change_variable = {
				name = mnt_wealth_national_gdp_total
				subtract = scope:mnt_wealth_moving_location.var:mnt_wealth_local_gdp
			}
			change_variable = {
				name = mnt_wealth_national_rural_gdp
				subtract = scope:mnt_wealth_moving_location.var:mnt_wealth_local_rural_gdp
			}
			change_variable = {
				name = mnt_wealth_national_urban_gdp
				subtract = scope:mnt_wealth_moving_location.var:mnt_wealth_local_urban_gdp
			}
			change_variable = {
				name = mnt_wealth_national_urban_ndp
				subtract = scope:mnt_wealth_moving_location.var:mnt_wealth_local_urban_ndp
			}
		}
	}

	scope:winner ?= {
		if = {
			limit = { NOT = { has_variable = mnt_wealth_national_gdp_total } }
			set_variable = { name = mnt_wealth_national_gdp_total value = 0 }
			set_variable = { name = mnt_wealth_national_rural_gdp value = 0 }
			set_variable = { name = mnt_wealth_national_urban_gdp value = 0 }
			set_variable = { name = mnt_wealth_national_urban_ndp value = 0 }
		}
		change_variable = {
			name = mnt_wealth_national_gdp_total
			add = scope:mnt_wealth_moving_location.var:mnt_wealth_local_gdp
		}
		change_variable = {
			name = mnt_wealth_national_rural_gdp
			add = scope:mnt_wealth_moving_location.var:mnt_wealth_local_rural_gdp
		}
		change_variable = {
			name = mnt_wealth_national_urban_gdp
			add = scope:mnt_wealth_moving_location.var:mnt_wealth_local_urban_gdp
		}
		change_variable = {
			name = mnt_wealth_national_urban_ndp
			add = scope:mnt_wealth_moving_location.var:mnt_wealth_local_urban_ndp
		}
	}
}

# Scope: location. Takes this location's estate raw power off the old owner's
# map and puts it on the new owner's.

mnt_wealth_move_location_estate_power_between_owners = {
	if = {
		limit = { exists = scope:loser }
		scope:loser = { save_temporary_scope_as = mnt_wealth_raw_power_country }
		mnt_wealth_apply_national_estate_raw_power = { SIGN = -1 }
	}

	if = {
		limit = { exists = scope:winner }
		scope:winner = { save_temporary_scope_as = mnt_wealth_raw_power_country }
		mnt_wealth_apply_national_estate_raw_power = { SIGN = 1 }
	}
}
