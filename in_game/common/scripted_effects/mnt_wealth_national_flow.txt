# GDP and Wages: gold moving between estate treasuries and their locations.
#
# One signed number per estate carries both directions: above its savings goal
# an estate sends gold down to its locations, below it takes gold back out.

# ── TREASURY DRIP ──

# Scope: country. Called monthly for AI through the EPBM loop. Combines the
# EPBM building maintenance charge and the savings valve flow into one gold
# move per estate, so AI treasuries change every month instead of in yearly
# lumps.
#
# Sign: delta = -(epbm_pay) - (flow_rate).
# Positive flow = payout = treasury drains; negative = seizure = treasury gains.

mnt_change_estate_gold = {
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = mnt_wealth_drip_estate

		prev = {
			set_local_variable = { name = mnt_wealth_drip_net value = 0 }

			# EPBM building maintenance from the pay variable map.
			if = {
				limit = {
					has_variable_map = epbm_pay
					is_key_in_variable_map = { name = epbm_pay target = scope:mnt_wealth_drip_estate }
					"variable_map(epbm_pay|scope:mnt_wealth_drip_estate)" > 0
				}
				set_local_variable = { name = mnt_wealth_drip_epbm value = "variable_map(epbm_pay|scope:mnt_wealth_drip_estate)" }
				change_local_variable = { name = mnt_wealth_drip_net subtract = local_var:mnt_wealth_drip_epbm }
			}

			# Savings valve flow rate (per month). The map is keyed by
			# estate_type, so the key goes through a local.
			set_local_variable = { name = mnt_wealth_drip_key value = scope:mnt_wealth_drip_estate.estate_type }
			if = {
				limit = {
					is_key_in_variable_map = {
						name = mnt_wealth_estate_flow
						target = local_var:mnt_wealth_drip_key
					}
				}
				set_local_variable = { name = mnt_wealth_drip_flow value = "variable_map(mnt_wealth_estate_flow|local_var:mnt_wealth_drip_key)" }
				change_local_variable = { name = mnt_wealth_drip_net subtract = local_var:mnt_wealth_drip_flow }
			}

			if = {
				limit = { local_var:mnt_wealth_drip_net != 0 }
				add_gold_to_estate = {
					estate_type = scope:mnt_wealth_drip_estate.estate_type
					value = local_var:mnt_wealth_drip_net
				}
			}
		}
	}
}

# ── NATIONAL SAVINGS VALVES ──

# Scope: country. MONTHS = 1 for the monthly player path, 12 for the yearly AI path.
# Computes each estate's savings gap and writes the per-month flow rate to the
# mnt_wealth_estate_flow map. Gold moves to the estate treasury on the player
# path only; AI treasuries are handled by the monthly drip in mnt_change_estate_gold.

mnt_wealth_set_national_savings_valves = {
	set_local_variable = { name = mnt_wealth_country_economical_base value = country_economical_base }
	set_local_variable = { name = mnt_wealth_valve_months value = $MONTHS$ }

	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = mnt_wealth_current_estate
		set_local_variable = { name = mnt_wealth_estate_key value = estate_type }

		prev = {
			set_local_variable = {
				name = mnt_wealth_national_savings_goal
				value = {
					value = local_var:mnt_wealth_country_economical_base
					multiply = "estate_power(scope:mnt_wealth_current_estate.estate_type)"
					min = 1
				}
			}

			# Last flow for this estate, scaled by the pass length.
			set_local_variable = { name = mnt_wealth_last_flow value = 0 }

			if = {
				limit = { is_key_in_variable_map = { name = mnt_wealth_estate_flow target = local_var:mnt_wealth_estate_key } }
				set_local_variable = {
					name = mnt_wealth_last_flow
					value = "variable_map(mnt_wealth_estate_flow|local_var:mnt_wealth_estate_key)"
				}
			}

			set_local_variable = {
				name = mnt_wealth_last_flow_scaled
				value = { value = local_var:mnt_wealth_last_flow min = 0 multiply = local_var:mnt_wealth_valve_months }
			}

			# How far under its effective goal the treasury is.
			set_local_variable = {
				name = mnt_wealth_estate_gold_gap
				value = {
					value = local_var:mnt_wealth_national_savings_goal
					subtract = scope:mnt_wealth_current_estate.estate_gold
					subtract = local_var:mnt_wealth_last_flow_scaled
				}
			}

			mnt_wealth_update_estate_seizure_demand = { MONTHS = $MONTHS$ }

			set_local_variable = {
				name = mnt_wealth_payout_cap
				value = { value = local_var:mnt_wealth_country_economical_base multiply = mnt_wealth_national_investment_limit_factor }
			}

			set_local_variable = {
				name = mnt_wealth_seizure_cap
				value = {
					value = mnt_wealth_estate_seizure_limit_factor
					add = local_var:mnt_wealth_seizure_demand
					multiply = local_var:mnt_wealth_national_savings_goal
				}
			}

			# Period total: the full amount moved this pass, clamped once.
			set_local_variable = { name = mnt_wealth_estate_period_total value = 0 }

			if = {
				limit = {
					is_key_in_variable_map = { name = mnt_wealth_national_estate_raw_power target = local_var:mnt_wealth_estate_key }
					"variable_map(mnt_wealth_national_estate_raw_power|local_var:mnt_wealth_estate_key)" > 0
				}
				if = {
					limit = { local_var:mnt_wealth_estate_gold_gap > 0 }
					set_local_variable = {
						name = mnt_wealth_estate_period_total
						value = {
							value = local_var:mnt_wealth_estate_gold_gap
							multiply = $MONTHS$
							divide = 12
							multiply = { value = 1 add = local_var:mnt_wealth_seizure_demand }
							max = local_var:mnt_wealth_estate_gold_gap
							max = { value = local_var:mnt_wealth_seizure_cap multiply = $MONTHS$ }
							multiply = -1
						}
					}
				}
				else = {
					set_local_variable = {
						name = mnt_wealth_estate_period_total
						value = {
							value = local_var:mnt_wealth_estate_gold_gap
							multiply = -1
							multiply = $MONTHS$
							divide = 12
							max = { value = local_var:mnt_wealth_payout_cap multiply = $MONTHS$ }
						}
					}
				}
			}

			# The map holds the per-month rate.
			set_local_variable = {
				name = mnt_wealth_estate_monthly_flow
				value = { value = local_var:mnt_wealth_estate_period_total divide = $MONTHS$ }
			}
			add_to_variable_map = {
				name = mnt_wealth_estate_flow
				key = local_var:mnt_wealth_estate_key
				value = local_var:mnt_wealth_estate_monthly_flow
			}

			# Player path applies gold here; AI treasuries move through the
			# monthly drip (mnt_change_estate_gold) instead.
			if = {
				limit = { local_var:mnt_wealth_valve_months = 1 }
				add_gold_to_estate = {
					estate_type = scope:mnt_wealth_current_estate.estate_type
					value = { value = local_var:mnt_wealth_estate_period_total multiply = -1 }
				}
			}
		}
	}
}

# Scope: country. Called by mnt_wealth_set_national_savings_valves once per estate.
# Tracks the estate's accumulated demand to seize, which grows each month the
# treasury sits under its goal and decays once the treasury recovers.
# Reads local_var:mnt_wealth_estate_gold_gap and local_var:mnt_wealth_national_savings_goal.
# Saves local_var:mnt_wealth_seizure_demand.

mnt_wealth_update_estate_seizure_demand = {
	set_local_variable = { name = mnt_wealth_seizure_demand value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_seizure_demand target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_seizure_demand
			value = "variable_map(mnt_wealth_estate_seizure_demand|local_var:mnt_wealth_estate_key)"
		}
	}

	if = {
		limit = { local_var:mnt_wealth_estate_gold_gap > 0 }
		change_local_variable = {
			name = mnt_wealth_seizure_demand
			add = {
				value = local_var:mnt_wealth_estate_gold_gap
				divide = local_var:mnt_wealth_national_savings_goal
				multiply = $MONTHS$
			}
		}
	}
	else = {
		# Monthly and yearly passes each have their own decay tuning value.
		set_local_variable = { name = mnt_wealth_demand_decay value = mnt_wealth_seizure_demand_decay }
		if = {
			limit = { local_var:mnt_wealth_valve_months > 1 }
			set_local_variable = { name = mnt_wealth_demand_decay value = mnt_wealth_seizure_demand_decay_yearly }
		}
		set_local_variable = {
			name = mnt_wealth_seizure_demand
			value = { value = local_var:mnt_wealth_seizure_demand multiply = local_var:mnt_wealth_demand_decay }
		}
	}

	add_to_variable_map = {
		name = mnt_wealth_estate_seizure_demand
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_seizure_demand
	}
}

# ── ONE LOCATION'S SHARE ──

# Scope: location. Called by mnt_wealth_resolve_estate_monthly_income and
# mnt_wealth_apply_national_flow_to_location.
# Saves local_var:mnt_wealth_gold_returned: one month of the estate's national flow at
# this location, the estate's payout split by this location's share of the
# estate's national raw power. Reads local_var:mnt_wealth_estate_key set by the caller.

mnt_wealth_set_estate_national_flow_share = {
	set_local_variable = { name = mnt_wealth_gold_returned value = 0 }
	# This location's raw power for the estate.
	set_local_variable = { name = mnt_wealth_local_raw_power value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_estate_raw_power target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_local_raw_power
			value = "variable_map(mnt_estate_raw_power|local_var:mnt_wealth_estate_key)"
		}
	}

	owner ?= {
		if = {
			limit = {
				is_key_in_variable_map = { name = mnt_wealth_estate_flow               target = local_var:mnt_wealth_estate_key }
				is_key_in_variable_map = { name = mnt_wealth_national_estate_raw_power target = local_var:mnt_wealth_estate_key }
				"variable_map(mnt_wealth_national_estate_raw_power|local_var:mnt_wealth_estate_key)" > 0
			}
			set_local_variable = {
				name = mnt_wealth_national_raw_power
				value = "variable_map(mnt_wealth_national_estate_raw_power|local_var:mnt_wealth_estate_key)"
			}

			# The national total already counts this location, so it can never be
			# smaller than this location's share.
			if = {
				limit = { local_var:mnt_wealth_local_raw_power > local_var:mnt_wealth_national_raw_power }
				error_log = "MNT_WEALTH-INVARIANT national estate raw power is smaller than this one location's share local=[SCOPE.GetLocalVariable('mnt_wealth_local_raw_power').GetValue|4] national=[SCOPE.GetLocalVariable('mnt_wealth_national_raw_power').GetValue|4]"
			}

			set_local_variable = {
				name = mnt_wealth_gold_returned
				value = {
					value = "variable_map(mnt_wealth_estate_flow|local_var:mnt_wealth_estate_key)"
					multiply = local_var:mnt_wealth_local_raw_power
					divide = local_var:mnt_wealth_national_raw_power
				}
			}
		}
	}
}

# Scope: location. Called by mnt_wealth_resolve_estate_monthly_income and
# mnt_wealth_apply_national_flow_to_location.
# Applies one estate's signed flow for this location. Reads
# local_var:mnt_wealth_gold_returned (positive invests, negative seizes) and saves
# local_var:mnt_wealth_pool_delta, the gold to add to or take from the pool.
# When seizing, the estate's own liquid pool wealth is taken first; whatever
# that cannot cover is ripped out of its invested wealth, which lowers its
# investment level toward neglect.

mnt_wealth_apply_estate_flow = {
	if = {
		limit = { local_var:mnt_wealth_gold_returned >= 0 }
		set_local_variable = { name = mnt_wealth_pool_delta value = local_var:mnt_wealth_gold_returned }
	}
	else = {
		set_local_variable = { name = mnt_wealth_seizing value = { value = local_var:mnt_wealth_gold_returned multiply = -1 } }
		# The estate's liquid wealth caps what can be seized from it.
		set_local_variable = { name = mnt_wealth_own_liquid value = 0 }

		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
			set_local_variable = {
				name = mnt_wealth_own_liquid
				value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
			}
		}

		set_local_variable = {
			name = mnt_wealth_from_liquid
			value = { value = local_var:mnt_wealth_seizing max = local_var:mnt_wealth_own_liquid }
		}
		set_local_variable = { name = mnt_wealth_pool_delta value = { value = local_var:mnt_wealth_from_liquid multiply = -1 } }
		# What the liquid share could not cover comes out of invested wealth.
		set_local_variable = {
			name = mnt_wealth_from_invested
			value = { value = local_var:mnt_wealth_seizing subtract = local_var:mnt_wealth_from_liquid }
		}

		set_local_variable = {
			name = mnt_wealth_investment_change
			value = { value = local_var:mnt_wealth_from_invested multiply = -1 }
		}

		mnt_wealth_add_investment_progress = yes
	}
}

# ── APPLYING THE FLOW ──

# Scope: location. Called by mnt_wealth_estate_month through
# mnt_wealth_for_each_cached_estate.
# Gold arriving for one estate this month: the kept wages from
# mnt_wealth_set_estate_wage_balance plus this location's share of the estate's
# national flow (negative when the estate is seizing).

mnt_wealth_resolve_estate_monthly_income = {
	# One month of the estate's national flow at this location.
	mnt_wealth_set_estate_national_flow_share = yes
	mnt_wealth_apply_estate_flow = yes

	set_local_variable = {
		name = mnt_wealth_estate_arriving
		value = {
			value = "local_variable_map(mnt_wealth_kept_wages|local_var:mnt_wealth_estate_key)"
			add = local_var:mnt_wealth_pool_delta
		}
	}

	# Write directly to the estate wealth map.
	set_local_variable = { name = mnt_wealth_arriving_current value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_arriving_current
			value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
		}
	}

	set_local_variable = {
		name = mnt_wealth_arriving_updated
		value = { value = local_var:mnt_wealth_arriving_current add = local_var:mnt_wealth_estate_arriving }
	}

	add_to_variable_map = {
		name = mnt_wealth_estate_wealth
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_arriving_updated
	}

	change_local_variable = { name = mnt_wealth_arriving_running_total add = local_var:mnt_wealth_estate_arriving }
}
