# Scope: location. Moves $amount$ pop units (thousands, same as pop_size)
# out of every $from$ pop into $to$ pops by splitting each source pop by the
# same fraction, so the mix of culture and religion is kept. Capped at 1.
#
# mnt_promote_pops = { from = pop_type:peasants to = pop_type:laborers amount = 1 }
mnt_promote_pops = {
	set_local_variable = { name = mnt_promote_source_size value = 0 }
	every_pop = {
		limit = { pop_type = $from$ }
		change_local_variable = { name = mnt_promote_source_size add = pop_size }
	}

	set_local_variable = { name = mnt_promote_fraction value = 0 }
	if = {
		limit = { local_var:mnt_promote_source_size > 0 }
		set_local_variable = {
			name = mnt_promote_fraction
			value = {
				value = $amount$
				divide = local_var:mnt_promote_source_size
				max = 1
			}
		}
	}

	if = {
		limit = { local_var:mnt_promote_fraction > 0 }
		save_temporary_scope_value_as = { name = mnt_promote_fraction value = local_var:mnt_promote_fraction }
		every_pop = {
			limit = { pop_type = $from$ }
			split_pop = {
				fraction = scope:mnt_promote_fraction
				type = $to$
			}
		}
	}
}

# Scope: location. Promotes peasants into laborers and burghers to fill open jobs. Setup can and does handle most of this
# but any place where we need to promote slaves is a problem that we need to sort out manually, along with any building setup
# that can't be done by setup. Even though this gives enough pops to fill these jobs, they won't actually be worked until the 
# following month.
mnt_fill_building_jobs = {
	set_local_variable = { name = mnt_fill_laborer_jobs value = 0 }
	set_local_variable = { name = mnt_fill_burgher_jobs value = 0 }
	every_buildings_in_location = {
		limit = { building_pop_type = laborers }
		change_local_variable = { name = mnt_fill_laborer_jobs add = building_employment_size_amount }
	}
	every_buildings_in_location = {
		limit = { building_pop_type = burghers }
		change_local_variable = { name = mnt_fill_burgher_jobs add = building_employment_size_amount }
	}
	# num_pop_type is cached per tick, so use iterate-and-sum script values instead;
	# only matters at game start where change_pop_type runs on the same day
	change_local_variable = { name = mnt_fill_laborer_jobs subtract = mnt_count_laborers }
	change_local_variable = { name = mnt_fill_burgher_jobs subtract = mnt_count_burghers }

	if = {
		limit = { local_var:mnt_fill_laborer_jobs > 0 }
		mnt_promote_pops = { from = pop_type:peasants to = pop_type:laborers amount = local_var:mnt_fill_laborer_jobs }
	}
	if = {
		limit = { local_var:mnt_fill_burgher_jobs > 0 }
		mnt_promote_pops = { from = pop_type:peasants to = pop_type:burghers amount = local_var:mnt_fill_burgher_jobs }
	}
}

# Scope: location. Sets pasture and hunting ground levels to their caps.
mnt_tribes_refresh_buildings = {

			floor = yes
		}
	}
}
