# The market half of the cost of the court and diplomatic spending systems. Their country events accumulate
# gold on each market center every month; these effects run once per market from the wealth dispatcher's
# monthly market walk, through mnt_monthly_instant_market_pulse, and turn that gold into basket demand.

# Scope: market. Called by mnt_monthly_instant_market_pulse.
# Prices one unit of the court demand basket, converts the gold accumulated
# on the market center by cost_of_the_court.1 into that many basket units,
# and resets the accumulator for next month.

apply_cost_of_the_court_demand = {
	# Calculate current price corresponding to 1 unit of the demand basket
	set_local_variable = {
		name = court_demand_basket_price
		value = {
			value = 0

			add = {
				# 1 furniture
				value = 1
				multiply = "market_price(goods:furniture)"
			}
			add = {
				# 0.25 jewelry
				value = 0.25
				multiply = "market_price(goods:jewelry)"
			}
			add = {
				# 0.5 glass
				value = 0.5
				multiply = "market_price(goods:glass)"
			}
			add = {
				# 1 fine_cloth
				value = 1
				multiply = "market_price(goods:fine_cloth)"
			}
			add = {
				# 1 cloth
				value = 1
				multiply = "market_price(goods:cloth)"
			}
			add = {
				# 1 leather
				value = 1
				multiply = "market_price(goods:leather)"
			}
			add = {
				# 1 weaponry
				value = 1
				multiply = "market_price(goods:weaponry)"
			}
			add = {
				# 0.5 books
				value = 0.5
				multiply = "market_price(goods:books)"
			}
			add = {
				# 1 paper
				value = 1
				multiply = "market_price(goods:paper)"
			}
		}
	}

	if = {
		# In the first month, there are ~10 markets that do not have this variable stored. Removing the condition leads to errors
		limit = { location = { has_variable = total_cost_of_the_court } }
		# Switching between two temporary demands
		# If we use only one temporary demand, the UI will not be updated properly
		if = {
			limit = { has_temporary_demand = demand:cost_of_the_court_demand1 }
			remove_temporary_demand = demand:cost_of_the_court_demand1

			add_temporary_demand = {
				type = demand:cost_of_the_court_demand2
				months = -1
				scale = { value = "location.var:total_cost_of_the_court" divide = "local_var:court_demand_basket_price" }
			}
		}
		else = {
			remove_temporary_demand = demand:cost_of_the_court_demand2
			add_temporary_demand = {
				type = demand:cost_of_the_court_demand1
				months = -1
				scale = { value = "location.var:total_cost_of_the_court" divide = "local_var:court_demand_basket_price" }
			}
		}

		# Clear total CotC value from the last month
		location = { set_variable = { name = total_cost_of_the_court value = 0 } }
	}
}
