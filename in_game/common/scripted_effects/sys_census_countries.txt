run_logging_credit = {
	if = { limit = { has_global_variable = is_logging_yearly_credit_information }
		print_country_credit_information = yes
	}
	error_log = "::Done logging country credit information for this year."
}

	every_country = {
		limit = { num_locations > 0 }
		set_local_variable = { name = cr_balance value = mnt_credit_tooltip_market_balance }

	remove_local_variable = cr_supply
