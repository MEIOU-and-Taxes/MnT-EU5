# Scope: none. Zeroes location wealth system variables on every ownable location
# and gives the owned ones their opening schedule.

mnt_wealth_seed_owned_locations = {
	set_global_variable = { name = mnt_wealth_location_serial value = 0 }

	every_ownable_location = {
		mnt_wealth_zero_location = yes
		if = {
			limit = { exists = owner }
			mnt_wealth_seed_location_schedule = yes
		}
	}
}

# Scope: none. Handles variable setup on every country and its locations

mnt_wealth_initialize_countries = {
	every_landed_country = {
		# The locations come first because country init needs to sum data from them.
		every_owned_location = { mnt_wealth_setup_location = yes }
		mnt_wealth_initialize_country = yes
	}
}

# Scope: country. Builds every country-level figure from the locations the
# country owns right now, assuming nothing about what it held before.
#
# The mirror of mnt_wealth_clear_annexed_country: everything cleared there is
# recounted here from the locations the country owns now.

mnt_wealth_initialize_country = {
	set_variable = { name = mnt_wealth_national_gdp_total value = mnt_wealth_owned_local_gdp_sum }
	set_variable = { name = mnt_wealth_national_rural_gdp value = mnt_wealth_owned_local_rural_gdp_sum }
	set_variable = { name = mnt_wealth_national_urban_gdp value = mnt_wealth_owned_local_urban_gdp_sum }
	set_variable = { name = mnt_wealth_national_urban_ndp value = mnt_wealth_owned_local_urban_ndp_sum }
	mnt_wealth_rebuild_national_estate_raw_power = yes
	mnt_wealth_set_national_savings_valves = { MONTHS = 1 }
}

# Scope: country (the one being annexed). Called by mnt_on_annexed.
# Country variables outlive the country, so a tag that is released again later
# comes back holding the totals it died with. Its national flow would then be
# computed against a treasury and a raw power total that no longer exist.

mnt_wealth_clear_annexed_country = {
	remove_variable = mnt_wealth_national_gdp_total
	remove_variable = mnt_wealth_national_rural_gdp
	remove_variable = mnt_wealth_national_urban_gdp
	remove_variable = mnt_wealth_national_urban_ndp
	clear_variable_map = mnt_wealth_national_estate_raw_power
	clear_variable_map = mnt_wealth_estate_flow
	clear_variable_map = mnt_wealth_estate_seizure_demand
}

# ── LOCATION SETUP ──

# Scope: location. Zeroes the wealth variables and maps this location owns, so
# later reads can count on them existing. The gdp survey owns the power maps.

mnt_wealth_zero_location = {
	set_variable = { name = mnt_wealth_pool                   value = 0 }
	set_variable = { name = mnt_wealth_local_gdp              value = 0 }
	set_variable = { name = mnt_wealth_local_rural_gdp          value = 0 }
	set_variable = { name = mnt_wealth_local_urban_gdp          value = 0 }
	set_variable = { name = mnt_wealth_local_urban_ndp           value = 0 }
	set_variable = { name = mnt_wealth_local_wages            value = 0 }
	set_variable = { name = mnt_wealth_current_wage_rate      value = 0 }
	set_variable = { name = mnt_wealth_kept_income_total      value = 0 }
	set_variable = { name = mnt_wealth_review_month           value = 0 }
	clear_variable_map = mnt_wealth_estate_wealth
	clear_variable_map = mnt_wealth_invested_since_review
	clear_variable_map = mnt_wealth_invested_wealth
}

# Scope: location. Sets up a location for the wealth system: surveys
# its goods and estate data, seeds per-estate wealth from GDP, and
# sets initial wage rates.
#
# Estate data is unreliable during on_game_start but tolerable and
# self-correcting, so no separate rebuild pass is needed.

mnt_wealth_setup_location = {
	mnt_wealth_ensure_survey_data = yes
	# Seed each estate's wealth from GDP, then set the pool to the total.
	set_local_variable = {
		name = mnt_wealth_starting_pool
		value = { value = var:mnt_wealth_local_gdp multiply = mnt_wealth_starting_gdp_months }
	}
	set_local_variable = { name = mnt_wealth_seed_running_total value = 0 }
	mnt_wealth_for_each_estate_in_power_cache = { EFFECT = mnt_wealth_seed_estate_wealth }
	set_variable = { name = mnt_wealth_pool value = local_var:mnt_wealth_seed_running_total }
	mnt_wealth_set_location_rates = yes
	# The review month is the campaign start, so the first service covers the
	# months that have passed since it.
}

mnt_wealth_set_location_rates = {
	mnt_wealth_set_location_wages = yes
	set_variable = { name = mnt_wealth_kept_income_total value = var:mnt_wealth_local_wages }
}

# ── ESTATE ITERATORS ──

# Scope: location. Iterates the owner's live estate list. Used by the
# resurvey, which needs the country's current estates regardless of what the
# power cache holds.

mnt_wealth_for_each_national_estate = {
	save_temporary_scope_as = mnt_wealth_location

	owner = {
		every_estate = {
			limit = { estate_type != estate_type:crown_estate }
			save_temporary_scope_as = mnt_wealth_estate
			set_local_variable = { name = mnt_wealth_estate_key value = estate_type }
			scope:mnt_wealth_location = { $EFFECT$ = yes }
		}
	}
}

# Scope: caller's scope. Hardcoded list of all seven non-crown estate types.
# Used by the survey to build the estate power cache. The survey has to iterate every
# estate type to initialize a location because the cache does not exist yet and every_estate
# is unreliable until the first tick.

mnt_wealth_for_all_estate_types = {
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:nobles_estate }
	$EFFECT$ = yes
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:clergy_estate }
	$EFFECT$ = yes
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:burghers_estate }
	$EFFECT$ = yes
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:peasants_estate }
	$EFFECT$ = yes
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:dhimmi_estate }
	$EFFECT$ = yes
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:tribes_estate }
	$EFFECT$ = yes
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:cossacks_estate }
	$EFFECT$ = yes
}

# Scope: location. Iterates the keys of mnt_wealth_cached_local_relative_estate_power
# directly. Used by game-start setup to read the cache the survey just built,
# visiting only estates that actually have power here. Also useful to identify
# estates that may have cached data but are no longer present, potentially post-conquest.

mnt_wealth_for_each_estate_in_power_cache = {
	save_temporary_scope_as = mnt_wealth_surveyed_location

	every_key_in_variable_map = {
		variable = mnt_wealth_cached_local_relative_estate_power
		save_temporary_scope_as = mnt_wealth_surveyed_key

		scope:mnt_wealth_surveyed_location = {
			set_local_variable = { name = mnt_wealth_estate_key value = scope:mnt_wealth_surveyed_key }
			$EFFECT$ = yes
		}
	}
}

# Scope: location. Iterates the owner's estates filtered by the power cache.
# Owner-authoritative: only estates the current owner has nationally and that
# hold an entry in mnt_wealth_cached_local_relative_estate_power. The most stable and
# consistent iterator for general work.

mnt_wealth_for_each_national_estate_in_power_cache = {
	save_temporary_scope_as = mnt_wealth_location

	owner = {
		every_estate = {
			limit = {
				estate_type != estate_type:crown_estate
				scope:mnt_wealth_location ?= { mnt_wealth_estate_is_present = { KEY = prev.estate_type } }
			}
			save_temporary_scope_as = mnt_wealth_estate
			set_local_variable = { name = mnt_wealth_estate_key value = estate_type }
			scope:mnt_wealth_location = { $EFFECT$ = yes }
		}
	}
}

# ── CACHED ESTATE SET ──
# The power-cache iterator's estate list, held for the location it was built
# for. Anything mutating mnt_wealth_cached_local_relative_estate_power mid-pass
# must call invalidate.

mnt_wealth_invalidate_estate_set = {
	remove_local_variable = mnt_wealth_estate_set_loc
}

# Scope: location. Captures the estate set into a local list.

mnt_wealth_build_estate_set = {
	clear_local_variable_list = mnt_wealth_estate_set
	save_temporary_scope_as = mnt_wealth_estate_set_location

	owner = {
		every_estate = {
			limit = {
				estate_type != estate_type:crown_estate
				scope:mnt_wealth_estate_set_location ?= { mnt_wealth_estate_is_present = { KEY = prev.estate_type } }
			}
			estate_type = { save_temporary_scope_as = mnt_wealth_estate_set_key }

			scope:mnt_wealth_estate_set_location = {
				add_to_local_variable_list = { name = mnt_wealth_estate_set target = scope:mnt_wealth_estate_set_key }
			}
		}
	}

	set_local_variable = { name = mnt_wealth_estate_set_loc value = -1 }
	if = {
		limit = { has_variable = mnt_wealth_location_id }
		set_local_variable = { name = mnt_wealth_estate_set_loc value = var:mnt_wealth_location_id }
	}
}

# Scope: location. Replays the cached set, rebuilding it when the recorded
# owner id is missing or belongs to another location. A location with no id
# records -1.

mnt_wealth_for_each_cached_estate = {
	set_local_variable = { name = mnt_wealth_estate_set_hit value = 0 }

	if = {
		limit = {
			has_local_variable = mnt_wealth_estate_set_loc
			has_variable = mnt_wealth_location_id
		}
		if = {
			limit = { local_var:mnt_wealth_estate_set_loc = var:mnt_wealth_location_id }
			set_local_variable = { name = mnt_wealth_estate_set_hit value = 1 }
		}
	}

	if = {
		limit = { local_var:mnt_wealth_estate_set_hit = 0 }
		mnt_wealth_build_estate_set = yes
	}

	save_temporary_scope_as = mnt_wealth_cached_set_location

	every_in_local_list = {
		variable = mnt_wealth_estate_set
		save_temporary_scope_as = mnt_wealth_cached_estate_key

		scope:mnt_wealth_cached_set_location = {
			set_local_variable = { name = mnt_wealth_estate_key value = scope:mnt_wealth_cached_estate_key }
			$EFFECT$ = yes
		}
	}
}

# ── LOCATION PASSES ──
#
# Player locations run the estate loop every month. AI locations get the
# closed-form advance at their review. The scheduler fires the matching pulse.

# Scope: location. The review month is the first month still to apply to the
# stored state. The advance and the display both count months from it. The
# monthly loop records the month after the one it just applied.

mnt_wealth_set_review_month = {
	set_variable = { name = mnt_wealth_review_month value = global_var:mnt_wealth_months_elapsed }
}

# Scope: location. One estate's monthly income: its share of the kept
# wages, then the arriving income with the national flow.

mnt_wealth_estate_month = {
	mnt_wealth_set_estate_wage_balance = yes
	mnt_wealth_resolve_estate_monthly_income = yes
}

# Scope: location (human-owned). Body of mnt_monthly_balanced_location_pulse.
# The full monthly cycle: advance to now, set wages, invest each estate,
# then collect each estate's income.

mnt_wealth_location_month = {
	# Apply any months still owed, such as those before the first service or
	# under a previous AI owner. In the usual month there are none.
	mnt_wealth_advance_location_to_now = yes
	# Frozen for the month so mid-loop estate changes do not shift the rate.
	# For if investment rate modifiers are added as investment/neglect modifiers
	set_local_variable = { name = mnt_wealth_cached_investment_rate value = mnt_wealth_investment_rate }
	# Accumulate each estate's kept income and investment spending across the
	# loop, then write the totals to variables and subtract from the pool.
	set_local_variable = { name = mnt_wealth_kept_income_running_total value = 0 }
	set_local_variable = { name = mnt_wealth_invested_running_total    value = 0 }
	set_local_variable = { name = mnt_wealth_arriving_running_total value = 0 }
	# Wage rates have to be set before the estate loop because each
	# estate's income depends on the location's total wages.
	mnt_wealth_set_location_wages = yes
	# Each estate first invests from the wealth it holds, the figure the
	# window shows, then the month's income lands.
	mnt_wealth_for_each_cached_estate = { EFFECT = mnt_wealth_invest_estate_wealth }
	change_variable = { name = mnt_wealth_pool subtract = local_var:mnt_wealth_invested_running_total }
	# Each estate earns its share of local wages and collects any national
	# flow the country sends.
	mnt_wealth_for_each_cached_estate = { EFFECT = mnt_wealth_estate_month }
	change_variable = { name = mnt_wealth_pool add = local_var:mnt_wealth_arriving_running_total }

	set_variable = { name = mnt_wealth_kept_income_total value = local_var:mnt_wealth_kept_income_running_total }
	mnt_wealth_raise_wealth_map_mode_ceiling = yes
	# This month is applied; the next one is the first still to apply.
	set_variable = {
		name = mnt_wealth_review_month
		value = { value = global_var:mnt_wealth_months_elapsed add = 1 }
	}
}

# Scope: location. Body of mnt_biyearly_balanced_location_pulse.
# The yearly review: advance to now, recompute investment levels, and
# drift wealth shares toward current power.

mnt_wealth_location_year = {
	# Advance stored state to now on the current wages, before anything else.
	mnt_wealth_advance_location_to_now = yes

	# Per-estate pass: investment levels and wealth drift.
	# Scale the drift step by age/12 so the monthly drift rate holds regardless
	# of whether the yearly review fires every 12 or 24 months.
	set_local_variable = {
		name = mnt_wealth_drift_age_factor
		value = { value = var:mnt_wealth_service_months divide = 12 }
	}
	set_local_variable = { name = mnt_wealth_drift_running_total value = 0 }
	mnt_wealth_for_each_cached_estate = { EFFECT = mnt_wealth_location_year_estate_pass }
	change_variable = { name = mnt_wealth_pool add = local_var:mnt_wealth_drift_running_total }

	# AI locations set their wages and kept income for the coming period.
	if = {
		limit = { owner ?= { is_ai = yes } }
		mnt_wealth_set_location_wages = yes
		set_variable = { name = mnt_wealth_kept_income_total value = var:mnt_wealth_local_wages }
	}

	mnt_wealth_raise_wealth_map_mode_ceiling = yes
}

# Scope: location. One estate's yearly pass: investment level and drift.

mnt_wealth_location_year_estate_pass = {
	mnt_wealth_update_estate_investment_level = yes
	mnt_wealth_drift_estate_wealth = yes
}
