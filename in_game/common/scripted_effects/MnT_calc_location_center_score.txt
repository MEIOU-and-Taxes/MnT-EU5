# Scope: location. Refreshes the urban goods value and all four center-of-importance
# scores for one location. Called from the yearly balanced location pass and from
# the game-start seeding event.

calc_location_center_scores = {
	calc_urban_goods_value_produced_location = yes
	calc_center_trade_score_location = yes
	calc_center_production_score_location = yes
	calc_center_culture_score_location = yes
	calc_center_education_score_location = yes
}

calc_center_trade_score_location = {
    scope:target_location = {
        set_variable = {
            name = mnt_center_trade_score_location
            value = mnt_center_trade_score
        }
    }
}
