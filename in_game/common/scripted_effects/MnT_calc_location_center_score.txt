# Scope: location. Refreshes all four center-of-importance scores for one
# location. Called from the yearly balanced location pass and from the
# game-start seeding event.

calc_location_center_scores = {
	calc_center_trade_score_location = yes
	calc_center_production_score_location = yes # Reads urban gross output, which the wealth census updates; kept here so the four scores sit together.
	calc_center_culture_score_location = yes
	calc_center_education_score_location = yes
}

calc_center_trade_score_location = {
    set_variable = {
        name = mnt_center_trade_score_location
        value = mnt_center_trade_score
    }
}
calc_center_production_score_location = {
    set_variable = {
        name = mnt_center_production_score_location
        value = mnt_center_production_score
    }
}
calc_center_culture_score_location = {
    set_variable = {
        name = mnt_center_culture_score_location
        value = mnt_center_culture_score
    }
}
calc_center_education_score_location = {
    set_variable = {
        name = mnt_center_education_score_location
        value = mnt_center_education_score
    }
}
