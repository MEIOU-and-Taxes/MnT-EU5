# Centers of Importance - leading urban industry of one location
#
# Called when an office appointment completes, to pick the office family. Reads the per-good values that
# calc_urban_goods_value_produced_location caches in mnt_gv_* (refreshed by the yearly centres pass), so
# this is 17 comparisons and no market lookups. Writes mnt_top_good (1-17) and mnt_top_good_value; the
# caller removes both afterwards.
#
# Index order matches the basket in calc_urban_goods_value_produced_location:
#   1 cloth          2 fine_cloth     3 furniture      4 glass          5 jewelry
#   6 lacquerware    7 naval_supplies 8 paper          9 porcelain     10 pottery
#  11 tools         12 books         13 weaponry      14 firearms      15 cannons
#  16 leather       17 beer

calc_top_industry_location = {
	set_variable = { name = mnt_top_good value = 0 }
	set_variable = { name = mnt_top_good_value value = 0 }

	# 1 cloth
	if = {
		limit = { var:mnt_gv_cloth > var:mnt_top_good_value }
		set_variable = { name = mnt_top_good_value value = var:mnt_gv_cloth }
		set_variable = { name = mnt_top_good value = 1 }
	}

	# 2 fine_cloth
