# Centers of Importance - top urban industry per location
#
# Called from calc_urban_goods_value_produced_location for locations that clear the local
# production threshold (production score = value x 10, so the gate is the local threshold / 10).
# A location that can hold a production centre therefore always has mnt_top_good.
#
# Index order matches the basket in calc_urban_goods_value_produced_location:
#   1 cloth          2 fine_cloth     3 furniture      4 glass          5 jewelry
#   6 lacquerware    7 naval_supplies 8 paper          9 porcelain     10 pottery
#  11 tools         12 books         13 weaponry      14 firearms      15 cannons
#  16 leather       17 beer
#
# Staleness: a location that falls below the threshold keeps its last value. Harmless - it
# cannot hold a production centre, so no consumer can reach it.

calc_top_industry_location = {

	set_variable = { name = mnt_top_good_value value = 0 }

	# 1 cloth
	set_variable = {
		name = mnt_good_value
		value = {
			value = "goods_output(goods:cloth)"
			multiply = "goods:cloth.price_in_market(scope:target_location.market)"
		}
	}
	if = {
		limit = { var:mnt_good_value > var:mnt_top_good_value }
		set_variable = { name = mnt_top_good_value value = var:mnt_good_value }
		set_variable = { name = mnt_top_good value = 1 }
