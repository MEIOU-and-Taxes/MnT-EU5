# Utility effects used across the wealth system: estate iterators, gold hooks, sized modifiers, map mode bands, lookup maps.

# Iterates the location's power cache keys. Needs no owner estate data, so setup and the rate cache use it.

mnt_wealth_for_each_estate_in_power_cache = {
	save_temporary_scope_as = mnt_wealth_cached_power_location

	every_key_in_variable_map = {
		variable = mnt_wealth_cached_local_relative_estate_power
		save_temporary_scope_as = mnt_wealth_cached_power_key

		scope:mnt_wealth_cached_power_location = {
			set_local_variable = { name = mnt_wealth_estate_key value = scope:mnt_wealth_cached_power_key }
			$EFFECT$ = yes
		}
	}
}

# National estates filtered to those present in this location's power cache.

mnt_wealth_for_each_national_estate_in_power_cache = {
	save_temporary_scope_as = mnt_wealth_location

	owner = {
		every_estate = {
			limit = {
				estate_type != estate_type:crown_estate
				scope:mnt_wealth_location = { mnt_wealth_estate_is_present = { KEY = prev.estate_type } }
			}
			save_temporary_scope_as = mnt_wealth_estate
			set_local_variable = { name = mnt_wealth_estate_key value = estate_type }
			scope:mnt_wealth_location = { $EFFECT$ = yes }
		}
	}
}

# Caches the filtered estate set as a local list so passes can replay without re-iterating.

mnt_wealth_build_location_estate_list = {
	clear_local_variable_list = mnt_wealth_location_estate_list
	save_temporary_scope_as = mnt_wealth_location_estate_list_scope

	# The estates the last census found here, whether or not the owner has them active. Filtering by the owner's
	# estates left wealth outside the pool: a country formed this tick has no estates yet, and a new owner can lack
	# one the pops here still belong to.
	every_key_in_variable_map = {
		variable = mnt_wealth_cached_local_relative_estate_power
		save_temporary_scope_as = mnt_wealth_location_estate_list_key

		scope:mnt_wealth_location_estate_list_scope = {
			add_to_local_variable_list = {
				name = mnt_wealth_location_estate_list
				target = scope:mnt_wealth_location_estate_list_key
			}
		}
	}

	# Tag the list with this location's id so the iterator knows when to rebuild.
	mnt_wealth_ensure_location_id = yes
	set_local_variable = { name = mnt_wealth_location_estate_list_location_id value = var:mnt_wealth_location_id }
	set_local_variable = { name = mnt_wealth_rebuild_location_estate_list value = no }
}

# Standard estate iterator. Replays the cached list, rebuilding if stale or from another location.

mnt_wealth_for_each_cached_estate = {
	if = {
		limit = {
			# Check existence first: an absent local reads as 0, and location id 0 exists.
			has_local_variable = mnt_wealth_location_estate_list_location_id
			local_var:mnt_wealth_rebuild_location_estate_list = no
			has_variable = mnt_wealth_location_id
			local_var:mnt_wealth_location_estate_list_location_id = var:mnt_wealth_location_id
		}
	}
	else = {
		mnt_wealth_build_location_estate_list = yes
	}
	save_temporary_scope_as = mnt_wealth_location_cached_estate

	every_in_local_list = {
		variable = mnt_wealth_location_estate_list
		save_temporary_scope_as = mnt_wealth_cached_estate_key

		scope:mnt_wealth_location_cached_estate = {
			set_local_variable = { name = mnt_wealth_estate_key value = scope:mnt_wealth_cached_estate_key }
			$EFFECT$ = yes
		}
	}
}

# Runs EFFECT with ESTATE set to the short estate name, for effects that name modifiers or variables per estate.

mnt_wealth_run_effect_with_estate_name = {
	switch = {
		trigger = local_var:mnt_wealth_estate_key
		estate_type:nobles_estate   = { $EFFECT$ = { ESTATE = nobles } }
		estate_type:clergy_estate   = { $EFFECT$ = { ESTATE = clergy } }
		estate_type:burghers_estate = { $EFFECT$ = { ESTATE = burghers } }
		estate_type:peasants_estate = { $EFFECT$ = { ESTATE = peasants } }
		estate_type:dhimmi_estate   = { $EFFECT$ = { ESTATE = dhimmi } }
		estate_type:tribes_estate   = { $EFFECT$ = { ESTATE = tribes } }
		estate_type:cossacks_estate = { $EFFECT$ = { ESTATE = cossacks } }
	}
}

# These add/remove helpers are hooks for other systems to move gold into or out of a location.

mnt_wealth_add_location_wealth = {
	set_local_variable = { name = mnt_wealth_add_amount value = { value = $AMOUNT$ min = 0 } }

	if = {
		limit = { var:mnt_wealth_pool > 0 }
		set_local_variable = {
			name = mnt_wealth_add_scale
			value = { value = local_var:mnt_wealth_add_amount divide = var:mnt_wealth_pool add = 1 }
		}

		save_temporary_scope_as = mnt_wealth_add_loc

		every_key_in_variable_map = {
			variable = mnt_wealth_estate_wealth
			save_temporary_scope_as = mnt_wealth_add_key

			scope:mnt_wealth_add_loc = {
				set_local_variable = {
					name = mnt_wealth_add_scaled
					value = {
						value = "variable_map(mnt_wealth_estate_wealth|scope:mnt_wealth_add_key)"
						multiply = local_var:mnt_wealth_add_scale
					}
				}

				add_to_variable_map = {
					name = mnt_wealth_estate_wealth
					key = scope:mnt_wealth_add_key
					value = local_var:mnt_wealth_add_scaled
				}
			}
		}
	}
	else = {
		save_temporary_scope_as = mnt_wealth_add_loc

		every_key_in_variable_map = {
			variable = mnt_wealth_cached_local_relative_estate_power
			save_temporary_scope_as = mnt_wealth_add_key

			scope:mnt_wealth_add_loc = {
				set_local_variable = { name = mnt_wealth_add_existing value = 0 }

				if = {
					limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = scope:mnt_wealth_add_key } }
					set_local_variable = {
						name = mnt_wealth_add_existing
						value = "variable_map(mnt_wealth_estate_wealth|scope:mnt_wealth_add_key)"
					}
				}

				set_local_variable = {
					name = mnt_wealth_add_by_power
					value = {
						value = local_var:mnt_wealth_add_existing
						add = {
							value = local_var:mnt_wealth_add_amount
							multiply = "variable_map(mnt_wealth_cached_local_relative_estate_power|scope:mnt_wealth_add_key)"
						}
					}
				}

				add_to_variable_map = {
					name = mnt_wealth_estate_wealth
					key = scope:mnt_wealth_add_key
					value = local_var:mnt_wealth_add_by_power
				}
			}
		}
	}

	change_variable = { name = mnt_wealth_pool add = local_var:mnt_wealth_add_amount }
}

mnt_wealth_remove_location_wealth = {
	set_local_variable = { name = mnt_wealth_remove_amount value = { value = $AMOUNT$ min = 0 } }

	# max = is a CEILING, so it caps remove_amount at the current wealth.
	set_local_variable = {
		name = mnt_wealth_removed_amount
		value = {
			value = local_var:mnt_wealth_remove_amount
			max = var:mnt_wealth_pool
			min = 0
		}
	}

	if = {
		limit = { local_var:mnt_wealth_removed_amount > 0 var:mnt_wealth_pool > 0 }
		set_local_variable = {
			name = mnt_wealth_remove_scale
			value = {
				value = var:mnt_wealth_pool
				subtract = local_var:mnt_wealth_removed_amount
				divide = var:mnt_wealth_pool
			}
		}

		save_temporary_scope_as = mnt_wealth_remove_loc

		every_key_in_variable_map = {
			variable = mnt_wealth_estate_wealth
			save_temporary_scope_as = mnt_wealth_remove_key

			scope:mnt_wealth_remove_loc = {
				set_local_variable = {
					name = mnt_wealth_remove_scaled
					value = {
						value = "variable_map(mnt_wealth_estate_wealth|scope:mnt_wealth_remove_key)"
						multiply = local_var:mnt_wealth_remove_scale
					}
				}

				add_to_variable_map = {
					name = mnt_wealth_estate_wealth
					key = scope:mnt_wealth_remove_key
					value = local_var:mnt_wealth_remove_scaled
				}
			}
		}
	}

	set_variable = {
		name = mnt_wealth_pool
		value = { value = var:mnt_wealth_pool subtract = local_var:mnt_wealth_removed_amount min = 0 }
	}

	set_local_variable = { name = mnt_wealth_removed_total value = local_var:mnt_wealth_removed_amount }
}

# Scope: location. Adds $AMOUNT$ gold to $ESTATE$'s wealth and the total.

mnt_wealth_add_local_estate_wealth = {
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:$ESTATE$_estate }
	set_local_variable = { name = mnt_wealth_add_amount value = { value = $AMOUNT$ min = 0 } }
	mnt_wealth_credit_estate_wealth = { AMOUNT = local_var:mnt_wealth_add_amount }
}

mnt_wealth_remove_local_estate_wealth = {
	set_local_variable = { name = mnt_wealth_estate_key value = estate_type:$ESTATE$_estate }
	set_local_variable = { name = mnt_wealth_local_estate_wealth value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_local_estate_wealth
			value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
		}
	}

	# max = is a CEILING.
	set_local_variable = {
		name = mnt_wealth_take_amount
		value = {
			value = $AMOUNT$
			min = 0
			max = local_var:mnt_wealth_local_estate_wealth
		}
	}

	set_local_variable = { name = mnt_wealth_removed_total value = local_var:mnt_wealth_take_amount }
	set_local_variable = { name = mnt_wealth_take_amount   value = { value = local_var:mnt_wealth_take_amount multiply = -1 } }
	mnt_wealth_credit_estate_wealth = { AMOUNT = local_var:mnt_wealth_take_amount }
}

mnt_wealth_credit_estate_wealth = {
	set_local_variable = { name = mnt_wealth_credit_current value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_credit_current
			value = "variable_map(mnt_wealth_estate_wealth|local_var:mnt_wealth_estate_key)"
		}
	}

	set_local_variable = {
		name = mnt_wealth_credit_new
		value = { value = local_var:mnt_wealth_credit_current add = $AMOUNT$ }
	}

	add_to_variable_map = {
		name = mnt_wealth_estate_wealth
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_credit_new
	}

	change_variable = { name = mnt_wealth_pool add = $AMOUNT$ }
}

mnt_wealth_apply_sized_modifier = {
	set_local_variable = {
		name = mnt_wealth_modifier_multiple
		value = {
			value = $SIZE$
			divide = $FACTOR$
			round = yes
			min = 0
		}
	}

	set_local_variable = { name = mnt_wealth_modifier_stale value = 1 }

	if = {
		limit = { has_location_modifier = $MODIFIER$ has_variable = mnt_wealth_applied_$MODIFIER$ }
		if = {
			limit = { var:mnt_wealth_applied_$MODIFIER$ = local_var:mnt_wealth_modifier_multiple }
			set_local_variable = { name = mnt_wealth_modifier_stale value = 0 }
		}
	}

	if = {
		limit = { local_var:mnt_wealth_modifier_multiple = 0 }
		mnt_wealth_remove_sized_modifier = { MODIFIER = $MODIFIER$ }
	}
	else_if = {
		limit = { local_var:mnt_wealth_modifier_stale = 1 }

		set_local_variable = {
			name = mnt_wealth_modifier_size
			value = { value = local_var:mnt_wealth_modifier_multiple multiply = $FACTOR$ }
		}

		add_location_modifier = {
			modifier = $MODIFIER$
			years = -1
			mode = replace
			size = local_var:mnt_wealth_modifier_size
		}

		set_variable = { name = mnt_wealth_applied_$MODIFIER$ value = local_var:mnt_wealth_modifier_multiple }
	}
}

mnt_wealth_remove_sized_modifier = {
	if = { limit = { has_location_modifier = $MODIFIER$ } remove_location_modifier = $MODIFIER$ }
	if = { limit = { has_variable = mnt_wealth_applied_$MODIFIER$ } remove_variable = mnt_wealth_applied_$MODIFIER$ }
}

mnt_wealth_set_wealth_map_mode_bands = {
	set_local_variable = { name = mnt_wealth_band_index value = 1 }

	while = {
		count = 5

		set_local_variable = {
			name = mnt_wealth_band_threshold
			value = {
				value = global_var:mnt_wealth_map_mode_max_wealth_pool
				multiply = local_var:mnt_wealth_band_index
				divide = 6
			}
		}

		add_to_global_variable_map = {
			name = mnt_wealth_bands
			key = local_var:mnt_wealth_band_index
			value = local_var:mnt_wealth_band_threshold
		}

		change_local_variable = { name = mnt_wealth_band_index add = 1 }
	}
}

mnt_wealth_raise_wealth_map_mode_ceiling = {
	# The ceiling is the largest pool on the map; the bands follow it at once.
	if = {
		limit = { var:mnt_wealth_pool > global_var:mnt_wealth_map_mode_max_wealth_pool }
		set_global_variable = { name = mnt_wealth_map_mode_max_wealth_pool value = var:mnt_wealth_pool }
		mnt_wealth_set_wealth_map_mode_bands = yes
	}
}

# $MODE$ is total, urban, or rural.

mnt_wealth_set_gross_output_map_mode_bands = {
	set_global_variable = { name = mnt_wealth_gross_output_band_$MODE$_1 value = { value = global_var:mnt_wealth_map_mode_max_gross_output_$MODE$ divide = 6 } }
	set_global_variable = { name = mnt_wealth_gross_output_band_$MODE$_2 value = { value = global_var:mnt_wealth_map_mode_max_gross_output_$MODE$ divide = 3 } }
	set_global_variable = { name = mnt_wealth_gross_output_band_$MODE$_3 value = { value = global_var:mnt_wealth_map_mode_max_gross_output_$MODE$ divide = 2 } }
	set_global_variable = { name = mnt_wealth_gross_output_band_$MODE$_4 value = { value = global_var:mnt_wealth_map_mode_max_gross_output_$MODE$ multiply = 2 divide = 3 } }
	set_global_variable = { name = mnt_wealth_gross_output_band_$MODE$_5 value = { value = global_var:mnt_wealth_map_mode_max_gross_output_$MODE$ multiply = 5 divide = 6 } }
}

mnt_wealth_raise_gross_output_map_mode_ceilings = {
	if = {
		limit = { var:mnt_wealth_local_gross_output > global_var:mnt_wealth_map_mode_max_gross_output_total }
		change_global_variable = { name = mnt_wealth_map_mode_max_gross_output_total add = 50 }
		mnt_wealth_set_gross_output_map_mode_bands = { MODE = total }
	}
	if = {
		limit = { var:mnt_wealth_local_urban_gross_output > global_var:mnt_wealth_map_mode_max_gross_output_urban }
		change_global_variable = { name = mnt_wealth_map_mode_max_gross_output_urban add = 50 }
		mnt_wealth_set_gross_output_map_mode_bands = { MODE = urban }
	}
	if = {
		limit = { var:mnt_wealth_local_rural_gross_output > global_var:mnt_wealth_map_mode_max_gross_output_rural }
		change_global_variable = { name = mnt_wealth_map_mode_max_gross_output_rural add = 50 }
		mnt_wealth_set_gross_output_map_mode_bands = { MODE = rural }
	}
}
