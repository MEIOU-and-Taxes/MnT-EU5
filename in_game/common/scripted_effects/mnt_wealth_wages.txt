# Wages: a share of gross output split by estate power. Maps are sparse; absence = zero.

mnt_wealth_set_location_wages = {
	set_local_variable = {
		name = mnt_wealth_wage_rate
		value = {
			value = modifier:mnt_wealth_local_wage_modifier
			add = owner.modifier:mnt_wealth_global_wage_modifier
			min = mnt_wealth_minimum_wage_rate
		}
	}

	set_variable = { name = mnt_wealth_current_wage_rate value = local_var:mnt_wealth_wage_rate }
	set_variable = {
		name = mnt_wealth_local_wages
		value = { value = local_var:mnt_wealth_wage_rate multiply = var:mnt_wealth_local_gross_output }
	}

	set_local_variable = {
		name = mnt_wealth_wage_burden_size
		value = { value = local_var:mnt_wealth_wage_rate multiply = 100 }
	}

	if = {
		limit = { local_var:mnt_wealth_wage_burden_size > 0 }
		mnt_wealth_apply_sized_modifier = {
			MODIFIER = mnt_wealth_wage_burden
			SIZE = local_var:mnt_wealth_wage_burden_size
			FACTOR = mnt_wealth_wage_rounding_factor
		}
	}
	else = {
		mnt_wealth_remove_sized_modifier = { MODIFIER = mnt_wealth_wage_burden }
	}

	# AI and slow-census locations have no monthly pass; set kept income here instead.
	if = {
		limit = { owner ?= { OR = { is_ai = yes mnt_wealth_census_frequency_is_slow = yes } } }
		set_variable = { name = mnt_wealth_kept_income_total value = var:mnt_wealth_local_wages }
	}
}

mnt_wealth_compute_estate_kept_wages = {
	# No power entry is no claim on the wages.
	set_local_variable = { name = mnt_wealth_kept_amount value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_cached_local_relative_estate_power target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_kept_amount
			value = {
				value = var:mnt_wealth_local_wages
				multiply = "variable_map(mnt_wealth_cached_local_relative_estate_power|local_var:mnt_wealth_estate_key)"
			}
		}
	}

	change_local_variable = { name = mnt_wealth_kept_income_running_total add = local_var:mnt_wealth_kept_amount }

	add_to_local_variable_map = {
		name = mnt_wealth_kept_wages
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_kept_amount
	}
}
