# Game start seeding and the location lifecycle: first setup, colonisation, and owner change.

# Zero every ownable location; owned ones also get a schedule slot and first census.
# Iterates ownable locations to keep location ids in map order for the schedule cohorts.

mnt_wealth_seed_owned_locations = {
	set_global_variable = { name = mnt_wealth_location_serial value = 0 }

	every_ownable_location = {
		mnt_wealth_zero_location = yes
		if = {
			limit = { has_owner = yes }
			mnt_wealth_seed_location_schedule = yes
			mnt_wealth_setup_location = yes
		}
	}
}

mnt_wealth_seed_economy_constants = {
	set_global_variable = { name = mnt_wealth_map_mode_max_wealth_pool value = 500 }
	mnt_wealth_set_wealth_map_mode_bands = yes
	# Gross output map modes: one ceiling each, raised as locations outgrow it.
	set_global_variable = { name = mnt_wealth_map_mode_max_gross_output_total value = 50 }
	set_global_variable = { name = mnt_wealth_map_mode_max_gross_output_urban value = 50 }
	set_global_variable = { name = mnt_wealth_map_mode_max_gross_output_rural value = 50 }
	mnt_wealth_set_gross_output_map_mode_bands = { MODE = total }
	mnt_wealth_set_gross_output_map_mode_bands = { MODE = urban }
	mnt_wealth_set_gross_output_map_mode_bands = { MODE = rural }
	# Lookup maps the rest of the system reads at runtime.
	mnt_wealth_build_power_per_pop_map = yes
	mnt_wealth_build_estate_short_key_map = yes
	mnt_wealth_build_days_in_month_map = yes
	mnt_wealth_build_good_classification_map = yes
}

# Every variable the system reads needs to exist before the first census or owner change touches the location.

mnt_wealth_zero_location = {
	set_variable = { name = mnt_wealth_pool                     value = 0 }
	set_variable = { name = mnt_wealth_local_gross_output       value = 0 }
	set_variable = { name = mnt_wealth_local_rural_gross_output value = 0 }
	set_variable = { name = mnt_wealth_local_urban_gross_output value = 0 }
	# set_variable = { name = mnt_wealth_local_urban_gdp           value = 0 }
	# set_variable = { name = mnt_wealth_local_gdp                value = 0 }
	set_variable = { name = mnt_wealth_local_wages       value = 0 }
	set_variable = { name = mnt_wealth_current_wage_rate value = 0 }
	set_variable = { name = mnt_wealth_kept_income_total value = 0 }
	set_variable = { name = mnt_wealth_review_month      value = 0 }
	set_variable = { name = mnt_wealth_reconciled_month  value = 0 }
	set_variable = { name = mnt_wealth_cached_investment_rate        value = 0 }
	set_variable = { name = mnt_wealth_cached_investment_maintenance value = 0 }
	clear_variable_map = mnt_wealth_goods_value
	clear_variable_map = mnt_wealth_estate_wealth
	clear_variable_map = mnt_wealth_invested_since_review
	clear_variable_map = mnt_wealth_invested_wealth
	clear_variable_map = mnt_wealth_cached_estate_decay_rate
	clear_variable_map = mnt_wealth_flow_paid_per_million_at_settle
}

mnt_wealth_setup_location = {
	# Estate data is unreliable during on_game_start; self-corrects at the first census.
	mnt_wealth_ensure_census_data = yes
	# Seed each estate's wealth from gross output, then set the pool to the total.
	set_local_variable = {
		name = mnt_wealth_starting_pool
		value = { value = var:mnt_wealth_local_gross_output multiply = mnt_wealth_starting_gross_output_months }
	}
	set_local_variable = { name = mnt_wealth_seed_running_total value = 0 }
	mnt_wealth_for_each_estate_in_power_cache = { EFFECT = mnt_wealth_seed_estate_wealth }
	# Running total avoids rounding gaps from seed_estate_wealth.
	set_variable = { name = mnt_wealth_pool value = local_var:mnt_wealth_seed_running_total }
	mnt_wealth_raise_wealth_map_mode_ceiling = yes
	mnt_wealth_set_location_wages = yes
	# Shares sum to the whole bill, so kept income total = wages.
	set_variable = { name = mnt_wealth_kept_income_total value = var:mnt_wealth_local_wages }
	# The review month stays at zero, the campaign start, so the first settle covers every month since.
	mnt_wealth_cache_location_rates = yes
}

mnt_wealth_ensure_census_data = {
	if = { limit = { var:mnt_wealth_local_gross_output = 0 } mnt_wealth_update_local_gross_output = yes }

	if = {
		limit = {
			NOT = {
				has_variable_map = mnt_wealth_cached_local_relative_estate_power
				has_variable_map = mnt_estate_raw_power
			}
		}
		mnt_wealth_census_estate_data = yes
	}
}

mnt_wealth_seed_estate_wealth = {
	set_local_variable = { name = mnt_wealth_seed_amount value = 0 }

	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_cached_local_relative_estate_power target = local_var:mnt_wealth_estate_key } }
		set_local_variable = {
			name = mnt_wealth_seed_amount
			value = {
				value = local_var:mnt_wealth_starting_pool
				multiply = "variable_map(mnt_wealth_cached_local_relative_estate_power|local_var:mnt_wealth_estate_key)"
			}
		}
	}

	add_to_variable_map = {
		name = mnt_wealth_estate_wealth
		key = local_var:mnt_wealth_estate_key
		value = local_var:mnt_wealth_seed_amount
	}

	change_local_variable = { name = mnt_wealth_seed_running_total add = local_var:mnt_wealth_seed_amount }
}

# A location with no service record, colonised or otherwise new to the schedule. No opening pool.

mnt_wealth_initialize_colonized_location = {
	# The census counts the estates into the owner's national raw power.
	mnt_wealth_census_location = yes
	set_variable = { name = mnt_wealth_kept_income_total value = var:mnt_wealth_local_wages }
	mnt_wealth_set_review_month = yes
	set_variable = { name = mnt_wealth_reconciled_month value = global_var:mnt_wealth_months_elapsed }
	mnt_wealth_cache_location_rates = yes
	mnt_wealth_record_flow_baseline = { COUNTRY = owner }
	# The last service is now, with no backdate.
	set_local_variable = { name = mnt_wealth_last_service value = global_var:mnt_wealth_months_elapsed }
	mnt_wealth_write_location_schedule = yes
}

# Existing land under a new owner: settle under the old owner's recorded rates, then census under the new one.

mnt_wealth_recensus_transferred_location = {
	mnt_wealth_reconcile_location = yes
	mnt_wealth_census_location = yes
}
