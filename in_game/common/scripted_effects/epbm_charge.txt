# The charge debits the same estate treasury the location wealth system's
# savings valves read (mnt_wealth_set_national_savings_valves), so maintenance
# costs raise an estate's seizure demand on its locations by themselves.
epbm_charge_estates = {
	every_estate = {
		limit = { estate_type != estate_type:crown_estate }
		save_scope_as = epbm_est

		prev = {
			if = {
				limit = {
					has_variable_map = epbm_pay
					is_key_in_variable_map = { name = epbm_pay target = scope:epbm_est }
					"variable_map(epbm_pay|scope:epbm_est)" > 0
				}
				add_gold_to_estate = {
					estate_type = scope:epbm_est.estate_type
					value = { value = "variable_map(epbm_pay|scope:epbm_est)" multiply = -1 }
				}
			}
		}
	}
}

# Copies pay vars to show/prev vars so tooltips display last month's values.
# Crown display reads epbm_crown_pool_cost live; only estate maps need snapshot.
epbm_snapshot_for_display = {
	epbm_snapshot_estates_dynamic = yes

	epbm_for_each_estate = { effect = epbm_snapshot_mod }
}
