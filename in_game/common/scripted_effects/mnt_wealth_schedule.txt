# Scheduler: decides when each location next needs work and fires the pulse
# for every cadence that is ready.
#
# The schedule is twelve global maps, one per month of the year. A location
# is filed in the bucket of its next service month:
#
#   mnt_wealth_service_bucket_1..12       - location -> its next service month
#   mnt_wealth_location_last_service  - location -> month of its last periodic
#                                       service
#   var:mnt_wealth_service_bucket         - which bucket this location is filed in
#   var:mnt_wealth_service_cycle      - completed periodic services, drives the
#                                       deep review and the gross output census
#
# The monthly dispatch walks only the current month's bucket, so the walk
# scales with the work that is ready, not with the world. Every periodic cadence is a
# multiple of twelve months, so a location keeps its bucket for life; only
# player-owned locations (serviced monthly) and owner-change requeues move.
#
# Markets have no schedule: they are few and run every month on the tick,
# straight from the dispatch.

# ── READING THE LAST SERVICE ──

# Scope: location. Leaves the month of the last periodic service in
# local_var:mnt_wealth_last_service, and flags a location the scheduler has
# never touched in local_var:mnt_wealth_first_service.

mnt_wealth_load_last_service = {
	set_local_variable = { name = mnt_wealth_last_service value = 0 }
	set_local_variable = { name = mnt_wealth_first_service value = 1 }
	if = {
		limit = { is_key_in_global_variable_map = { name = mnt_wealth_location_last_service target = this } }
		set_local_variable = {
			name = mnt_wealth_last_service
			value = "global_variable_map(mnt_wealth_location_last_service|this)"
		}
		set_local_variable = { name = mnt_wealth_first_service value = 0 }
	}
}

# ── THE SERVICE BUCKETS ──

# Scope: location. Files the location in the bucket of the next service month in
# local_var:mnt_wealth_next_service. Map names cannot be built at runtime, so the
# bucket pick is a spelled-out branch per map.

mnt_wealth_schedule_next_service = {
	# The bucket is the next service month mod twelve, one-based.
	set_local_variable = {
		name = mnt_wealth_service_bucket_index
		value = { value = local_var:mnt_wealth_next_service modulo = 12 add = 1 }
	}

	# Moving buckets (player monthly, owner-change requeue): pull the old
	# entry first. The same bucket just overwrites in place, the common AI case.
	if = {
		limit = {
			has_variable = mnt_wealth_service_bucket
			NOT = { var:mnt_wealth_service_bucket = local_var:mnt_wealth_service_bucket_index }
		}
		mnt_wealth_unschedule_location = yes
	}
	set_variable = { name = mnt_wealth_service_bucket value = local_var:mnt_wealth_service_bucket_index }

	save_temporary_scope_as = mnt_wealth_bucket_entity

	if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 1 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_1 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 2 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_2 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 3 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_3 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 4 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_4 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 5 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_5 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 6 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_6 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 7 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_7 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 8 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_8 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 9 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_9 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 10 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_10 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else_if = {
		limit = { local_var:mnt_wealth_service_bucket_index = 11 }
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_11 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
	else = {
		add_to_global_variable_map = { name = mnt_wealth_service_bucket_12 key = scope:mnt_wealth_bucket_entity value = local_var:mnt_wealth_next_service }
	}
}

# Scope: location. Removes the location from the bucket recorded on it.
# Safe on a location that was never filed.

mnt_wealth_unschedule_location = {
	if = {
		limit = { has_variable = mnt_wealth_service_bucket }
		save_temporary_scope_as = mnt_wealth_bucket_entity

		if = {
			limit = { var:mnt_wealth_service_bucket = 1 }
			remove_from_global_variable_map = { name = mnt_wealth_service_bucket_1 key = scope:mnt_wealth_bucket_entity }
		}

# ── WRITING THE NEXT SERVICE MONTH ──

# Scope: location. Reads local_var:mnt_wealth_is_player and
# local_var:mnt_wealth_last_service. Players come back next month; everyone
# else sleeps until the year boundary after their last periodic service. A
# location pulled in early by an owner change kept its old last-service month,
# so this files it right back into its old bucket.

mnt_wealth_write_location_next_service = {
	if = {
		limit = { local_var:mnt_wealth_is_player = 1 }
		set_local_variable = {
			name = mnt_wealth_next_service
			value = { value = global_var:mnt_wealth_months_elapsed add = 1 }
		}
	}
	else = {
		set_local_variable = {
			name = mnt_wealth_next_service
			value = { value = local_var:mnt_wealth_last_service add = 12 }
		}
	}

	mnt_wealth_schedule_next_service = yes
}

# Scope: location. Brings the next service forward to next month, so the next
# dispatch picks the location up and rebuilds its schedule against its new
# owner.

mnt_wealth_require_location_service = {
	set_local_variable = {
		name = mnt_wealth_next_service
		value = { value = global_var:mnt_wealth_months_elapsed add = 1 }
	}
	mnt_wealth_schedule_next_service = yes
}

# ── LOCATION IDENTITY ──
# A serial number the estate-set cache, the day spread, and the logs use to tell
# locations apart.

mnt_wealth_ensure_location_id = {
	if = {
		limit = { NOT = { has_variable = mnt_wealth_location_id } }
		if = {
			limit = { NOT = { has_global_variable = mnt_wealth_location_serial } }
			set_global_variable = { name = mnt_wealth_location_serial value = 0 }
		}
		set_variable = { name = mnt_wealth_location_id value = global_var:mnt_wealth_location_serial }
		change_global_variable = { name = mnt_wealth_location_serial add = 1 }
	}
}

# ── OPENING SPREAD ──

# Scope: location. Backdates the last service by the location's serial, so the
# first services spread evenly over the first year, and every location keeps
# that bucket from then on.

mnt_wealth_seed_location_schedule = {
	mnt_wealth_ensure_location_id = yes

	# last service = -(id mod 12), so the first service month lands on 1..12.
	set_local_variable = {
		name = mnt_wealth_last_service
		value = {
			value = var:mnt_wealth_location_id
			modulo = 12
			multiply = -1
		}
	}
	save_temporary_scope_as = mnt_wealth_seed_location
	add_to_global_variable_map = {
		name = mnt_wealth_location_last_service
		key = scope:mnt_wealth_seed_location
		value = local_var:mnt_wealth_last_service
	}

	# Spread the deep review and gross output census across cohorts.
	set_local_variable = {
		name = mnt_wealth_seed_cycle
		value = { value = var:mnt_wealth_location_id modulo = 10 }
	}
	set_variable = { name = mnt_wealth_service_cycle value = local_var:mnt_wealth_seed_cycle }

	set_local_variable = { name = mnt_wealth_is_player value = 0 }
	mnt_wealth_write_location_next_service = yes
}

# ── THE SERVICE ──

# Scope: location. Run once by the dispatch on each location whose service month arrived. The pulses run in the order the work depends on: the gross output refresh and
# the deep review both feed the census, and the scoring pulse reads what the
# census produced.

mnt_wealth_service_location = {
	mnt_wealth_ensure_location_id = yes

	set_local_variable = { name = mnt_wealth_is_player value = 0 }
	if = {
		limit = { owner ?= { is_human = yes } }
		set_local_variable = { name = mnt_wealth_is_player value = 1 }
	}

	# Months since the last periodic pass; a fresh location reads as a whole
	# campaign and bootstraps everything below.
	mnt_wealth_load_last_service = yes
	set_local_variable = {
		name = mnt_wealth_service_age
		value = {
			value = global_var:mnt_wealth_months_elapsed
			subtract = local_var:mnt_wealth_last_service
		}
	}

	# Player locations do their month work on every visit; visits are monthly.
	if = {
		limit = { local_var:mnt_wealth_is_player = 1 }
		trigger_event_silently = { on_action = mnt_monthly_balanced_location_pulse }
	}

	set_local_variable = { name = mnt_wealth_censused_full value = 0 }
	set_local_variable = { name = mnt_wealth_censused_yearly value = 0 }

	# The periodic ladder: only when a full year has passed. An owner-change
	# requeue arrives early, skips it, and keeps its old rhythm.
	if = {
		limit = { local_var:mnt_wealth_service_age >= 12 }

		# One more completed service on the cycle counter.
		if = {
			limit = { NOT = { has_variable = mnt_wealth_service_cycle } }
			set_local_variable = {
				name = mnt_wealth_seed_cycle
				value = { value = var:mnt_wealth_location_id modulo = 10 }
			}
			set_variable = { name = mnt_wealth_service_cycle value = local_var:mnt_wealth_seed_cycle }
		}
		change_variable = { name = mnt_wealth_service_cycle add = 1 }

		# Months the deep review will cover, accumulated service by service.
		if = {
			limit = { NOT = { has_variable = mnt_wealth_review_backlog } }
			set_variable = { name = mnt_wealth_review_backlog value = 0 }
		}
		change_variable = { name = mnt_wealth_review_backlog add = local_var:mnt_wealth_service_age }

		# Gross output census: every fifth service is five years, since services are
		# yearly. A never-serviced location is censused immediately.
		set_local_variable = {
			name = mnt_wealth_census_phase
			value = { value = var:mnt_wealth_service_cycle modulo = 5 }
		}
		if = {
			limit = {
				OR = {
					local_var:mnt_wealth_census_phase = 0
					local_var:mnt_wealth_first_service = 1
				}
			}
			trigger_event_silently = { on_action = mnt_five_yearly_balanced_location_pulse }
			set_local_variable = { name = mnt_wealth_censused_full value = 1 }
		}

		# Deep review: players every year, AI every second year, and always on
		# the bootstrap pass.
		set_local_variable = {
			name = mnt_wealth_deep_phase
			value = { value = var:mnt_wealth_service_cycle modulo = 2 }
		}
		if = {
			limit = {
				OR = {
					local_var:mnt_wealth_is_player = 1
					local_var:mnt_wealth_deep_phase = 0
					local_var:mnt_wealth_first_service = 1
				}
			}
			# The review prorates by the months it covers, handed over as a
			# location variable because locals do not reliably cross the event
			# boundary.
			set_variable = {
				name = mnt_wealth_service_months
				value = { value = var:mnt_wealth_review_backlog min = 1 }
			}
			set_variable = { name = mnt_wealth_review_backlog value = 0 }
			trigger_event_silently = { on_action = mnt_biyearly_balanced_location_pulse }
			set_local_variable = { name = mnt_wealth_censused_yearly value = 1 }
		}

		# Record this pass; the next service month builds on it.
		set_local_variable = { name = mnt_wealth_last_service value = global_var:mnt_wealth_months_elapsed }
		save_temporary_scope_as = mnt_wealth_service_entity
		add_to_global_variable_map = {
			name = mnt_wealth_location_last_service
			key = scope:mnt_wealth_service_entity
			value = local_var:mnt_wealth_last_service
		}
	}

	# Full census: pops, raw power, relative cache, and departure reconcile.
	if = {
		limit = { local_var:mnt_wealth_censused_full = 1 }
		mnt_wealth_retake_estate_census = yes
	}
	# Yearly lite: relative cache only, when the full path did not fire.
	else_if = {
		limit = { local_var:mnt_wealth_censused_yearly = 1 }
		mnt_wealth_refresh_relative_power = yes
	}

	if = {
		limit = { OR = { local_var:mnt_wealth_censused_full = 1 local_var:mnt_wealth_censused_yearly = 1 } }
		mnt_wealth_invalidate_estate_set = yes
	}

	# True yearly for every location: the non-wealth systems.
	if = {
		limit = { local_var:mnt_wealth_service_age >= 12 }
		trigger_event_silently = { on_action = mnt_yearly_balanced_location_pulse }
	}

	mnt_wealth_write_location_next_service = yes
}
