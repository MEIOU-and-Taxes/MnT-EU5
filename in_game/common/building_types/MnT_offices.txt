# Centers of Importance - Offices and Boards of Manufactories
#
# Event-only buildings granted by the cabinet actions appoint_inspector_of_manufactures and
# elevate_office_of_manufactories. Tier 1 (Office) gives +10% to the city's leading industry,
# tier 2 (Board) +20%. Both carry a paper, books, glass, wax and furniture maintenance cost and employ burghers.
#
# Families cover the 17 urban goods of the centres basket:
#   textile   cloth, fine_cloth
#   ceramics  glass, porcelain, pottery
#   metalwork tools, weaponry, firearms, cannons
#   naval     naval_supplies
#   learning  paper, books
#   luxury    furniture, jewelry, lacquerware
#   sundries  leather, beer

### TEXTILE

mnt_office_textile = {
	audio_tier = 3
	is_special = yes
	is_foreign = no
	max_levels = 1

	can_close = no
	ai_forbid_shutdown = yes
	ai_ignore_maintenance = yes
	automation_build_allowed = no

	pop_type = burghers
	employment_size = generic_burgher_employment

	category = government_category

	town = yes
	city = yes
	megalopolis = yes

	build_time = guild_build_time

	country_potential = { always = no }
	allow = { always = no }

	modifier = {
		local_cloth_output_modifier = 0.10
		local_fine_cloth_output_modifier = 0.10
	}

	unique_production_methods = {
		mnt_office_textile_maintenance = {
			icon_type = building_type

		}
	}
