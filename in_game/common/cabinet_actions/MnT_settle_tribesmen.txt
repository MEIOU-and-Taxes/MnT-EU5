		num_pop_type_in_country:tribesmen > 0
		#MnT restrictions on tribal and steppe countries
		has_tribal_government = no
	}

	allow = {
		trigger_if = {
			limit = {
				has_reform = government_reform:tle_support_of_banu_al_wad
			}
			NOT = { has_reform = government_reform:tle_support_of_banu_al_wad } # When removing, make sure to update pure_tooltip_entry = disables_tribal_cabinet_action in the gov reform
		}
		trigger_if = {
			limit = {
				has_estate_privilege = estate_privilege:tun_arab_bedouins
			}
			NOT = { has_estate_privilege = estate_privilege:tun_arab_bedouins } # When removing, make sure to update pure_tooltip_entry = disables_tribal_cabinet_action in the privilege
		}
	}

	allow_multiple = yes
	
	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		name = "settle_tribesmen_select_province"
		none_available_msg_key = "settle_tribesmen_no_provinces"
		column = {
			data = name
		}

			data = province_tribesmen
		}
		column = {
			data = population

		column = {
			data = province_tax_base
		}
		visible = {

		}
		map_color = {

			}
		}
	}

	location_modifier = {
		local_tribal_promotion = 5.0
	}

	is_finished = {
		scope:target ?= {
			num_pop_type_in_province:tribesmen <= 0
		}
	}

	map_marker = {
		show_for_owner = yes
	}
}
