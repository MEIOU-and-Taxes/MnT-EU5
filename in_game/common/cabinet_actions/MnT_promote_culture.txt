REPLACE:promote_culture = {
	ability = dip
	
	allow_multiple = yes
	
	potential = {
		num_locations > 0 
	}
	
	allow = {
		cultural_unity < 1.0
		NAND = { #mnt adds this to prevent subjects of the same culture as the overlord from also culture converting
			has_limited_diplomacy = yes
			top_overlord ?= { 
				culture = {
					is_primary_in = root
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = culture
		target_flag = target
		source_ai_override = actor
		source_flags_ai_override = include_any_present
		name = "promote_culture_select_culture"
		none_available_msg_key = "promote_culture_no_culture"
		column = {
			data = name
		}
		column = {
			data = culture_percentage_in_country
		}
		default_sort = culture_percentage_in_country
		visible = {
			scope:actor = {
				OR = {
					is_ai = no
					OR = {
						culture_percentage_in_country = {
							culture = root
							value >= 0.01
						}
						root = {
							is_primary_in = scope:actor
						}
					}
				}
			}
		}
		enabled = {
			is_primary_or_accepted_in = scope:actor
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_1
		name = "promote_culture_select_province"
		none_available_msg_key = "promote_culture_no_provinces"
		column = {
			data = name
		}
		column = {
			data = province_culture_percentage
		}
		column = {
			data = population
		}
		default_sort = province_culture_percentage
		visible = {
			any_location_in_province = {
				OR = {
					NOT = { dominant_culture = scope:target }
					local_cultural_unity < 0.95
				}
			}
			OR = {
				"culture_percentage(scope:target)" < 0.95
				any_location_in_province = { NOT = { dominant_culture = scope:target } }
			}
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = promote_culture interaction_target = target_1 } }
		}
		enabled = {
			OR = {
				scope:actor = { is_ai = no }
				AND = {
					trigger_if = {
						limit = { this = scope:actor.capital.province }
						scope:target = scope:actor.culture
					}
					any_location_in_province = { naturally_assimilating_culture = scope:target }
					trigger_if = {
						limit = { exists = dominant_culture }
						OR = {
							AND = {
								NOT = { dominant_culture = scope:target }
								dominant_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
							}
							AND = {
								dominant_culture = scope:target
								any_location_in_province = {
									exists = secondary_culture
									secondary_culture = { NOT = { is_primary_or_accepted_in = scope:actor } }
								}
							}
						}
					}
					trigger_else = { always = no }
				}
			}
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = "province.culture_percentage(scope:target)"
			}
		}
	}

	is_finished = {
		scope:target_1 = {
			"culture_percentage(scope:target)" >= 0.95
			not = {
				any_location_in_province = {
					NOT = { dominant_culture = scope:target }
				}
			}
		}
	}

	on_activate = {
		if = {
			limit = {
				exists = scope:target_1
			}
			scope:target_1 = {
				set_variable = {
					name = promoting_culture
					value = scope:target
				}
			}
		}
	}

	on_deactivate = {
		if = {
			limit = {
				exists = scope:target_1
			}
			scope:target_1 = {
				remove_variable = promoting_culture
			}
		}
	}

	location_modifier = {
		local_pop_assimilation_speed = 0.016 #mnt: from 0.04
		local_migration_attraction = -0.15
		local_pop_assimilation_speed_modifier = 0.2 #added by M&T
	}

	progress = {
		scope:target_1 = {
			value = "culture_percentage(scope:target)"
		}
	}

	map_marker = {
		show_for_owner = yes
	}

	ai_will_do = {
		if = {
			limit = { scope:target_1 = { NOT = { dominant_culture = scope:target } } }
			if = {
				limit = { scope:target_1.dominant_culture = { is_discriminated_in = scope:actor } }
				add = 6
			}
			else_if = {
				limit = {
					scope:target_1.dominant_culture = {
						NOT = { is_primary_or_accepted_in = scope:actor }
						NOT = { is_tolerated_in = scope:actor }
						NOT = { is_discriminated_in = scope:actor }
					}
				}
				add = 4
			}
			else_if = {
				limit = { scope:target_1.dominant_culture = { is_tolerated_in = scope:actor } }
				add = 2
			}
			subtract = "scope:target_1.dominant_culture.culture_opinion_impact(scope:actor.culture)"
		}
		else = {
			if = {
				limit = { scope:target_1 = { any_location_in_province = { exists = secondary_culture  secondary_culture = { is_discriminated_in = scope:actor } } } }
				add = 6
			}
			else_if = {
				limit = { scope:target_1 = { any_location_in_province = { exists = secondary_culture  secondary_culture = { NOT = { is_primary_or_accepted_in = scope:actor } NOT = { is_tolerated_in = scope:actor } NOT = { is_discriminated_in = scope:actor } } } } }
				add = 4
			}
			else_if = {
				limit = { scope:target_1 = { any_location_in_province = { exists = secondary_culture  secondary_culture = { is_tolerated_in = scope:actor } } } }
				add = 2
			}
			if = {
				limit = { exists = scope:target_1.assimilating_from_culture }
				subtract = "scope:target_1.assimilating_from_culture.culture_opinion_impact(scope:actor.culture)"
			}
		}
		multiply = {
			value = 1
			subtract = "scope:target_1.culture_percentage(scope:target)"
		}
	}
}
