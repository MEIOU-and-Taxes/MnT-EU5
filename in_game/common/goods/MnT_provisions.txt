# Provisions are the only good that yields food. Raw foodstuffs are turned into
# provisions before anyone eats them.

provisions = {
	category = produced
	color = goods_provisions

	default_market_price = 3
	food = 30

	# Untradeable: ban_exports/ban_imports in country_base_values
	transport_cost = 1

	# One provisions is 30 food, so the base is one food's worth and each pop
	# multiplies it by what it actually eats, from MnT_pop_types.txt.
	demand_add = {
		all = 0.03333
	}

	demand_multiply = {
		nobles = 5
		clergy = 2
		burghers = 3
		soldiers = 2
		laborers = 1
		peasants = 1
		slaves = 1
		tribesmen = 0
	}

}
