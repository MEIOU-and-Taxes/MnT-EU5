# Named conditions shared across the wealth system.

mnt_wealth_estate_is_present = {
	is_key_in_variable_map = { name = mnt_wealth_cached_local_relative_estate_power target = $KEY$ }
}

# Scope: country. The cadence: 0 fast, 1 medium, 2 slow; unset is medium. Every AI country is slow.
# Slow runs the country's locations at the AI pace, with no monthly pass. Fast runs the reconciliation and census every month.

mnt_wealth_cadence_is_slow = {
	OR = {
		is_ai = yes
		AND = {
			has_variable = mnt_wealth_cadence
			var:mnt_wealth_cadence = 2
		}
	}
}

mnt_wealth_cadence_is_fast = {
	is_human = yes
	has_variable = mnt_wealth_cadence
	var:mnt_wealth_cadence = 0
}

mnt_wealth_cadence_is_medium = {
	is_human = yes
	NOT = { mnt_wealth_cadence_is_fast = yes }
	NOT = { mnt_wealth_cadence_is_slow = yes }
}
