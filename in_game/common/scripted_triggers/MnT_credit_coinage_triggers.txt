# PM access is checked here so banking countries can bypass advance unlocks.
mnt_credit_bank_pm_available = {
	OR = {
		has_reform = government_reform:banking_country
		AND = {
			has_advance = $advance$
			mnt_credit_coinage_$coinage$ = yes
		}
	}
}

# Historical local houses can operate before the advance unlocks new construction.

	}
}

# Location scope, used only by the hidden banking-company expansion event.
mnt_credit_ai_bank_location = {
	exists = owner
	exists = market
	NOT = { location_rank = location_rank:rural_settlement }
	num_civil_constructions = 0
	can_build_building = scope:mnt_credit_ai_bank_type
	location_building_level = {
		building_type = scope:mnt_credit_ai_bank_type
		owner = scope:mnt_credit_ai_bank_builder
		value < mnt_bank_max_level
	}
	market = {
		"goods_demand_in_market(goods:credit)" > "goods_supply_in_market(goods:credit)"
	}
}
