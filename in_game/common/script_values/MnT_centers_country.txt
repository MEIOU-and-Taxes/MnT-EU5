# M&T Centers of Importance - culture centre scales per tier
#
# `scales_with` in mnt_center_culture_capacity_country / _prestige_country multiplies the auto modifier
# effects by these. The modifier bases are the world-tier values (cultures_capacity 1, monthly_prestige 0.1),
# so a scale of 1 = world and the ladder reproduces M&T's own per-tier numbers exactly:
#   capacity 0.025 / 0.05 / 0.25 / 1        prestige 0.01 / 0.025 / 0.05 / 0.1
# The tier variable (0-4) is written by the yearly pass and refreshed when a centre changes hands.

center_culture_capacity_scale = {
	value = 0
	if = { limit = { var:mnt_center_culture_tier = 4 } value = 1 }
	else_if = { limit = { var:mnt_center_culture_tier = 3 } value = 0.25 }
	else_if = { limit = { var:mnt_center_culture_tier = 2 } value = 0.05 }
	else_if = { limit = { var:mnt_center_culture_tier = 1 } value = 0.025 }
}

center_culture_prestige_scale = {
	value = 0
	if = { limit = { var:mnt_center_culture_tier = 4 } value = 1 }
	else_if = { limit = { var:mnt_center_culture_tier = 3 } value = 0.5 }
	else_if = { limit = { var:mnt_center_culture_tier = 2 } value = 0.25 }
	else_if = { limit = { var:mnt_center_culture_tier = 1 } value = 0.1 }
}
