transhumant_wool_pasture_cap = { #M&T created for special pastoral flocks
	add = { # Transhumance grows with how many tribesmen there are
		desc = "BUILDING_LEVEL_TRIBESMEN"
		value = num_pop_type:tribesmen
	}
	subtract = { # Transhumance is deterred by too many people
		desc = "BUILDING_LEVEL_LAND"
		add = num_pop_type:tribesmen
		multiply = {
			value = location_population_percentage
		}
	}
	# No description for this condition since it will never be visible
	if = {
		limit = {			
			# Limitations based on where wooly critters were herdered
			MnT_can_have_nomadic_wool_pasture = no
		}
		multiply = 0
	}
}

REPLACE:market_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 2 #M&T changed from 1 to 2
	}
	add = {
		desc = "BUILDING_LEVEL_DEVELOPMENT"
		value = development
		multiply = 0.1
	}
	add = {
		desc = "BUILDING_LEVEL_POPULATION"
		value = population
		multiply = 0.05
	}
	if = {
		limit = {
			location_rank ?= location_rank:city
		}
		add = {
			desc = "BUILDING_LEVEL_IS_CITY"
			value = 5
		}
	}
	if = {
		limit = {
			location_rank ?= location_rank:megalopolis
		}
		add = {
			desc = "BUILDING_LEVEL_IS_MEGAPOLIS"
			value = 10
		}
	}
	
	if = {
		limit = {
			owner ?= { modifier:market_building_levels != 0 }
		}
		multiply = {
			desc = "BUILDING_LEVEL_FROM_OWNER_MODIFIERS"
			value = 1
			add = owner.modifier:market_building_levels
		}
	}
}



urban_amenities_max_level = { #M&T created to model the infrastructure of cities, should help throttle urbanization
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	if = {
		limit = {
			location_rank = location_rank:town
		}
		add = {
			desc = "BUILDING_LEVEL_IS_TOWN"
			value = 4
		}
	}
	if = {
		limit = {
			location_rank = location_rank:city
		}
		add = {
			desc = "BUILDING_LEVEL_IS_CITY"
			value = 8
		}
	}
	if = {
		limit = {
			location_rank = location_rank:megalopolis
		}
		add = {
			desc = "BUILDING_LEVEL_IS_MEGAPOLIS"
			value = 12
		}
	}
	if = {
		limit = {
			owner = { has_advance = province_governors }
		}
		add = {
			desc = "province_governors"
			value = 1
		}
	}
	if = {
		limit = {
			modifier:local_urban_amenities_building_levels > 0
		}
		add = {
			desc = "major_center"
			value = modifier:local_urban_amenities_building_levels
		}
	}
	add = {
		desc = "BUILDING_LEVEL_DEVELOPMENT"
		value = development
		multiply = 0.04	#adds one maximum per 25 development
	}
	add = { 
		desc = "BUILDING_LEVEL_POPULATION"
		value = population 
		multiply = 0.002
	}
	if = {
		limit = {
			location_rank = rural_settlement
		}
		set_variable = {
			desc = "RURAL_SETTLEMENT_CAP" #rural settlements should not be able to build more than one level maximum, needs to be at the end
			name = urban_amenities_max_level
			value = 1
		}
	}

}
