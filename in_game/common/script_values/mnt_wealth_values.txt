# Simulation values. Display values are in mnt_wealth_display_values.txt.

mnt_wealth_months_since_campaign_start = {
	value = current_year
	subtract = mnt_wealth_campaign_start_year
	multiply = 12
	add = current_month
	subtract = 4
}

# The estate savings goal grows smoothly with game time: 2 to the power of the doubling periods passed.
mnt_wealth_estate_savings_time_multiplier = {
	value = mnt_wealth_months_since_campaign_start
	divide = 12
	divide = mnt_wealth_estate_savings_goal_doubling_years
	min = 0
	save_temporary_value_as = mnt_wealth_savings_doubling_periods
	value = 2
	pow = scope:mnt_wealth_savings_doubling_periods
}

mnt_wealth_minimum_wage_rate = {
	value = modifier:mnt_wealth_local_minimum_wage
	if = { limit = { has_owner = yes } add = owner.modifier:mnt_wealth_global_minimum_wage }
}

mnt_wealth_investment_rate_multiplier = {
	value = 1
	add = modifier:mnt_wealth_local_investment_rate_efficiency
	if = { limit = { has_owner = yes } add = owner.modifier:mnt_wealth_global_investment_rate_efficiency }
	min = 0.5
}

mnt_wealth_investment_rate = {
	value = modifier:mnt_wealth_local_investment_rate
	if = { limit = { has_owner = yes } add = owner.modifier:mnt_wealth_global_investment_rate }
	multiply = mnt_wealth_investment_rate_multiplier
}

mnt_wealth_investment_efficiency_multiplier = {
	value = 1
	add = modifier:mnt_wealth_local_investment_efficiency
	add = owner.modifier:mnt_wealth_global_investment_efficiency
	min = 0
}

# 1 / (1 + resistance), floored at 0.5; higher resistance is cheaper decay.

# The rates recorded at the last settle, which the advance applies to the owed months and the display projects with.
# Before a location's first settle records them, the live value stands in.

mnt_wealth_recorded_investment_rate = {
	value = mnt_wealth_investment_rate
	if = { limit = { has_variable = mnt_wealth_cached_investment_rate } value = var:mnt_wealth_cached_investment_rate }
}

mnt_wealth_recorded_investment_maintenance = {
	value = modifier:mnt_wealth_local_investment_maintenance
	if = { limit = { has_variable = mnt_wealth_cached_investment_maintenance } value = var:mnt_wealth_cached_investment_maintenance }
}

mnt_wealth_recorded_investment_efficiency = {
	value = mnt_wealth_investment_efficiency_multiplier
	if = { limit = { has_variable = mnt_wealth_cached_investment_efficiency } value = var:mnt_wealth_cached_investment_efficiency }
}

mnt_wealth_recorded_decay_resistance = {
	value = mnt_wealth_decay_resistance_multiplier
	if = { limit = { has_variable = mnt_wealth_cached_decay_resistance } value = var:mnt_wealth_cached_decay_resistance }
}

mnt_wealth_decay_resistance_multiplier = {
	value = 1

	divide = {
		value = 1
		add = modifier:mnt_wealth_local_decay_resistance
		add = owner.modifier:mnt_wealth_global_decay_resistance
		min = 0.5
	}
}

mnt_wealth_estate_flow_total = {
	value = 0
	save_temporary_scope_as = mnt_wealth_flow_location

	if = {
		limit = {
			has_owner = yes
			owner = { has_variable_map = mnt_wealth_estate_flow has_variable_map = mnt_wealth_national_estate_raw_power }
		}
		owner = {
			every_key_in_variable_map = {
				variable = mnt_wealth_estate_flow
				save_temporary_scope_as = mnt_wealth_flow_key

				if = {
					limit = {
						scope:mnt_wealth_flow_location = {
							has_variable_map = mnt_estate_raw_power
							is_key_in_variable_map = { name = mnt_estate_raw_power target = scope:mnt_wealth_flow_key }
						}
						prev = {
							has_variable_map = mnt_wealth_national_estate_raw_power is_key_in_variable_map = { name = mnt_wealth_national_estate_raw_power target = scope:mnt_wealth_flow_key }
							"variable_map(mnt_wealth_national_estate_raw_power|scope:mnt_wealth_flow_key)" > 0
						}
					}
					prev = {
						add = {
							value = "variable_map(mnt_wealth_estate_flow|scope:mnt_wealth_flow_key)"
							multiply = "scope:mnt_wealth_flow_location.variable_map(mnt_estate_raw_power|scope:mnt_wealth_flow_key)"
							divide = "variable_map(mnt_wealth_national_estate_raw_power|scope:mnt_wealth_flow_key)"
						}
					}
				}
			}
		}
	}
}

mnt_wealth_local_estate_wealth_fraction = {
	value = 0

	scope:mnt_wealth_location = {
		if = {
			limit = {
				var:mnt_wealth_pool > 0
				has_variable_map = mnt_wealth_estate_wealth is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = scope:mnt_wealth_estate }
			}
			add = "variable_map(mnt_wealth_estate_wealth|scope:mnt_wealth_estate)"
			divide = var:mnt_wealth_pool
		}
	}
}

# The flow running totals are kept per million raw power, not per unit. The engine holds five decimals, and a
# month's flow divided by a large country's raw power is a few hundred-thousandths, which would store with one or
# two significant digits and drift the credits by hundredths of a ducat a month. Scaled up they keep full precision.
mnt_wealth_flow_raw_power_scale = 1000000
