#  Which locations qualify for a building: can_build_*_RGO in in_game/common/scripted_triggers/MnT_RGO_substitution.txt. These are now cached into location variables for a few of the more complex triggers.

Add a new modifier for it here: modifier type in main_menu/common/modifier_type_definitions/mnt_modifier_types.txt, icon, loc, then a row in mnt_location_max_rgo_building_level. Then use that number however you want in the specific limits you're looking at. And add it to an INJECT/REPLACE on any systems you want it to scale with.

# Max level calculation for the RGO replacement buildings.
# The whole base is one breakdown row: mnt_rgo_base_size, plus mnt_rgo_mining_size for mining goods, multiplied by the ideal area bonus where the location's raw material matches the building's good.
# Its label spells out that arithmetic, so every row under it is plainly a percentage of the number the base row shows. Base size modifiers are set in main_menu/common/static_modifiers/MnT_location.txt.
# The ideal rows are written out rather than factored into a shared value. A script value whose own root level ends in multiply breaks the breakdown when it is read from one of these rows, so the flat base is saved and multiplied back in place.
# Unowned locations return 0. Final value floors at 1 where the raw material matches and at 0 elsewhere.

# Local, national and settlement rank size modifiers. All are percentages of scope:base_value, which already includes the ideal area bonus.
mnt_rgo_size_modifiers = {
	if = {
		limit = { modifier:local_max_rgo_size_modifier != 0 }
		add = { desc = "LOCAL_MAX_RGO"
			value = scope:base_value
			multiply = modifier:local_max_rgo_size_modifier
		}
	}
	if = {
		limit = { owner.modifier:global_max_rgo_size_modifier != 0 }
		add = { desc = "GLOBAL_MAX_RGO"
			value = scope:base_value
			multiply = owner.modifier:global_max_rgo_size_modifier
		}
	}
	if = {
		limit = {
			location_rank = location_rank:rural_settlement
			owner.modifier:global_max_rgo_size_modifier_in_rural != 0
		}
		add = { desc = "RURAL_MAX_RGO"
			value = scope:base_value
			multiply = owner.modifier:global_max_rgo_size_modifier_in_rural
		}
	}
	else_if = {
		limit = {
			NOT = { location_rank = location_rank:rural_settlement }
			owner.modifier:global_max_rgo_size_modifier_in_non_rural != 0
		}
		add = { desc = "URBAN_MAX_RGO"
			value = scope:base_value
			multiply = owner.modifier:global_max_rgo_size_modifier_in_non_rural
		}
	}
}

# Goods whose RGO building can only stand on its own raw material, so the ideal area bonus always applies.
rgo_standard_max_level = {
	if = {
		limit = { has_owner = yes }
		add = { desc = "MNT_RGO_BASE_IDEAL"
			value = modifier:mnt_rgo_base_size
			save_temporary_value_as = base_before_ideal
			value = 1
			add = modifier:local_raw_material_output
			add = owner.modifier:global_raw_material_output
			multiply = scope:base_before_ideal
		}
		save_temporary_value_as = base_value
		add = mnt_rgo_size_modifiers
		min = 1
	}
}
