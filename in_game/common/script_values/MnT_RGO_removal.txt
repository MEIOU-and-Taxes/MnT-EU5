#Max level calculation for the RGO replacement buildings.
#Every good adds mnt_rgo_base_size; categories add their mnt_rgo_*_size delta on top. Both set in main_menu/common/static_modifiers/MnT_location.txt.
#Unowned locations return 0.

#Farming goods only buildable on their raw material, level floors at 1.
rgo_standard_max_level = {
	if = {
		limit = { exists = owner }
		add = { desc = "MNT_RGO_BASE_SIZE"
			value = modifier:mnt_rgo_base_size
		}
		save_temporary_value_as = base_value
		min = 1
		if = {
			limit = { modifier:local_max_rgo_size_modifier != 0 }
			add = { desc = "LOCAL_MAX_RGO"
				value = scope:base_value
				multiply = modifier:local_max_rgo_size_modifier
			}
		}
		if = {
			limit = { owner.modifier:global_max_rgo_size_modifier != 0 }
			add = { desc = "GLOBAL_MAX_RGO"
				value = scope:base_value
				multiply = owner.modifier:global_max_rgo_size_modifier
			}
		}
		add = { desc = "BUILDING_LEVEL_RAW_MATERIALS_BONUS_LOCAL"
			value = scope:base_value
			multiply = modifier:local_raw_material_output
		}
		if = {
			limit = { owner.modifier:global_raw_material_output != 0 }
			add = { desc = "BUILDING_LEVEL_RAW_MATERIALS_BONUS_GLOBAL"
				value = scope:base_value
				multiply = owner.modifier:global_raw_material_output
			}
		}
		if = {
			limit = {
				location_rank = location_rank:rural_settlement
				owner.modifier:global_max_rgo_size_modifier_in_rural != 0
			}
			add = { desc = "RURAL_MAX_RGO"
				value = scope:base_value
				multiply = owner.modifier:global_max_rgo_size_modifier_in_rural
			}
		}
		else_if = {
			limit = {
				NOT = { location_rank = location_rank:rural_settlement }
				owner.modifier:global_max_rgo_size_modifier_in_non_rural != 0
			}
			add = { desc = "URBAN_MAX_RGO"
				value = scope:base_value
				multiply = owner.modifier:global_max_rgo_size_modifier_in_non_rural
			}
		}
	}
}
