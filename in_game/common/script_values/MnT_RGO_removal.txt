#Where to tune the RGO max level system:
#  Base sizes: mnt_rgo_<method>_size for each goods method, plus mnt_rgo_<good>_size for each good, in main_menu/common/static_modifiers/MnT_location.txt (location_base_values for the flat part, development for the per-dev part).

#  Bog iron ceiling: bog_iron_cap_level in country_base_values, in_game/common/auto_modifiers/MnT_country.txt; advances add to it.
#  Which locations qualify for a building: can_build_*_RGO in in_game/common/scripted_triggers/MnT_RGO_substitution.txt. These are now cached into location variables for a few of the more complex triggers.
#  I want to specifically change one RGO limit. Add a new modifier for it here: modifier type in main_menu/common/modifier_type_definitions/mnt_modifier_types.txt, icon, loc, then a row in mnt_location_max_rgo_building_level. Then use that number however you want in the specific limits you're looking at. And add it to an INJECT/REPLACE on any systems you want it to scale with.
#  New RGO good or a change to the max level layout: edit the max_levels templates of tools/generators_from_game_data/generators/rgo_building/rgo_building.py, run it, and paste its output.

# Max level calculation for the RGO replacement buildings.
# Base = method size + good size + ideal area bonus (matching raw material only). Size modifier rows are percentages of the base.
# Every operation in the breakdown has its own desc; an operation without one shows as a "missing key" row.
# Each percentage row shows the levels it adds, so the rows add up to the total.
# Unowned locations return 0. Final value floors at 1 where the raw material matches and at 0 elsewhere.

# Local, national and settlement rank size modifier rows.
mnt_rgo_size_modifiers = {
	if = {
		limit = { modifier:local_max_rgo_size_modifier != 0 }
		save_temporary_value_as = {
			name = local_max_rgo_amount
			value = {
				value = scope:base_value
				multiply = modifier:local_max_rgo_size_modifier
			}
		}
		add = { desc = "LOCAL_MAX_RGO" value = scope:local_max_rgo_amount }
	}
	if = {
		limit = { owner.modifier:global_max_rgo_size_modifier != 0 }
		save_temporary_value_as = {
			name = global_max_rgo_amount
			value = {
				value = scope:base_value
				multiply = owner.modifier:global_max_rgo_size_modifier
			}
		}
		add = { desc = "GLOBAL_MAX_RGO" value = scope:global_max_rgo_amount }
	}
	if = {
		limit = {
			location_rank = location_rank:rural_settlement
			owner.modifier:global_max_rgo_size_modifier_in_rural != 0
		}
		save_temporary_value_as = {
			name = rural_max_rgo_amount
			value = {
				value = scope:base_value
				multiply = owner.modifier:global_max_rgo_size_modifier_in_rural
			}
		}
		add = { desc = "RURAL_MAX_RGO" value = scope:rural_max_rgo_amount }
	}
	else_if = {
		limit = {
			NOT = { location_rank = location_rank:rural_settlement }
			owner.modifier:global_max_rgo_size_modifier_in_non_rural != 0
		}
		save_temporary_value_as = {
			name = urban_max_rgo_amount
			value = {
				value = scope:base_value
				multiply = owner.modifier:global_max_rgo_size_modifier_in_non_rural
			}
		}
		add = { desc = "URBAN_MAX_RGO" value = scope:urban_max_rgo_amount }
	}
}

# Farming goods

rgo_beeswax_max_level = {
	if = {
		limit = { has_owner = yes }
		if = {
			limit = { raw_material = goods:beeswax }
			add = {
				desc = "MNT_RGO_BASE_IDEAL_BEESWAX"
				add = { desc = "MNT_RGO_SIZE_FARMING" value = modifier:mnt_rgo_farming_size }
				add = { desc = "MNT_RGO_SIZE_BEESWAX" value = modifier:mnt_rgo_beeswax_size }
				#Work out each amount first, so the tooltip shows it as one row instead of one row per step.
				save_temporary_value_as = {
					name = ideal_area_local_amount
					value = {
						value = modifier:mnt_rgo_farming_size
						add = modifier:mnt_rgo_beeswax_size
						multiply = modifier:local_raw_material_output
					}
				}
				add = { desc = "MNT_IDEAL_AREA_LOCAL_SIZE_MODIFIER" value = scope:ideal_area_local_amount }
				save_temporary_value_as = {
					name = ideal_area_amount
					value = {
						value = modifier:mnt_rgo_farming_size
						add = modifier:mnt_rgo_beeswax_size
						multiply = owner.modifier:global_raw_material_output
					}
				}
				add = { desc = "MNT_IDEAL_AREA_SIZE_MODIFIER" value = scope:ideal_area_amount }
			}
			save_temporary_value_as = base_value
			if = {
				limit = { mnt_rgo_size_modifiers != 0 }
				add = mnt_rgo_size_modifiers
			}
			save_temporary_value_as = mnt_rgo_level
			if = {
				limit = { scope:mnt_rgo_level < 1 }
				min = { desc = "MNT_RGO_MINIMUM_LEVEL" value = 1 }
			}
		}
		else = {
			add = {
				desc = "MNT_RGO_BASE_BEESWAX"
				add = { desc = "MNT_RGO_SIZE_FARMING" value = modifier:mnt_rgo_farming_size }
				add = { desc = "MNT_RGO_SIZE_BEESWAX" value = modifier:mnt_rgo_beeswax_size }
			}
			save_temporary_value_as = base_value
			if = {
				limit = { mnt_rgo_size_modifiers != 0 }
				add = mnt_rgo_size_modifiers
			}
		}
		save_temporary_value_as = mnt_rgo_level
		if = {
			limit = { scope:mnt_rgo_level < 0 }
			min = { desc = "MNT_RGO_MINIMUM_LEVEL" value = 0 }
		}
	}
}

				add = { desc = "MNT_RGO_SIZE_SAND" value = modifier:mnt_rgo_sand_size }
				#Work out each amount first, so the tooltip shows it as one row instead of one row per step.
				save_temporary_value_as = {
					name = ideal_area_local_amount
					value = {
						value = modifier:mnt_rgo_gathering_size
						add = modifier:mnt_rgo_sand_size
						multiply = modifier:local_raw_material_output
					}
				}
				add = { desc = "MNT_IDEAL_AREA_LOCAL_SIZE_MODIFIER" value = scope:ideal_area_local_amount }
				save_temporary_value_as = {
					name = ideal_area_amount
					value = {
						value = modifier:mnt_rgo_gathering_size
						add = modifier:mnt_rgo_sand_size
						multiply = owner.modifier:global_raw_material_output
					}
				}
				add = { desc = "MNT_IDEAL_AREA_SIZE_MODIFIER" value = scope:ideal_area_amount }
			}
			save_temporary_value_as = base_value
			if = {
				limit = { mnt_rgo_size_modifiers != 0 }
				add = mnt_rgo_size_modifiers
			}
			save_temporary_value_as = mnt_rgo_level
			if = {
				limit = { scope:mnt_rgo_level < 1 }
				min = { desc = "MNT_RGO_MINIMUM_LEVEL" value = 1 }
			}
		}
		else = {
			add = {
				desc = "MNT_RGO_BASE_SAND"
				add = { desc = "MNT_RGO_SIZE_GATHERING" value = modifier:mnt_rgo_gathering_size }
				add = { desc = "MNT_RGO_SIZE_SAND" value = modifier:mnt_rgo_sand_size }

				limit = { mnt_rgo_size_modifiers != 0 }
				add = mnt_rgo_size_modifiers
			}
		}
