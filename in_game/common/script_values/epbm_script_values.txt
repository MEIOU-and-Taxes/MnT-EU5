# The auto_modifiers read the last_month scale, copied at the moment of the
# charge, so the refund matches the bill the estates actually paid. The map is
# keyed by estate_type; an estate the country lacks has no key and reads 0.

epbm_nobles_upkeep_scale = {
	if = {
		limit = { is_key_in_variable_map = { name = epbm_upkeep_scale_last_month target = estate_type:nobles_estate } }
		value = "variable_map(epbm_upkeep_scale_last_month|estate_type:nobles_estate)"
	}
}

# GUI display script values. Evaluated from estate scope:
# estate -> owner (country) -> read variable map with prev (estate) as key.
#
# The estate tooltip sits beside vanilla's "Events and other sources", which
# reports the tick before last, so it reads the two_months_ago tier. The
# economic balance breakdown reports the last tick, so it reads last_month.

epbm_maintenance_charged_last_month_display = {
	owner = {
		if = {
			limit = { has_variable_map = epbm_maintenance_charged_last_month is_key_in_variable_map = { name = epbm_maintenance_charged_last_month target = prev } }
			value = "variable_map(epbm_maintenance_charged_last_month|prev)"
		}
	}
}
