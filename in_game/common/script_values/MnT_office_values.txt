# Centers of Importance - office subvention cost
#
# Mirrors vanilla's university construction basket (goods_demand/building_construction_costs.txt:
# masonry 2, paper 2, lumber 1, books 2, tools 1) priced at the local market, so the sum scales
# with the economy instead of being a fixed number.
#
# Requires scope:mnt_office_location, saved by the mnt_offices.1 event's immediate block.

mnt_office_subvention = {
	value = 0
	add = {
		value = 2
		multiply = "goods:masonry.price_in_market(scope:mnt_office_location.market)"
	}
	add = {
		value = 2
		multiply = "goods:paper.price_in_market(scope:mnt_office_location.market)"
	}
	add = {
		value = 1
		multiply = "goods:lumber.price_in_market(scope:mnt_office_location.market)"
	}
	add = {
		value = 2
		multiply = "goods:books.price_in_market(scope:mnt_office_location.market)"
	}
	add = {
		value = 1
		multiply = "goods:tools.price_in_market(scope:mnt_office_location.market)"
	}
}
