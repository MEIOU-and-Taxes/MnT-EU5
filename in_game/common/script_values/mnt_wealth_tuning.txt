# Tuning constants for the wealth system.

# Yearly drift of local wealth percentage toward actual power. 0.06 halves the gap in about eleven years.
mnt_wealth_local_wealth_share_trend_rate = 0.06
# Monthly upkeep change per investment or neglect level.
mnt_wealth_upkeep_change_per_investment_level = 0.01
# Controls how fast desired wealth drops with neglect depth: base / (1 + this x depth).
mnt_wealth_desired_wealth_per_investment_level = 0.5
# Gold cost of the first investment level. Each deeper level costs this times level squared.
mnt_wealth_investment_level_cost_base = 50
# How often the building-type lookup table is rebuilt in months.
mnt_wealth_building_table_lifetime_months = 120
# Months in a full GDP rotation. 60 means each location is surveyed once every 5 years.
mnt_wealth_gdp_cycle_months = 60

# How long each load balancer index stays valid. The volatile ones expire within
# the month so the first worker of the next month rebuilds them. The location
# index is maintained in place rather than rebuilt, so it never expires.
mnt_wealth_month_lifetime = 27

mnt_wealth_country_index_lifetime = 27

mnt_wealth_human_location_index_lifetime = 27

mnt_wealth_market_index_lifetime = 27

mnt_wealth_location_index_lifetime = -1


# Factor for rounding the wage penalty before applying it.
mnt_wealth_wage_rounding_factor = 0.1
# Factor for rounding the investment and neglect modifiers.
mnt_wealth_investment_rounding_factor = 0.1
# Starting pool as a multiple of desired wealth. 1 means the world begins at desired wealth.
mnt_wealth_starting_savings_multiplier = 1
# Caps how fast an estate whose treasury exceeds its savings goal sends national
# investment to its locations, measured in economic bases per month. The cap
# ignores estate power so a numerous but powerless estate can pay out as fast as
# a strong one once it has the gold to do it.
mnt_wealth_national_investment_limit_factor = 0.25
# The seizure cap starts here and the estate's shortfall demand is added on top,
# measured in savings goals per month.
mnt_wealth_estate_seizure_limit_factor = 0.25
# How fast an estate's demand for seizure fades once its treasury is back at the
# savings goal. At 0.9 per month, about a hundredth of the demand remains after
# four years.
mnt_wealth_seizure_demand_decay = 0.9
# The initial resistance a freshly looted location gains and its decay time are
# both set on the mnt_wealth_recently_looted static modifier, not here.
# This caps total looting resistance so looting can never be fully blocked.
mnt_wealth_loot_resistance_cap = 0.9
# How much of each pool is looted per sacking is set on the two looted modifiers
# in country base values, the same place as the liquid share.
# Political weight per person for each estate. Used to convert raw power into a
# head count. Seeded into the mnt_wealth_power_per_pop global map at game start.
# These must match power_per_pop in common/estates: M&T_default.txt overrides
# peasants, dhimmi, tribes and cossacks; the rest fall through to vanilla.
mnt_wealth_nobles_power_per_pop = 25

mnt_wealth_clergy_power_per_pop = 10

mnt_wealth_burghers_power_per_pop = 4

mnt_wealth_peasants_power_per_pop = 0.15

mnt_wealth_dhimmi_power_per_pop = 0.1

mnt_wealth_tribes_power_per_pop = 0.2

mnt_wealth_cossacks_power_per_pop = 0.2
