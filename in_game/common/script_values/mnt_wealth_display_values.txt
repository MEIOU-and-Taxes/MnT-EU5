# Script values the interface reads: the location window, its tooltips, the map
# modes and the map mode tooltip strings. Nothing here is read by the
# simulation; those values live in mnt_wealth_values.txt.

# --- the display month ---

# Months still to apply to the stored state. Every projection below replays
# exactly these months with the formulas the advance uses, so the window
# shows what the next review will write.

mnt_wealth_months_since_review = {
	value = mnt_wealth_months_since_campaign_start
	subtract = var:mnt_wealth_review_month
}

# --- pool and health ---

# The share of the wealth pool left after a month's investment, and after
# all the months since the review. A negative rate spends nothing.

mnt_wealth_projected_pool_remaining = {
	value = 1
	subtract = { value = mnt_wealth_investment_rate min = 0 }
}

mnt_wealth_projected_pool_remaining_over_period = {
	value = mnt_wealth_projected_pool_remaining
	pow = mnt_wealth_projected_months
}

# What a steady monthly income adds up to over the period, once the
# investment taken from it each month is counted. Just the month count when
# nothing is invested.

mnt_wealth_projected_income_over_period = {
	value = mnt_wealth_investment_rate
	min = 0
	save_temporary_value_as = mnt_wealth_investment_rate

	if = {
		limit = { scope:mnt_wealth_investment_rate > 0 }
		value = 1
		subtract = mnt_wealth_projected_pool_remaining_over_period
		divide = scope:mnt_wealth_investment_rate
	}
	else = {
		value = mnt_wealth_projected_months
	}
}

# The wealth pool now: what was there decays by the investment taken from
# it, and the period's income, kept wages plus net estate flow, adds up on
# top. The same series the advance runs, summed over the estates.

mnt_wealth_displayed_wealth_pool = {
	value = var:mnt_wealth_pool
	multiply = mnt_wealth_projected_pool_remaining_over_period
	add = {
		value = var:mnt_wealth_kept_income_total
		add = mnt_wealth_estate_flow_total
		multiply = mnt_wealth_projected_income_over_period
	}
	min = 0
}

mnt_wealth_local_wealth_total = {
	value = mnt_wealth_displayed_wealth_pool
}

# One estate's monthly income: its wage share plus its national flow share.
# The wage share is its pool fraction, or its relative power when the pool
# is empty. Reads scope:mnt_wealth_location and scope:mnt_wealth_estate.

mnt_wealth_projected_estate_income = {
	value = 0

	scope:mnt_wealth_location = {
		if = {
			limit = { var:mnt_wealth_pool > 0 }
			add = mnt_wealth_local_estate_wealth_fraction
		}
		else_if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_cached_local_relative_estate_power target = scope:mnt_wealth_estate } }
			add = "variable_map(mnt_wealth_cached_local_relative_estate_power|scope:mnt_wealth_estate)"
		}

		multiply = var:mnt_wealth_kept_income_total
		add = mnt_wealth_equilibrium_estate_flow
	}
}

# One estate's share of the location's raw power: its raw power over the
# location total the survey cached. Reads scope:mnt_wealth_location and
# scope:mnt_wealth_estate.

mnt_wealth_local_estate_raw_power_fraction = {
	value = 0

	scope:mnt_wealth_location = {
		if = {
			limit = {
				var:mnt_wealth_local_raw_power_total > 0
				is_key_in_variable_map = { name = mnt_estate_raw_power target = scope:mnt_wealth_estate }
			}
			add = {
				value = "variable_map(mnt_estate_raw_power|scope:mnt_wealth_estate)"
				divide = var:mnt_wealth_local_raw_power_total
			}
		}
	}
}

# One estate's share of the location's Local Investment Maintenance: the flat
# monthly cost split by the estate's share of the location's raw power.
# Reads scope:mnt_wealth_location and scope:mnt_wealth_estate.

mnt_wealth_displayed_estate_maintenance = {
	value = mnt_wealth_local_estate_raw_power_fraction

	scope:mnt_wealth_location = {
		multiply = modifier:mnt_wealth_local_investment_maintenance
	}
}

# One estate's wealth now: the estate's own run of the pool series.

mnt_wealth_displayed_estate_wealth = {
	value = 0

	scope:mnt_wealth_location = {
		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = scope:mnt_wealth_estate } }
			add = "variable_map(mnt_wealth_estate_wealth|scope:mnt_wealth_estate)"
		}

		multiply = mnt_wealth_projected_pool_remaining_over_period
		add = {
			value = mnt_wealth_projected_estate_income
			multiply = mnt_wealth_projected_income_over_period
		}
	}

	min = 0
}

mnt_wealth_baseline = {
	value = var:mnt_wealth_local_gross_output
	multiply = mnt_wealth_starting_gross_output_months
}

# Zero when there is no desired wealth to measure against, which means no estate
# is present here or the location has not been surveyed yet.

mnt_wealth_health_ratio = {
	value = 0

	if = {
		limit = { mnt_wealth_baseline > 0 }
		value = mnt_wealth_local_wealth_total
		divide = mnt_wealth_baseline
	}
}

mnt_wealth_band = {
	value = 0
	if = { limit = { mnt_wealth_health_ratio >= 0.5 } add = 1 }
	if = { limit = { mnt_wealth_health_ratio >= 0.9 } add = 1 }
	if = { limit = { mnt_wealth_health_ratio >= 1.1 } add = 1 }
}

# --- flow map mode ratio ---

# Net estate capital flow / monthly investment. Positive = investing, negative = seizing.
# flow_ratio = estate_investment_total / (wealth * investment_rate).
# Zero where there is no monthly investment to measure the flow against.

mnt_wealth_estate_flow_ratio = {
	value = 0

	if = {
		limit = { mnt_wealth_local_wealth_total > 0 mnt_wealth_investment_rate > 0 }
		value = mnt_wealth_estate_flow_total
		divide = { value = mnt_wealth_local_wealth_total multiply = mnt_wealth_investment_rate }
	}
}

# --- investment equilibrium ---

# The level an estate settles at is where its monthly investment exactly pays its
# monthly decay. Working that out means undoing the desired wealth curve, which
# is a square root that cannot be computed in the data binding language.
#
# Every value in this section runs in location scope and reads scope:mnt_wealth_estate,
# the estate type the tooltip row is showing.

# Entry point for the location window. The interface hands in the location and
# the estate as named scopes because the tooltip row knows both.

mnt_wealth_displayed_investment_equilibrium = {
	value = 0
	scope:mnt_wealth_location = { add = mnt_wealth_investment_equilibrium_level }
}

mnt_wealth_displayed_monthly_decay = {

}

# Entry point for the decay-rate macro: one estate's live monthly decay rate.

mnt_wealth_displayed_estate_decay_rate = {
	value = 0
	scope:mnt_wealth_location = { add = mnt_wealth_estate_decay_rate }
}

# The estate's projected absolute wealth at this location.

mnt_wealth_equilibrium_estate_wealth = {
	value = mnt_wealth_displayed_estate_wealth
}

# This location's slice of the estate's national flow: the national flow split
# by how much of the estate's raw power sits here. Positive when the estate is
# investing gold into the location, negative when it is seizing gold out.

mnt_wealth_equilibrium_estate_flow = {
	value = 0
	save_temporary_scope_as = mnt_wealth_equilibrium_location

	if = {
		limit = {
			has_variable_map = mnt_estate_raw_power
			is_key_in_variable_map = { name = mnt_estate_raw_power target = scope:mnt_wealth_estate }

			owner ?= {
				has_variable_map = mnt_wealth_estate_flow
				has_variable_map = mnt_wealth_national_estate_raw_power
				is_key_in_variable_map = { name = mnt_wealth_estate_flow               target = scope:mnt_wealth_estate }
				is_key_in_variable_map = { name = mnt_wealth_national_estate_raw_power target = scope:mnt_wealth_estate }
				"variable_map(mnt_wealth_national_estate_raw_power|scope:mnt_wealth_estate)" > 0
			}
		}
		owner = {
			add = {
				value = "variable_map(mnt_wealth_estate_flow|scope:mnt_wealth_estate)"
				multiply = "scope:mnt_wealth_equilibrium_location.variable_map(mnt_estate_raw_power|scope:mnt_wealth_estate)"
				divide = "variable_map(mnt_wealth_national_estate_raw_power|scope:mnt_wealth_estate)"
			}
		}
	}
}

# Gold a national seizure tears out of invested wealth this month: the part of a
# negative flow that the estate's liquid wealth cannot cover. Zero while the
# estate is investing or while its liquid wealth covers the whole seizure.

mnt_wealth_equilibrium_seizure_from_invested = {
	value = 0
	subtract = mnt_wealth_equilibrium_estate_flow
	subtract = mnt_wealth_equilibrium_estate_wealth
	min = 0
}

# The invested wealth an estate settles at: where a month's progress is zero.
# Spending at the rate, scaled by efficiency, less the seizure and the
# maintenance share, over the decay it pays per gold held. Negative when the
# charges outrun the spending: the estate settles at a neglect level. A
# negative rate spends nothing.

mnt_wealth_equilibrium_invested_wealth = {
	value = 0

	if = {
		limit = { mnt_wealth_estate_decay_rate > 0 }
		value = mnt_wealth_equilibrium_estate_wealth
		multiply = { value = mnt_wealth_investment_rate min = 0 }
		multiply = mnt_wealth_investment_efficiency_multiplier
		subtract = mnt_wealth_equilibrium_seizure_from_invested
		subtract = mnt_wealth_displayed_estate_maintenance
		divide = { value = mnt_wealth_estate_decay_rate multiply = mnt_wealth_decay_resistance_multiplier }
	}
}

# The per-estate monthly decay rate: local plus national modifier. Resolved by
# estate type because the modifier name carries it. Runs in location scope with
# scope:mnt_wealth_estate set, and reads the modifiers live.

mnt_wealth_estate_decay_rate = {
	value = 0
	if = {
		limit = { scope:mnt_wealth_estate = estate_type:nobles_estate }
		value = modifier:mnt_wealth_local_nobles_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_nobles_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:clergy_estate }
		value = modifier:mnt_wealth_local_clergy_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_clergy_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:burghers_estate }
		value = modifier:mnt_wealth_local_burghers_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_burghers_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:peasants_estate }
		value = modifier:mnt_wealth_local_peasants_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_peasants_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:dhimmi_estate }
		value = modifier:mnt_wealth_local_dhimmi_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_dhimmi_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:tribes_estate }
		value = modifier:mnt_wealth_local_tribes_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_tribes_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:cossacks_estate }
		value = modifier:mnt_wealth_local_cossacks_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_cossacks_investment_decay }
	}
}

# The level an estate settles at: the square-root curve over the equilibrium
# invested wealth, run on the magnitude and signed, so a neglect equilibrium
# reads negative. Zero is skipped because the engine's pow returns 1 for a
# zero base.

mnt_wealth_investment_equilibrium_level = {
	value = mnt_wealth_equilibrium_invested_wealth
	divide = mnt_wealth_investment_level_cost_base
	save_temporary_value_as = mnt_wealth_equilibrium_level_ratio

	value = 0
	if = {
		limit = { scope:mnt_wealth_equilibrium_level_ratio > 0 }
		value = scope:mnt_wealth_equilibrium_level_ratio
		pow = 0.5
	}
	else_if = {
		limit = { scope:mnt_wealth_equilibrium_level_ratio < 0 }
		value = scope:mnt_wealth_equilibrium_level_ratio
		multiply = -1
		pow = 0.5
		multiply = -1
	}
}

# Income is the kept wage total alone, matching the pass; estate flow arrives
# separately through the national payout.

mnt_wealth_projected_investment_paid = {
	value = 0

	if = {
		limit = {
			mnt_wealth_projected_months > 0
			mnt_wealth_investment_rate > 0
		}
		value = mnt_wealth_projected_effective_months
		save_temporary_value_as = mnt_wealth_proj_effective_months
		value = var:mnt_wealth_pool
		add = var:mnt_wealth_kept_income_total

			value = var:mnt_wealth_kept_income_total
			multiply = { value = mnt_wealth_projected_months subtract = scope:mnt_wealth_proj_effective_months }
			divide = mnt_wealth_review_spend_rate
		}

		multiply = mnt_wealth_investment_rate
	}

# One estate's invested wealth projected forward from the review: what it
# held decays, each month's investment lands and decays from then on, and
# each month's maintenance share is taken. Runs in location scope with
# scope:mnt_wealth_estate set.

mnt_wealth_projected_invested_wealth = {
	value = 0
	if = {
		limit = {
			has_variable_map = mnt_wealth_invested_wealth
			is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = scope:mnt_wealth_estate }
		}
		add = "variable_map(mnt_wealth_invested_wealth|scope:mnt_wealth_estate)"
	}
	if = {
		limit = {
			has_variable_map = mnt_wealth_invested_since_review
			is_key_in_variable_map = { name = mnt_wealth_invested_since_review target = scope:mnt_wealth_estate }
		}
		add = "variable_map(mnt_wealth_invested_since_review|scope:mnt_wealth_estate)"
	}
	save_temporary_value_as = mnt_wealth_invested_pool_at_review

	value = mnt_wealth_projected_months
	save_temporary_value_as = mnt_wealth_months
	# A negative rate spends nothing.
	value = mnt_wealth_investment_rate
	min = 0
	save_temporary_value_as = mnt_wealth_investment_rate
	value = mnt_wealth_projected_pool_remaining
	save_temporary_value_as = mnt_wealth_pool_remaining
	value = mnt_wealth_projected_pool_remaining_over_period
	save_temporary_value_as = mnt_wealth_pool_remaining_over_period

	# The estate's own income and wealth after income.
	value = 0
	if = {
		limit = { is_key_in_variable_map = { name = mnt_wealth_estate_wealth target = scope:mnt_wealth_estate } }
		add = "variable_map(mnt_wealth_estate_wealth|scope:mnt_wealth_estate)"
	}
	save_temporary_value_as = mnt_wealth_wealth_at_review
	value = mnt_wealth_projected_estate_income
	save_temporary_value_as = mnt_wealth_estate_income
	value = mnt_wealth_displayed_estate_maintenance
	save_temporary_value_as = mnt_wealth_estate_maintenance

	# The share of invested wealth left after a month's decay, and after the
	# period. Decay is toward zero for either sign.
	value = 1
	subtract = {
		value = mnt_wealth_estate_decay_rate
		multiply = mnt_wealth_decay_resistance_multiplier
	}
	save_temporary_value_as = mnt_wealth_invested_pool_remaining
	value = scope:mnt_wealth_invested_pool_remaining
	pow = scope:mnt_wealth_months
	save_temporary_value_as = mnt_wealth_invested_pool_remaining_over_period

	# Sum over the period of the flat part of each month's investment.
	value = 1
	subtract = scope:mnt_wealth_invested_pool_remaining
	save_temporary_value_as = mnt_wealth_invested_pool_change
	if = {
		limit = { NOT = { scope:mnt_wealth_invested_pool_change = 0 } }
		value = 1
		subtract = scope:mnt_wealth_invested_pool_remaining_over_period
		divide = scope:mnt_wealth_invested_pool_change
	}
	else = {
		value = scope:mnt_wealth_months
	}
	save_temporary_value_as = mnt_wealth_flat_inflow_sum

	# Sum over the period of the sloped part.
	value = scope:mnt_wealth_invested_pool_remaining
	subtract = scope:mnt_wealth_pool_remaining
	save_temporary_value_as = mnt_wealth_remaining_gap
	if = {
		limit = { NOT = { scope:mnt_wealth_remaining_gap = 0 } }
		value = scope:mnt_wealth_invested_pool_remaining_over_period
		subtract = scope:mnt_wealth_pool_remaining_over_period
		divide = scope:mnt_wealth_remaining_gap
	}
	else_if = {
		limit = { NOT = { scope:mnt_wealth_pool_remaining = 0 } }
		value = scope:mnt_wealth_months
		multiply = scope:mnt_wealth_pool_remaining_over_period
		divide = scope:mnt_wealth_pool_remaining
	}
	else = {
		value = 0
	}
	save_temporary_value_as = mnt_wealth_sloped_inflow_sum

	value = scope:mnt_wealth_investment_rate
	multiply = scope:mnt_wealth_wealth_at_review
	subtract = scope:mnt_wealth_estate_income
	save_temporary_value_as = mnt_wealth_inflow_slope

	value = scope:mnt_wealth_invested_pool_at_review
	multiply = scope:mnt_wealth_invested_pool_remaining_over_period
	add = {
		value = scope:mnt_wealth_estate_income
		multiply = scope:mnt_wealth_flat_inflow_sum
		add = {
			value = scope:mnt_wealth_inflow_slope
			multiply = scope:mnt_wealth_sloped_inflow_sum
		}
		multiply = mnt_wealth_investment_efficiency_multiplier
	}
	subtract = {
		value = scope:mnt_wealth_estate_maintenance
		multiply = scope:mnt_wealth_flat_inflow_sum
	}
}

# The investment level from the projected invested wealth.

mnt_wealth_projected_investment_level = {
	value = mnt_wealth_projected_invested_wealth
	save_temporary_value_as = mnt_wealth_level_invested_pool
	value = 1
	if = { limit = { scope:mnt_wealth_level_invested_pool < 0 } value = -1 }
	save_temporary_value_as = mnt_wealth_level_sign
	value = scope:mnt_wealth_level_invested_pool
	multiply = scope:mnt_wealth_level_sign
	save_temporary_value_as = mnt_wealth_level_invested_magnitude
	# Below the first level the whole level is 0 and the root is skipped: the
	# engine's pow returns 1 for a zero base.
	value = scope:mnt_wealth_level_invested_magnitude
	divide = mnt_wealth_investment_level_cost_base
	save_temporary_value_as = mnt_wealth_level_ratio
	value = 0
	if = {
		limit = { scope:mnt_wealth_level_ratio >= 1 }
		value = scope:mnt_wealth_level_ratio
		pow = 0.5
		floor = yes
	}
	save_temporary_value_as = mnt_wealth_level_whole
	value = scope:mnt_wealth_level_whole
	multiply = scope:mnt_wealth_level_whole
	multiply = mnt_wealth_investment_level_cost_base
	save_temporary_value_as = mnt_wealth_level_reached_cost
	value = scope:mnt_wealth_level_whole
	add = 1
	save_temporary_value_as = mnt_wealth_level_next_whole
	value = scope:mnt_wealth_level_next_whole
	multiply = scope:mnt_wealth_level_next_whole
	multiply = mnt_wealth_investment_level_cost_base
	subtract = scope:mnt_wealth_level_reached_cost
	save_temporary_value_as = mnt_wealth_level_step_cost
	value = scope:mnt_wealth_level_invested_magnitude
	subtract = scope:mnt_wealth_level_reached_cost
	divide = scope:mnt_wealth_level_step_cost
	add = scope:mnt_wealth_level_whole
	multiply = scope:mnt_wealth_level_sign
}

# --- aggregate display values ---

# Net monthly wealth change: wages paid in, plus the net estate flow (positive
# when estates invest, negative when they seize), minus what is invested.
# The invested term floors the rate at zero: a negative rate spends nothing,
# it never pays the pool.

mnt_wealth_local_wealth_change = {
	value = mnt_wealth_local_wage_pool
	add = mnt_wealth_estate_flow_total
	subtract = {
		value = mnt_wealth_local_wealth_total
		multiply = { value = mnt_wealth_investment_rate min = 0 }
	}
}

# Monthly fraction of the yearly wealth-share drift, for tooltip display.

mnt_wealth_local_wealth_share_monthly_trend_rate = {
	value = mnt_wealth_local_wealth_share_trend_rate
	divide = 12
}

# --- estate savings (estate scope) ---

# A newly formed country has no economic base yet, so the goal is held at a
# gold minimum. The ratios measured against it below then always have something
# to divide by.

mnt_wealth_estate_savings_goal = {
	value = 0
	owner = { add = { value = country_economical_base multiply = "estate_power(prev.estate_type)" } }
	min = 1
}

mnt_wealth_estate_savings_ratio = {
	value = estate_gold
	divide = mnt_wealth_estate_savings_goal
}

# --- estate seizure demand (estate scope) ---

# The demand the estate has built up over every month its treasury has sat under
# the savings goal.

mnt_wealth_estate_seizure_demand_level = {
	value = 0

	owner = {
		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_seizure_demand target = prev.estate_type } }
			add = "variable_map(mnt_wealth_estate_seizure_demand|prev.estate_type)"
		}
	}
}

# What the built-up demand multiplies this month's seizure by.

mnt_wealth_estate_seizure_multiplier = {
	value = 1
	add = mnt_wealth_estate_seizure_demand_level
}

# The share of the goal the estate can seize this month, demand included.

mnt_wealth_estate_seizure_cap = {
	value = mnt_wealth_estate_seizure_limit_factor
	add = mnt_wealth_estate_seizure_demand_level
	multiply = mnt_wealth_estate_savings_goal
}

# The most any estate can pay into its locations in a month. A share of the
# whole economic base, so it does not shrink with the estate's power.

mnt_wealth_estate_payout_cap = {
	value = 0
	owner = { add = country_economical_base }
	multiply = mnt_wealth_national_investment_limit_factor
}

# How much of this month's shortfall is being added to the demand: the gap
# between the effective goal and the treasury, as a share of the savings goal.
# The effective goal adds last month's seizure so the demand keeps growing until
# organic income alone would cover the goal.

mnt_wealth_estate_seizure_demand_growth = {
	value = mnt_wealth_estate_savings_goal
	subtract = estate_gold

	owner = {
		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_flow target = prev.estate_type } }
			subtract = { value = "variable_map(mnt_wealth_estate_flow|prev.estate_type)" min = 0 }
		}
	}

	min = 0
	divide = mnt_wealth_estate_savings_goal
}
