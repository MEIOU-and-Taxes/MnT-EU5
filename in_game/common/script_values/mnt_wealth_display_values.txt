# Script values the interface reads: the location window, its tooltips, the map
# modes and the map mode tooltip strings. Nothing here is read by the
# simulation; those values live in mnt_wealth_values.txt.

# --- the display month ---

# How many months have passed since this location's last yearly review.
# AI locations do a whole year of work in one tick, so the display projects
# forward from the review state using this count.

mnt_wealth_months_since_review = {
	value = mnt_wealth_months_since_campaign_start
	subtract = var:mnt_wealth_review_month
}

# --- pool and health ---

# For AI locations, projects the wealth pool forward from the review state
# using the same geometric series the yearly shortcut will eventually apply.
# For human locations, just reads the pool directly.

mnt_wealth_displayed_wealth_pool = {
	value = var:mnt_wealth_pool

	if = {
		limit = {
			exists = owner
			owner = { is_ai = yes }
			var:mnt_wealth_local_wealth_share_sum > 0
			mnt_wealth_investment_rate > 0
		}
		value = mnt_wealth_investment_rate
		multiply = var:mnt_wealth_local_wealth_share_sum
		save_temporary_value_as = mnt_wealth_display_spend_rate
		value = 1
		subtract = scope:mnt_wealth_display_spend_rate
		save_temporary_value_as = mnt_wealth_display_decay
		value = scope:mnt_wealth_display_decay
		pow = mnt_wealth_months_since_review
		save_temporary_value_as = mnt_wealth_display_decay_n
		# Monthly inflow: wages kept + national estate investment returned to the pool.
		value = var:mnt_wealth_kept_income_total
		add = mnt_wealth_estate_flow_total
		save_temporary_value_as = mnt_wealth_display_income
		value = var:mnt_wealth_pool
		add = scope:mnt_wealth_display_income
		save_temporary_value_as = mnt_wealth_display_pool_plus_income
		value = scope:mnt_wealth_display_pool_plus_income
		multiply = scope:mnt_wealth_display_decay_n

		add = {
			value = scope:mnt_wealth_display_income
			multiply = { value = scope:mnt_wealth_display_decay subtract = scope:mnt_wealth_display_decay_n }
			divide = scope:mnt_wealth_display_spend_rate
		}
	}

	min = 0
}

mnt_wealth_local_wealth_total = {
	value = mnt_wealth_displayed_wealth_pool
}

mnt_wealth_baseline = {
	value = var:mnt_wealth_local_gdp
	multiply = mnt_wealth_starting_gdp_months
}

# Zero when there is no desired wealth to measure against, which means no estate
# is present here or the location has not been surveyed yet.

mnt_wealth_health_ratio = {
	value = 0

	if = {
		limit = { mnt_wealth_baseline > 0 }
		value = mnt_wealth_local_wealth_total
		divide = mnt_wealth_baseline
	}
}

mnt_wealth_band = {
	value = 0
	if = { limit = { mnt_wealth_health_ratio >= 0.5 } add = 1 }
	if = { limit = { mnt_wealth_health_ratio >= 0.9 } add = 1 }
	if = { limit = { mnt_wealth_health_ratio >= 1.1 } add = 1 }
}

# --- flow map mode ratio ---

# Net estate capital flow / monthly investment. Positive = investing, negative = seizing.
# flow_ratio = estate_investment_total / (wealth * investment_rate).
# Zero where there is no monthly investment to measure the flow against.

mnt_wealth_estate_flow_ratio = {
	value = 0

	if = {
		limit = { mnt_wealth_local_wealth_total > 0 mnt_wealth_investment_rate > 0 }
		value = mnt_wealth_estate_flow_total
		divide = { value = mnt_wealth_local_wealth_total multiply = mnt_wealth_investment_rate }
	}
}

# --- investment equilibrium ---

# The level an estate settles at is where its monthly investment exactly pays its
# monthly upkeep. Working that out means undoing the desired wealth curve, which
# is a square root that cannot be computed in the data binding language.
#
# Every value in this section runs in location scope and reads scope:mnt_wealth_estate,
# the estate type the tooltip row is showing.

# Entry point for the location window. The interface hands in the location and
# the estate as named scopes because the tooltip row knows both.

mnt_wealth_displayed_investment_equilibrium = {
	value = 0
	scope:mnt_wealth_location = { add = mnt_wealth_investment_equilibrium_level }
}

# Entry point for the upkeep macro: one estate's monthly decay upkeep, before
# the upkeep efficiency the macro applies. Location and estate come in as scopes.

mnt_wealth_displayed_monthly_upkeep = {
	value = 0
	scope:mnt_wealth_location = { add = mnt_wealth_monthly_estate_upkeep }
}

# Entry point for the decay-rate macro: one estate's live monthly decay rate.

mnt_wealth_displayed_estate_decay_rate = {
	value = 0
	scope:mnt_wealth_location = { add = mnt_wealth_estate_decay_rate }
}

# The estate's share of the location's wealth pool.

mnt_wealth_equilibrium_estate_wealth = {
	value = 0

	if = {
		limit = {
			has_variable_map = mnt_wealth_local_wealth_share
			is_key_in_variable_map = { name = mnt_wealth_local_wealth_share target = scope:mnt_wealth_estate }
		}
		value = mnt_wealth_local_wealth_total
		multiply = "variable_map(mnt_wealth_local_wealth_share|scope:mnt_wealth_estate)"
	}
}

# This location's slice of the estate's national flow: the national flow split
# by how much of the estate's raw power sits here. Positive when the estate is
# investing gold into the location, negative when it is seizing gold out.

mnt_wealth_equilibrium_estate_flow = {
	value = 0
	save_temporary_scope_as = mnt_wealth_equilibrium_location

	if = {
		limit = {
			has_variable_map = mnt_estate_raw_power
			is_key_in_variable_map = { name = mnt_estate_raw_power target = scope:mnt_wealth_estate }

			owner ?= {
				has_variable_map = mnt_wealth_estate_flow
				has_variable_map = mnt_wealth_national_estate_raw_power
				is_key_in_variable_map = { name = mnt_wealth_estate_flow               target = scope:mnt_wealth_estate }
				is_key_in_variable_map = { name = mnt_wealth_national_estate_raw_power target = scope:mnt_wealth_estate }
				"variable_map(mnt_wealth_national_estate_raw_power|scope:mnt_wealth_estate)" > 0
			}
		}
		owner = {
			add = {
				value = "variable_map(mnt_wealth_estate_flow|scope:mnt_wealth_estate)"
				multiply = "scope:mnt_wealth_equilibrium_location.variable_map(mnt_estate_raw_power|scope:mnt_wealth_estate)"
				divide = "variable_map(mnt_wealth_national_estate_raw_power|scope:mnt_wealth_estate)"
			}
		}
	}
}

# Gold a national seizure tears out of invested wealth this month: the part of a
# negative flow that the estate's liquid wealth cannot cover. Zero while the
# estate is investing or while its liquid wealth covers the whole seizure.

mnt_wealth_equilibrium_seizure_from_invested = {
	value = 0
	subtract = mnt_wealth_equilibrium_estate_flow
	subtract = mnt_wealth_equilibrium_estate_wealth
	min = 0
}

# The wealth the equilibrium is measured from: estate wealth scaled by investment
# efficiency, minus any seized invested wealth converted back to its wealth cost,
# all divided by upkeep efficiency.

mnt_wealth_equilibrium_wealth = {
	value = mnt_wealth_equilibrium_estate_wealth
	multiply = mnt_wealth_investment_efficiency_multiplier

	if = {
		limit = { mnt_wealth_investment_rate > 0 }
		subtract = { value = mnt_wealth_equilibrium_seizure_from_invested divide = mnt_wealth_investment_rate }
	}

	divide = mnt_wealth_upkeep_efficiency_multiplier
}

# The per-estate monthly decay rate: local plus national modifier. Resolved by
# estate type because the modifier name carries it. Runs in location scope with
# scope:mnt_wealth_estate set, and reads the modifiers live.

mnt_wealth_estate_decay_rate = {
	value = 0
	if = {
		limit = { scope:mnt_wealth_estate = estate_type:nobles_estate }
		value = modifier:mnt_wealth_local_nobles_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_nobles_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:clergy_estate }
		value = modifier:mnt_wealth_local_clergy_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_clergy_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:burghers_estate }
		value = modifier:mnt_wealth_local_burghers_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_burghers_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:peasants_estate }
		value = modifier:mnt_wealth_local_peasants_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_peasants_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:dhimmi_estate }
		value = modifier:mnt_wealth_local_dhimmi_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_dhimmi_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:tribes_estate }
		value = modifier:mnt_wealth_local_tribes_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_tribes_investment_decay }
	}
	else_if = {
		limit = { scope:mnt_wealth_estate = estate_type:cossacks_estate }
		value = modifier:mnt_wealth_local_cossacks_investment_decay
		if = { limit = { exists = owner } add = owner.modifier:mnt_wealth_global_cossacks_investment_decay }
	}
}

# The level an estate settles at: where monthly investment spending exactly pays
# the monthly decay. Invested wealth settles at spending / decay, and the level
# is the unchanged square-root curve over that pool.

mnt_wealth_investment_equilibrium_level = {
	value = 0

	if = {
		limit = {
			mnt_wealth_equilibrium_wealth > 0
			mnt_wealth_estate_decay_rate > 0
		}
		value = mnt_wealth_equilibrium_wealth
		multiply = mnt_wealth_investment_rate
		divide = mnt_wealth_estate_decay_rate
		divide = mnt_wealth_investment_level_cost_base
		pow = 0.5
		min = 0
	}
}

# --- projected investment level ---

# mnt_wealth_set_ai_yearly_investment settles a whole year in one tick: it
# leaves var:mnt_wealth_pool holding the pool twelve months from the review and
# mnt_wealth_invested_wealth holding a year of credit. Both therefore run ahead
# of the calendar, which is what made the window read thirteen months of
# spending on the first month of a campaign.
#
# The pass is a geometric series, so it inverts. These values read the stored
# state back to the review, replay only the months that have actually elapsed,
# and write nothing.

mnt_wealth_review_spend_rate = {
	value = mnt_wealth_investment_rate
	multiply = var:mnt_wealth_local_wealth_share_sum
}

mnt_wealth_review_decay = {
	value = 1
	subtract = mnt_wealth_review_spend_rate
}

mnt_wealth_projected_effective_months = {
	value = 0

	if = {
		limit = { mnt_wealth_review_spend_rate > 0 }
		value = mnt_wealth_review_decay
		pow = mnt_wealth_projected_months
		save_temporary_value_as = mnt_wealth_review_decay_n
		value = 1
		subtract = scope:mnt_wealth_review_decay_n
		divide = mnt_wealth_review_spend_rate
	}
}

# Replays mnt_wealth_set_ai_yearly_investment for the months elapsed so far.
# Runs in location scope and reads scope:mnt_wealth_estate.

mnt_wealth_projected_months = {
	value = mnt_wealth_months_since_review
	min = 0
	max = 12
}

mnt_wealth_projected_stored_level = {
	value = 0

	if = {
		limit = {
			has_variable_map = mnt_wealth_investment_level
			is_key_in_variable_map = { name = mnt_wealth_investment_level target = scope:mnt_wealth_estate }
		}
		value = "variable_map(mnt_wealth_investment_level|scope:mnt_wealth_estate)"
	}
}

# Income is the kept wage total alone, matching the pass; estate flow arrives
# separately through the national payout.

mnt_wealth_projected_investment_paid = {
	value = 0

	if = {
		limit = {
			mnt_wealth_projected_months > 0
			var:mnt_wealth_local_wealth_share_sum > 0
			mnt_wealth_investment_rate > 0
		}
		value = mnt_wealth_projected_effective_months
		save_temporary_value_as = mnt_wealth_proj_effective_months
		value = var:mnt_wealth_pool
		add = var:mnt_wealth_kept_income_total
		multiply = scope:mnt_wealth_proj_effective_months

		add = {
			value = var:mnt_wealth_kept_income_total
			multiply = { value = mnt_wealth_projected_months subtract = scope:mnt_wealth_proj_effective_months }
			divide = mnt_wealth_review_spend_rate
		}

		multiply = mnt_wealth_investment_rate
	}
}

# The invested pool decayed over the elapsed months, matching
# mnt_wealth_compute_estate_upkeep with MONTHS = projected months.

mnt_wealth_projected_estate_upkeep = {
	value = 0

	if = {
		limit = {
			has_variable_map = mnt_wealth_invested_wealth
			is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = scope:mnt_wealth_estate }
			"variable_map(mnt_wealth_invested_wealth|scope:mnt_wealth_estate)" > 0
		}
		value = "variable_map(mnt_wealth_invested_wealth|scope:mnt_wealth_estate)"
		save_temporary_value_as = mnt_wealth_proj_invested
		multiply = mnt_wealth_estate_decay_rate
		multiply = mnt_wealth_projected_months
		max = scope:mnt_wealth_proj_invested
	}
}

# An estate with no ownership share still owes upkeep, as the pass has it.

mnt_wealth_projected_estate_progress = {
	value = 0

	if = {
		limit = {
			has_variable_map = mnt_wealth_local_wealth_share
			is_key_in_variable_map = { name = mnt_wealth_local_wealth_share target = scope:mnt_wealth_estate }
		}
		value = mnt_wealth_projected_investment_paid
		multiply = "variable_map(mnt_wealth_local_wealth_share|scope:mnt_wealth_estate)"
	}

	multiply = mnt_wealth_investment_efficiency_multiplier
	subtract = { value = mnt_wealth_projected_estate_upkeep multiply = mnt_wealth_upkeep_efficiency_multiplier }
}

# One month of this estate's progress at the location's current state: the same
# figure the window prints as the monthly change. Everything below is expressed
# in multiples of it, so the accumulated number and the monthly number can never
# disagree about how much a month is worth.

mnt_wealth_monthly_estate_upkeep = {
	value = 0

	if = {
		limit = {
			has_variable_map = mnt_wealth_invested_wealth
			is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = scope:mnt_wealth_estate }
			"variable_map(mnt_wealth_invested_wealth|scope:mnt_wealth_estate)" > 0
		}
		value = "variable_map(mnt_wealth_invested_wealth|scope:mnt_wealth_estate)"
		save_temporary_value_as = mnt_wealth_month_invested
		multiply = mnt_wealth_estate_decay_rate
		max = scope:mnt_wealth_month_invested
	}
}

mnt_wealth_monthly_estate_progress = {
	value = 0

	if = {
		limit = {
			has_variable_map = mnt_wealth_local_wealth_share
			is_key_in_variable_map = { name = mnt_wealth_local_wealth_share target = scope:mnt_wealth_estate }
		}
		value = mnt_wealth_displayed_wealth_pool
		multiply = "variable_map(mnt_wealth_local_wealth_share|scope:mnt_wealth_estate)"
		multiply = mnt_wealth_investment_rate
	}

	multiply = mnt_wealth_investment_efficiency_multiplier
	subtract = { value = mnt_wealth_monthly_estate_upkeep multiply = mnt_wealth_upkeep_efficiency_multiplier }
}

# Invested wealth as of this month. AI locations settle yearly, so the stored
# figure is the value at the last review; it is projected forward over the
# months elapsed as a geometric series, the invested pool decaying at the
# estate's rate while it takes in the monthly investment, mirroring the
# wealth-pool projection. A location the pass has never reached projects from
# its estate progress instead.

mnt_wealth_projected_invested_wealth = {
	value = 0

	if = {
		limit = {
			has_variable_map = mnt_wealth_invested_wealth
			is_key_in_variable_map = { name = mnt_wealth_invested_wealth target = scope:mnt_wealth_estate }
		}
		value = "variable_map(mnt_wealth_invested_wealth|scope:mnt_wealth_estate)"

		if = {
			limit = {
				exists = owner
				owner = { is_ai = yes }
				mnt_wealth_estate_decay_rate > 0
			}
			save_temporary_value_as = mnt_wealth_proj_invested_0
			# Monthly drain on the invested pool, and its survival over the months
			# elapsed since the review.
			value = mnt_wealth_estate_decay_rate
			multiply = mnt_wealth_upkeep_efficiency_multiplier
			save_temporary_value_as = mnt_wealth_proj_drain
			value = 1
			subtract = scope:mnt_wealth_proj_drain
			pow = mnt_wealth_projected_months
			save_temporary_value_as = mnt_wealth_proj_survival_n
			# The monthly investment paid in at the review, efficiency-scaled.
			value = var:mnt_wealth_pool
			multiply = "variable_map(mnt_wealth_local_wealth_share|scope:mnt_wealth_estate)"
			multiply = mnt_wealth_investment_rate
			multiply = mnt_wealth_investment_efficiency_multiplier
			save_temporary_value_as = mnt_wealth_proj_inflow
			# invested(n) = invested_0 * survival^n + inflow * (1 - survival^n) / drain
			value = scope:mnt_wealth_proj_invested_0
			multiply = scope:mnt_wealth_proj_survival_n
			add = {
				value = 1
				subtract = scope:mnt_wealth_proj_survival_n
				divide = scope:mnt_wealth_proj_drain
				multiply = scope:mnt_wealth_proj_inflow
			}
		}
	}
	else = {
		value = mnt_wealth_projected_estate_progress
	}

	# Progress the monthly loop has banked but the yearly review has not settled.
	if = {
		limit = {
			has_variable_map = mnt_wealth_invested_since_review
			is_key_in_variable_map = { name = mnt_wealth_invested_since_review target = scope:mnt_wealth_estate }
		}
		add = "variable_map(mnt_wealth_invested_since_review|scope:mnt_wealth_estate)"
	}
}

# Human locations tick monthly, so their stored level is already current.

mnt_wealth_projected_investment_level = {
	value = mnt_wealth_projected_stored_level

	if = {
		limit = {
			exists = owner
			owner = { is_ai = yes }
			var:mnt_wealth_local_wealth_share_sum > 0
			mnt_wealth_investment_rate > 0
			has_variable_map = mnt_wealth_investment_level
			is_key_in_variable_map = { name = mnt_wealth_investment_level target = scope:mnt_wealth_estate }
		}
		value = mnt_wealth_projected_invested_wealth
		save_temporary_value_as = mnt_wealth_proj_signed
		value = 1
		if = { limit = { scope:mnt_wealth_proj_signed < 0 } value = -1 }
		save_temporary_value_as = mnt_wealth_proj_sign
		value = scope:mnt_wealth_proj_signed
		multiply = scope:mnt_wealth_proj_sign
		save_temporary_value_as = mnt_wealth_proj_magnitude
		value = scope:mnt_wealth_proj_magnitude
		divide = mnt_wealth_investment_level_cost_base
		pow = 0.5
		floor = yes
		min = 0
		save_temporary_value_as = mnt_wealth_proj_whole
		value = scope:mnt_wealth_proj_whole
		multiply = scope:mnt_wealth_proj_whole
		multiply = mnt_wealth_investment_level_cost_base
		save_temporary_value_as = mnt_wealth_proj_reached
		value = scope:mnt_wealth_proj_whole
		add = 1
		save_temporary_value_as = mnt_wealth_proj_next_whole
		value = scope:mnt_wealth_proj_next_whole
		multiply = scope:mnt_wealth_proj_next_whole
		multiply = mnt_wealth_investment_level_cost_base
		subtract = scope:mnt_wealth_proj_reached
		save_temporary_value_as = mnt_wealth_proj_level_cost
		value = scope:mnt_wealth_proj_magnitude
		subtract = scope:mnt_wealth_proj_reached
		divide = scope:mnt_wealth_proj_level_cost
		add = scope:mnt_wealth_proj_whole
		multiply = scope:mnt_wealth_proj_sign
	}
}

# Entry point for the interface; location and estate come in as named scopes.

mnt_wealth_displayed_investment_level = {
	value = 0
	scope:mnt_wealth_location = { add = mnt_wealth_projected_investment_level }
}

# --- aggregate display values ---

# Net monthly wealth change: wages paid in, plus the net estate flow (positive
# when estates invest, negative when they seize), minus what is invested.

mnt_wealth_local_wealth_change = {
	value = mnt_wealth_local_wage_pool
	add = mnt_wealth_estate_flow_total
	subtract = { value = mnt_wealth_local_wealth_total multiply = mnt_wealth_investment_rate }
}

# Monthly fraction of the yearly wealth-share drift, for tooltip display.

mnt_wealth_local_wealth_share_monthly_trend_rate = {
	value = mnt_wealth_local_wealth_share_trend_rate
	divide = 12
}

# --- estate savings (estate scope) ---

# A newly formed country has no economic base yet, so the goal is held at a
# gold minimum. The ratios measured against it below then always have something
# to divide by.

mnt_wealth_estate_savings_goal = {
	value = 0
	owner = { add = { value = country_economical_base multiply = "estate_power(prev.estate_type)" } }
	min = 1
}

mnt_wealth_estate_savings_ratio = {
	value = estate_gold
	divide = mnt_wealth_estate_savings_goal
}

# --- estate seizure demand (estate scope) ---

# The demand the estate has built up over every month its treasury has sat under
# the savings goal.

mnt_wealth_estate_seizure_demand_level = {
	value = 0

	owner = {
		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_seizure_demand target = prev.estate_type } }
			add = "variable_map(mnt_wealth_estate_seizure_demand|prev.estate_type)"
		}
	}
}

# What the built-up demand multiplies this month's seizure by.

mnt_wealth_estate_seizure_multiplier = {
	value = 1
	add = mnt_wealth_estate_seizure_demand_level
}

# The share of the goal the estate can seize this month, demand included.

mnt_wealth_estate_seizure_cap = {
	value = mnt_wealth_estate_seizure_limit_factor
	add = mnt_wealth_estate_seizure_demand_level
	multiply = mnt_wealth_estate_savings_goal
}

# The most any estate can pay into its locations in a month. A share of the
# whole economic base, so it does not shrink with the estate's power.

mnt_wealth_estate_payout_cap = {
	value = 0
	owner = { add = country_economical_base }
	multiply = mnt_wealth_national_investment_limit_factor
}

# How much of this month's shortfall is being added to the demand: the gap
# between the effective goal and the treasury, as a share of the savings goal.
# The effective goal adds last month's seizure so the demand keeps growing until
# organic income alone would cover the goal.

mnt_wealth_estate_seizure_demand_growth = {
	value = mnt_wealth_estate_savings_goal
	subtract = estate_gold

	owner = {
		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_flow target = prev.estate_type } }
			subtract = { value = "variable_map(mnt_wealth_estate_flow|prev.estate_type)" min = 0 }
		}
	}

	min = 0
	divide = mnt_wealth_estate_savings_goal
}

# --- estate capital flow (estate scope) ---

# One estate's net monthly flow with its locations: positive when it invests
# gold into them, negative when it seizes gold out to cover a shortfall.

mnt_wealth_estate_location_investment = {
	value = 0

	owner = {
		if = {
			limit = { is_key_in_variable_map = { name = mnt_wealth_estate_flow target = prev.estate_type } }
			add = "variable_map(mnt_wealth_estate_flow|prev.estate_type)"
		}
	}
}
