mnt_credit_loan_demand_multiplier = {
	if = {
		limit = { has_advance = clearing_house_advance }
		add = {
			desc = "MNT_CREDIT_LOAN_DEMAND_CLEARING_HOUSE"
			value = 0.00033
		}
	}
	else_if = {
		limit = { has_advance = stock_exchange_advance }
		add = {
			desc = "MNT_CREDIT_LOAN_DEMAND_STOCK_EXCHANGE"
			value = 0.0005
		}
	}
	else_if = {
		limit = { has_advance = global_trade_advance }
		add = {
			desc = "MNT_CREDIT_LOAN_DEMAND_GLOBAL_TRADE"
			value = 0.00083
		}
	}
	else_if = {
		limit = { has_advance = basic_financial_instruments }
		add = {
			desc = "MNT_CREDIT_LOAN_DEMAND_BASIC_FINANCE"
			value = 0.00125
		}
	}
	else_if = {
		limit = { has_advance = banking_advance }
		add = {
			desc = "MNT_CREDIT_LOAN_DEMAND_BANKING"
			value = 0.0025
		}
	}
	else_if = {
		limit = { always = yes }
		add = {
			desc = "MNT_CREDIT_LOAN_DEMAND_EARLY_FINANCE"
			value = 0.005
		}
	}
}

# Country scope: replace only the cached loan component with the live ledger.
mnt_credit_lending_demand = {
	if = {
		limit = { capital.market.location = { has_variable = mnt_credit_quote_other_demand } }
		value = "capital.market.location.var:mnt_credit_quote_other_demand"
		add = "capital.market.location.var:mnt_credit_loan_final_scale"
	}
	else = {
		value = "capital.market.goods_demand_in_market(goods:credit)"
	}
}

# Credit has base price 1. The one-unit buffer keeps empty markets finite.
mnt_credit_lending_quote = {
	value = mnt_credit_lending_demand
	add = 1
	divide = {
		value = "capital.market.goods_supply_in_market(goods:credit)"
		add = 1
	}
	min = 0.01
}
