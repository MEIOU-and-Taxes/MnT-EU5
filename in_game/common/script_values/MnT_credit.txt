MnT_credit_output_from_inflation = {
	value = inflation
	multiply = 5
}

# Location scope. Pop counts are already expressed in thousands.
mnt_credit_burgher_trade_rate = {
	value = 1
	add = modifier:local_trades_per_burgher
	if = {
		limit = { exists = owner }
		add = owner.modifier:global_trades_per_burgher
	}
}

# Monthly Credit per unit of debt: 1 per 1,000, falling to 0.5 with advances.
mnt_credit_loan_demand_multiplier = {
	if = { limit = { has_advance = clearing_house_advance }
		value = 0.0005
	}
	else_if = { limit = { has_advance = stock_exchange_advance }
		value = 0.0006
	}
	else_if = { limit = { has_advance = global_trade_advance }
		value = 0.0007
	}
	else_if = { limit = { has_advance = basic_financial_instruments }
		value = 0.0008
	}
	else_if = { limit = { has_advance = banking_advance }
		value = 0.0009
	}
	else = {
		value = 0.001
	}
}

# Native supply, read only when capturing a new interest calculation.
mnt_credit_current_market_supply = {
    value = 0
    if = { limit = { exists = capital.market }
        value = "capital.market.goods_supply_in_market(goods:credit)"
    }
}

# Country scope: replace only the cached loan component with the live ledger.
mnt_credit_current_lending_demand = {
    if = { limit = {
            exists = capital.market
            exists = capital.market.location
            capital.market.location = {
                has_variable = mnt_credit_quote_other_demand
                has_variable = mnt_credit_loan_final_scale
            }
        }
        value = "capital.market.location.var:mnt_credit_quote_other_demand"
        add = "capital.market.location.var:mnt_credit_loan_final_scale"
    }
    else_if = { limit = { exists = capital.market }
        value = "capital.market.goods_demand_in_market(goods:credit)"
    }
    else = { value = 0 }
}

# Market contribution to annual interest. Above quote 1, a 10% baseline
# approaches 50%; quote 2 gives 12.5%. Surplus discounts approach -9 percentage points (1% baseline).
mnt_credit_market_interest_adjustment = {
	if = { limit = { mnt_credit_lending_quote > 1 }
		value = mnt_credit_lending_quote
		subtract = 1
		divide = { value = mnt_credit_lending_quote add = 14 }
		multiply = 0.40
		max = 0.40
	}
	else = {
		value = mnt_credit_lending_quote
		subtract = 1
		multiply = 0.09
	}
}

# Credit has base price 1. The one-unit buffer keeps empty markets finite.
mnt_credit_lending_quote = {
	value = mnt_credit_lending_demand
	add = 1
	divide = { value = mnt_credit_tooltip_market_supply
		add = 1
	}
	min = 0.01
}

# All interest tooltip inputs come from the calculation that was actually applied.
# Fallbacks keep old saves readable until their first refresh.
mnt_credit_lending_demand = {
    if = { limit = { has_variable = mnt_credit_rate_demand }
        value = var:mnt_credit_rate_demand
    }
    else = { value = mnt_credit_current_lending_demand }
}

mnt_credit_tooltip_market_supply = {
    if = { limit = { has_variable = mnt_credit_rate_supply }
        value = var:mnt_credit_rate_supply
    }
    else = { value = mnt_credit_current_market_supply }
}

mnt_credit_tooltip_market_balance = {
    value = mnt_credit_tooltip_market_supply
    subtract = mnt_credit_lending_demand
}

# Market scope; the goods price is separate from the inputs used for loan interest.
mnt_credit_tooltip_market_price = { value = "goods:credit.price_in_market(root)" }

# Use the documented comparison form; the numeric readback returned zero in-game.
mnt_credit_rate_matches_applied = {
    if = { limit = {
        country_modifier_strength = {
            modifier = mnt_credit_interest_market_price
            value = mnt_credit_market_interest_adjustment
        }
    }
        value = 1
    }
    else = { value = 0 }
}
