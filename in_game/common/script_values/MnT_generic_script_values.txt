# For when num_pop_type or num_pop_type_in_country are stale, such as on the same day after any effect 
# that changes pop counts. MnT_tribes and MnT_slaves on_game_start are good examples.
mnt_count_laborers = {
	every_pop = {
		limit = { pop_type = pop_type:laborers }
		add = pop_size
	}
}

mnt_count_burghers = {
	every_pop = {
		limit = { pop_type = pop_type:burghers }
		add = pop_size
	}
}

mnt_count_tribesmen = {
	every_pop = {
		limit = { pop_type = pop_type:tribesmen }
		add = pop_size
	}
}
