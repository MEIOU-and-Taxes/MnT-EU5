namespace = cost_of_the_court

# Recalculate demands from cost of the court
cost_of_the_court.1 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	trigger = {
		country_type = location
	}
	immediate = {
		capital.market.location = {
			# Initialize the variable if not exists
			if = {
				limit = {
					not = { has_variable = total_cost_of_the_court }
				}
				set_variable = {
					name = total_cost_of_the_court
					value = 0
				}
			}
			change_variable = {
				name = total_cost_of_the_court
				add = {
					# Base, court maintenance slider 0 to 1
					value = "root.court_maintenance"
					# Multiply by economic base
					multiply = "root.country_economical_base"
					#Mepper: Old value broke the economy, demand for 10% of slider
                    multiply = 0.1
					# Court Spending Effiency:
					# The more efficient your court spending, the less goods you demand.
					# Cap to always demand at least 1/3 of original demand, and max 3x original demand to prevent weirdness.
					divide = {
						value = 1
						add = "root.modifier:court_spending_efficiency"
						min = 0.33		# So effectively multiply demand by 3x worst-case
						max = 3			# So effectively divide demand by 3x best-case
					}

				}
			}
		}
	}
}
