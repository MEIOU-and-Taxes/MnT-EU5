
namespace = mnt_wealth_economy

# Fired one day after a location changes owner. New colonies are initialized;
# existing locations are advanced to now and resurveyed.

mnt_wealth_economy.30 = {
	type = location_event
	hidden = yes
	title = empty_text
	trigger = { has_owner = yes }
	immediate = {
		save_temporary_scope_as = mnt_wealth_transfer_loc
		if = {
			limit = {
				NOT = { is_key_in_global_variable_map = { name = mnt_wealth_location_last_service target = scope:mnt_wealth_transfer_loc } }
			}
			# New colony: initialize from scratch, no opening pool. The resurvey
			# counts the estates into the owner's national raw power.
			mnt_wealth_update_local_gdp = yes
			mnt_wealth_resurvey_estate_data = yes
			mnt_wealth_invalidate_estate_set = yes
			mnt_wealth_set_location_rates = yes
			mnt_wealth_set_review_month = yes
			# Seed the service cycle from the id, last service = now with no backdate.
			mnt_wealth_ensure_location_id = yes
			set_local_variable = { name = mnt_wealth_last_service value = global_var:mnt_wealth_months_elapsed }
			save_temporary_scope_as = mnt_wealth_seed_location
			add_to_global_variable_map = {
				name = mnt_wealth_location_last_service
				key = scope:mnt_wealth_seed_location
				value = local_var:mnt_wealth_last_service
			}
			set_local_variable = {
				name = mnt_wealth_seed_cycle
				value = { value = var:mnt_wealth_location_id modulo = 10 }
			}
			set_variable = { name = mnt_wealth_service_cycle value = local_var:mnt_wealth_seed_cycle }
			set_local_variable = { name = mnt_wealth_is_player value = 0 }
			mnt_wealth_write_location_next_service = yes
		}
		else = {
			# Existing land: advance to now and resurvey under the new owner.
			mnt_wealth_advance_location_to_now = yes
			mnt_wealth_update_investment_levels = yes
			mnt_wealth_resurvey_estate_data = yes
			mnt_wealth_invalidate_estate_set = yes
			mnt_wealth_require_location_service = yes
		}
	}
}

# Fired one day after a civil war annexation. The map reset at resolution can
# land after every hook and wipe the same-day rebuild, so the winner's raw
# power totals are recounted once the reset is over.

mnt_wealth_economy.31 = {
	type = country_event
	hidden = yes
	title = empty_text
	trigger = { num_locations > 0 }
	immediate = { mnt_wealth_rebuild_national_estate_raw_power = yes }
}

# Monthly: recalculates each estate's savings valve and national payout.

mnt_wealth_economy.10 = {
	type = country_event
	hidden = yes
	title = empty_text
	trigger = { num_locations > 0 is_ai = no }
	immediate = { mnt_wealth_set_national_savings_valves = { MONTHS = 1 } }
}

# Yearly: AI countries run the valve with twelve months of application.

mnt_wealth_economy.11 = {
	type = country_event
	hidden = yes
	title = empty_text
	trigger = { num_locations > 0 is_ai = yes }
	immediate = { mnt_wealth_set_national_savings_valves = { MONTHS = 12 } }
}
