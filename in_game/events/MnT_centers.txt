namespace = mnt_centers

# Yearly recalculation of centers (trade / production / culture / education)
mnt_centers.1 = {
	type = country_event
	title = hidden_event.t
	hidden = yes

	immediate = {
		# Reset the country-wide culture centre tier; re-raised as awards are handed out
		every_country = {
			set_variable = { name = mnt_center_culture_tier value = 0 }
		}

		# Clear previous center modifiers
		every_ownable_location = {
			limit = {
				# By checking the conditions here, later we can simply check for the existence of the variable
				has_owner = yes
			}
			remove_location_modifier = mnt_center_trade_local
			remove_location_modifier = mnt_center_trade_regional
			remove_location_modifier = mnt_center_trade_continental
			remove_location_modifier = mnt_center_trade_world

			remove_location_modifier = mnt_center_production_local
			remove_location_modifier = mnt_center_production_regional
			remove_location_modifier = mnt_center_production_continental
			remove_location_modifier = mnt_center_production_world

			remove_location_modifier = mnt_center_culture_local
			remove_location_modifier = mnt_center_culture_regional
			remove_location_modifier = mnt_center_culture_continental
			remove_location_modifier = mnt_center_culture_world

			remove_location_modifier = mnt_center_education_local
			remove_location_modifier = mnt_center_education_regional
			remove_location_modifier = mnt_center_education_continental
			remove_location_modifier = mnt_center_education_world

			# Production heuristic: urban manufactured goods output per head
			save_scope_as = target_location
			calc_urban_goods_value_produced_location = yes

			# Cache scores to avoid recalculating during location iteration
			calc_center_trade_score_location = yes
			calc_center_production_score_location = yes
			calc_center_culture_score_location = yes
			calc_center_education_score_location = yes
		}

		# Trade Centers
		remove_global_variable = world_center
		every_continent = {
			remove_global_variable = continent_center

			every_sub_continent_in_continent = {
				every_region_in_continent = {
					remove_global_variable = region_center

					every_area_in_region = {
						ordered_location_in_area = {
							limit = {
								has_variable = mnt_center_trade_score_location
								var:mnt_center_trade_score_location >= center_trade_local_threshold
							}
							order_by = { value = var:mnt_center_trade_score_location }
							max = 1
							check_range_bounds = no
							add_location_modifier = { modifier = mnt_center_trade_local days = -1 }
							if = {
								limit = {
									var:mnt_center_trade_score_location >= center_trade_regional_threshold
									OR = {
										NOT = { has_global_variable = region_center }
										var:mnt_center_trade_score_location > global_var:region_center.var:mnt_center_trade_score_location
									}
								}
								set_global_variable = {
									name = region_center
									value = this
								}
							}
						}
					}
					global_var:region_center ?= {
						remove_location_modifier = mnt_center_trade_local
						add_location_modifier = { modifier = mnt_center_trade_regional days = -1 }

						if = {
							limit = {
								var:mnt_center_trade_score_location >= center_trade_continental_threshold
								OR = {
									NOT = { has_global_variable = continent_center }
									var:mnt_center_trade_score_location > global_var:continent_center.var:mnt_center_trade_score_location
								}
							}
							set_global_variable = {
								name = continent_center
								value = this
							}
						}
					}
				}
			}
			
			global_var:continent_center ?= {
				remove_location_modifier = mnt_center_trade_regional
				add_location_modifier = { modifier = mnt_center_trade_continental days = -1 }

				if = {
					limit = {
						# World centers (late game, very scarce)
						current_age_or_later = { age = age_5_absolutism }
						var:mnt_center_trade_score_location >= center_trade_world_threshold
						OR = {
							NOT = { has_global_variable = world_center }
							var:mnt_center_trade_score_location > global_var:world_center.var:mnt_center_trade_score_location
						}
					}
					set_global_variable = {
						name = world_center
						value = this
					}
				}
			}
		}
		global_var:world_center ?= {
			remove_location_modifier = mnt_center_trade_continental
			add_location_modifier = { modifier = mnt_center_trade_world days = -1 }
		}

		# Production Centers
		remove_global_variable = world_center
		every_continent = {
			remove_global_variable = continent_center

			every_sub_continent_in_continent = {
				every_region_in_continent = {
					remove_global_variable = region_center

					every_area_in_region = {
						ordered_location_in_area = {
							limit = {
								has_variable = mnt_center_production_score_location
								var:mnt_center_production_score_location >= center_production_local_threshold
							}
							order_by = { value = var:mnt_center_production_score_location }
							max = 1
							check_range_bounds = no
							add_location_modifier = { modifier = mnt_center_production_local days = -1 }
							if = {
								limit = {
									var:mnt_center_production_score_location >= center_production_regional_threshold
									OR = {
										NOT = { has_global_variable = region_center }
										var:mnt_center_production_score_location > global_var:region_center.var:mnt_center_production_score_location
									}
								}
								set_global_variable = {
									name = region_center
									value = this
								}
							}
						}
					}
					global_var:region_center ?= {
						remove_location_modifier = mnt_center_production_local
						add_location_modifier = { modifier = mnt_center_production_regional days = -1 }

						if = {
							limit = {
								var:mnt_center_production_score_location >= center_production_continental_threshold
								OR = {
									NOT = { has_global_variable = continent_center }
									var:mnt_center_production_score_location > global_var:continent_center.var:mnt_center_production_score_location
								}
							}
							set_global_variable = {
								name = continent_center
								value = this
							}
						}
					}
				}
			}

			global_var:continent_center ?= {
				remove_location_modifier = mnt_center_production_regional
				add_location_modifier = { modifier = mnt_center_production_continental days = -1 }

				if = {
					limit = {
						# World centers (late game, very scarce)
						current_age_or_later = { age = age_5_absolutism }
						var:mnt_center_production_score_location >= center_production_world_threshold
						OR = {
							NOT = { has_global_variable = world_center }
							var:mnt_center_production_score_location > global_var:world_center.var:mnt_center_production_score_location
						}
					}
					set_global_variable = {
						name = world_center
						value = this
					}
				}
			}
		}
		global_var:world_center ?= {
			remove_location_modifier = mnt_center_production_continental
			add_location_modifier = { modifier = mnt_center_production_world days = -1 }
		}

		# Culture Centers
		remove_global_variable = world_center
		every_continent = {
			remove_global_variable = continent_center

			every_sub_continent_in_continent = {
				every_region_in_continent = {
					remove_global_variable = region_center

					every_area_in_region = {
						ordered_location_in_area = {
							limit = {
								has_variable = mnt_center_culture_score_location
								var:mnt_center_culture_score_location >= center_culture_local_threshold
							}
							order_by = { value = var:mnt_center_culture_score_location }
							max = 1
							check_range_bounds = no
							add_location_modifier = { modifier = mnt_center_culture_local days = -1 }
							owner ?= { set_variable = { name = mnt_center_culture_tier value = { value = var:mnt_center_culture_tier min = 1 } } }
							if = {
								limit = {
									var:mnt_center_culture_score_location >= center_culture_regional_threshold
									OR = {
										NOT = {has_global_variable = region_center}
										var:mnt_center_culture_score_location > global_var:region_center.var:mnt_center_culture_score_location
									}
								}
								set_global_variable = {
									name = region_center
									value = this
								}
							}
						}
					}
					global_var:region_center ?= {
						remove_location_modifier = mnt_center_culture_local
						add_location_modifier = { modifier = mnt_center_culture_regional days = -1 }
						owner ?= { set_variable = { name = mnt_center_culture_tier value = { value = var:mnt_center_culture_tier min = 2 } } }

						if = {
							limit = {
								var:mnt_center_culture_score_location >= center_culture_continental_threshold
								OR = {
									NOT = { has_global_variable = continent_center }
									var:mnt_center_culture_score_location > global_var:continent_center.var:mnt_center_culture_score_location
								}
							}
							set_global_variable = {
								name = continent_center
								value = this
							}
						}
					}
				}
			}

			global_var:continent_center ?= {
				remove_location_modifier = mnt_center_culture_regional
				add_location_modifier = { modifier = mnt_center_culture_continental days = -1 }
				owner ?= { set_variable = { name = mnt_center_culture_tier value = { value = var:mnt_center_culture_tier min = 3 } } }

				if = {
					limit = {
						# World centers (late game, very scarce)
						current_age_or_later = { age = age_5_absolutism }
						var:mnt_center_culture_score_location >= center_culture_world_threshold
						OR = {
							NOT = { has_global_variable = world_center }
							var:mnt_center_culture_score_location > global_var:world_center.var:mnt_center_culture_score_location
						}
					}
					set_global_variable = {
						name = world_center
						value = this
					}
				}
			}
		}
		global_var:world_center ?= {
			remove_location_modifier = mnt_center_culture_continental
			add_location_modifier = { modifier = mnt_center_culture_world days = -1 }
			owner ?= { set_variable = { name = mnt_center_culture_tier value = { value = var:mnt_center_culture_tier min = 4 } } }
		}

		# Education Centers
		remove_global_variable = world_center
		every_continent = {
			remove_global_variable = continent_center

			every_sub_continent_in_continent = {
				every_region_in_continent = {
					remove_global_variable = region_center

					every_area_in_region = {
						ordered_location_in_area = {
							limit = {
								has_variable = mnt_center_education_score_location
								var:mnt_center_education_score_location >= center_education_local_threshold
							}
							order_by = { value = var:mnt_center_education_score_location }
							max = 1
							check_range_bounds = no
							add_location_modifier = { modifier = mnt_center_education_local days = -1 }
							if = {
								limit = {
									var:mnt_center_education_score_location >= center_education_regional_threshold
									OR = {
										NOT = {has_global_variable = region_center}
										var:mnt_center_education_score_location > global_var:region_center.var:mnt_center_education_score_location
									}
								}
								set_global_variable = {
									name = region_center
									value = this
								}
							}
						}
					}
					global_var:region_center ?= {
						remove_location_modifier = mnt_center_education_local
						add_location_modifier = { modifier = mnt_center_education_regional days = -1 }

						if = {
							limit = {
								var:mnt_center_education_score_location >= center_education_continental_threshold
								OR = {
									NOT = { has_global_variable = continent_center }
									var:mnt_center_education_score_location > global_var:continent_center.var:mnt_center_education_score_location
								}
							}
							set_global_variable = {
								name = continent_center
								value = this
							}
						}
					}
				}
			}

			global_var:continent_center ?= {
				remove_location_modifier = mnt_center_education_regional
				add_location_modifier = { modifier = mnt_center_education_continental days = -1 }

				if = {
					limit = {
						# World centers (late game, very scarce)
						current_age_or_later = { age = age_5_absolutism }
						var:mnt_center_education_score_location >= center_education_world_threshold
						OR = {
							NOT = { has_global_variable = world_center }
							var:mnt_center_education_score_location > global_var:world_center.var:mnt_center_education_score_location
						}
					}
					set_global_variable = {
						name = world_center
						value = this
					}
				}
			}
		}
		global_var:world_center ?= {
			remove_location_modifier = mnt_center_education_continental
			add_location_modifier = { modifier = mnt_center_education_world days = -1 }
		}
		# Offices whose city no longer holds a production centre: yearly chance to open the inquiry.
		every_ownable_location = {
			limit = {
				has_owner = yes
				OR = {
					has_building = building_type:mnt_office_textile
					has_building = building_type:mnt_office_ceramics
					has_building = building_type:mnt_office_metalwork
					has_building = building_type:mnt_office_naval
					has_building = building_type:mnt_office_learning
					has_building = building_type:mnt_office_luxury
					has_building = building_type:mnt_office_sundries
					has_building = building_type:mnt_office_textile_board
					has_building = building_type:mnt_office_ceramics_board
					has_building = building_type:mnt_office_metalwork_board
					has_building = building_type:mnt_office_naval_board
					has_building = building_type:mnt_office_learning_board
					has_building = building_type:mnt_office_luxury_board
					has_building = building_type:mnt_office_sundries_board
				}
				NOT = {
					OR = {
						has_location_modifier = mnt_center_production_local
						has_location_modifier = mnt_center_production_regional
						has_location_modifier = mnt_center_production_continental
						has_location_modifier = mnt_center_production_world
					}
				}
			}
			on_center_of_production_status_loss = yes
		}

	}
}
