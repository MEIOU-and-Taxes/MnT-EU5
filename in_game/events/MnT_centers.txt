namespace = mnt_centers

# Game-start seeding: scores every owned land location and ranks them into
# center-of-importance tiers so the modifiers are present from the first day.
# After game start, scoring is handled by the yearly balanced location pass
# and ranking by mnt_centers.2.

mnt_centers.1 = {
	type = country_event
	title = hidden_event.t
	hidden = yes

	immediate = {
		# Clear previous center modifiers
		every_ownable_location = {
			limit = {
				# By checking the conditions here, later we can simply check for the existence of the variable
				has_owner = yes
			}
			calc_location_center_scores = yes
		}

		mnt_rank_centers = yes
	}
}

# Monthly ranking pass: the balanced scoring pass finishes its twelfth chunk
# that month, and this event runs one day after the monthly pulse, so every
# location's score is at most a year old when ranked.

mnt_centers.2 = {
	type = country_event
	title = hidden_event.t
	hidden = yes

	trigger = {
		great_power_ranking = 1
	}

	immediate = {
		set_local_variable = {
			name = mnt_centers_cycle_month
			value = { value = global_var:mnt_wealth_months_elapsed modulo = 12 }
		}
		if = {
			limit = { local_var:mnt_centers_cycle_month = 11 }
			mnt_rank_centers = yes
		}
	}
}
