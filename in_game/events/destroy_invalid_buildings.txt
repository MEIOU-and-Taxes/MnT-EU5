namespace = destroy_invalid_buildings
destroy_invalid_buildings.1 = {
    type = country_event
    title = hidden_event.t
    hidden = yes
    immediate = {
        every_owned_location = {
            if = {
                limit = {
                    has_building_with_at_least_one_level = fruit_orchard
                }
                change_building_level_in_location = {
                    building = building_type:RGO_building_fruit
                    value = {
                        add = "location_building_level(building_type:fruit_orchard)"
                        max = rgo_fruit_max_level
                        floor = yes	#the caps are fractional; round down rather than over
                    }
                }
                destroy_all_buildings_of_type = building_type:fruit_orchard
            }
            if = {
                limit = {
                    has_building_with_at_least_one_level = lumber_mill
                }
                change_building_level_in_location = {
                    building = building_type:RGO_building_lumber
                    value = {
                        add = "location_building_level(building_type:lumber_mill)"
                        max = rgo_lumber_max_level
                        floor = yes	#the caps are fractional; round down rather than over
                    }
                }
                destroy_all_buildings_of_type = building_type:lumber_mill
            }
            if = {
                limit = {
                    has_building_with_at_least_one_level = sand_pit
                }
                change_building_level_in_location = {
                    building = building_type:RGO_building_bad_sand
                    value = {
                        add = "location_building_level(building_type:sand_pit)"
                        max = bad_sand_max_level
                        floor = yes	#the caps are fractional; round down rather than over
                    }
                }
                destroy_all_buildings_of_type = building_type:sand_pit
            }
            if = {
                limit = {
                    has_building_with_at_least_one_level = clay_pit
                }
                change_building_level_in_location = {
                    building = building_type:RGO_building_clay
                    value = {
                        add = "location_building_level(building_type:clay_pit)"
                        max = rgo_clay_max_level
                        floor = yes	#the caps are fractional; round down rather than over
                    }
                }
                destroy_all_buildings_of_type = building_type:clay_pit
            }
            if = {
                limit = {
                    has_building_with_at_least_one_level = sheep_farms

                }
                change_building_level_in_location = {
                    building = building_type:RGO_building_wool
                    value = {
                        add = "location_building_level(building_type:sheep_farms)"
                        max = rgo_wool_max_level
                        floor = yes	#the caps are fractional; round down rather than over
                    }
                }
                destroy_all_buildings_of_type = building_type:sheep_farms
            }
            if = {
                limit = {
                    has_building = building_type:fishing_village
                }
                destroy_all_buildings_of_type = building_type:fishing_village
            }
            if = {
                limit = {
                    has_building = building_type:forest_village
                }
                destroy_all_buildings_of_type = building_type:forest_village
            }
        }
    }
}
