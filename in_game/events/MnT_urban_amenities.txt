namespace = mnt_urban_amenities

#fires once at startup
mnt_urban_amenities.1 = {
    type = country_event
    title = hidden_event.t
    hidden = yes

    immediate = {
		every_owned_location = {
			if = {
				limit = {
					location_rank ?= location_rank:town
				}
				change_building_level_in_location = {
					building = building_type:urban_amenities
					value = 2
				}
				set_local_variable = { name = mnt_fill_amount value = { value = 2 multiply = large_burgher_employment } }
				mnt_promote_pops = { from = pop_type:peasants to = pop_type:burghers amount = local_var:mnt_fill_amount }
			}
			else_if = {
				limit = {
					location_rank ?= location_rank:city
				}
				change_building_level_in_location = {
					building = building_type:urban_amenities
					value = 6
				}
				set_local_variable = { name = mnt_fill_amount value = { value = 6 multiply = large_burgher_employment } }
				mnt_promote_pops = { from = pop_type:peasants to = pop_type:burghers amount = local_var:mnt_fill_amount }
			}
			else_if = {
				limit = {
					location_rank ?= location_rank:megalopolis
				}
				change_building_level_in_location = {
					building = building_type:urban_amenities
					value = 10
				}
				set_local_variable = { name = mnt_fill_amount value = { value = 10 multiply = large_burgher_employment } }
				mnt_promote_pops = { from = pop_type:peasants to = pop_type:burghers amount = local_var:mnt_fill_amount }
			}
			remove_variable = building_urban_amenities
		} 
    }
}
