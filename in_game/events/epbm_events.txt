
namespace = epbm

# AI monthly: aggregate location and estate costs, copy mod scales for
# refund (AI charges and calcs same tick), then apply the combined
# EPBM + savings-valve drip so AI treasuries move every month.

epbm.1 = {
	type = country_event
	title = epbm_hidden_event.t
	hidden = yes
	outcome = neutral

	trigger = { is_ai = yes }

	immediate = {
		epbm_aggregate_maintenance = yes
		epbm_for_each_estate = { effect = epbm_snapshot_mod }
		mnt_change_estate_gold = yes
	}

	option = { name = "OK" }
}
