
namespace = epbm

# Monthly, every country with an active estate. AI aggregates its cached
# costs first and pays the fresh bill; the player pays last month's
# calculation and recalculates after. mnt_change_estate_gold is the only
# place estate gold moves.

epbm.1 = {
	type = country_event
	title = epbm_hidden_event.t
	hidden = yes
	outcome = neutral

	immediate = {
		if = { limit = { is_ai = yes } epbm_aggregate_maintenance = yes }
		mnt_change_estate_gold = yes
		if = { limit = { is_ai = no } epbm_calculate_maintenance = yes }
	}

	option = { name = "OK" }
}

# AI yearly: reprice estate-owned and foreign buildings.

epbm.2 = {
	type = country_event
	title = epbm_hidden_event.t
	hidden = yes
	outcome = neutral

	trigger = { is_ai = yes }

	immediate = { epbm_calculate_estate_and_foreign = yes }

	option = { name = "OK" }
}
