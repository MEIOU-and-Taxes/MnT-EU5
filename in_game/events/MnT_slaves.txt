namespace = mnt_slaves

mnt_slaves.1 = {
    type = country_event
    title = hidden_event.t
    hidden = yes

    immediate = {
		every_owned_location = {
			limit = { any_pop = { pop_type = pop_type:slaves } }
			# Since vanilla assumes mass enslavement, remove all slaves on startup for now until we can better address this
			every_pop = {
				limit = {
					pop_type = pop_type:slaves
				}
				change_pop_type = pop_type:peasants
			}
			# Setup filled the building jobs while these pops were still slaves, so fill them from the new peasants
			mnt_fill_building_jobs = yes
		}
	}
}
