namespace = diplomatic_spending

# Recalculate demands from diplomatic spending
# Unlike cost of the court, a proportion of spending will be directly transferred to estate
diplomatic_spending.1 = {
    type = country_event
    title = hidden_event.t
    hidden = yes
    trigger = {
        country_type = location
    }
    immediate = {
        set_local_variable = {
            name = country_diplomatic_spending
            value = {
                value = 0
                add = {
                    value = "root.diplomatic_maintenance"
                    multiply = "root.modifier:diplomatic_spending_cost"
                    multiply = "root.country_economical_base"
                }
            }
        }

        # Add a 1 month delayed display so the tooltip shows accurately to whats being charged.
        if = { 
            limit = {
                has_variable = noble_diplomatic_spending_transfer
            }
            set_variable = {
                name = noble_diplomatic_spending_transfer_tooltip_display
                value = var:noble_diplomatic_spending_transfer
            }
        }

        if = {      #Failsafe: Don't do this in the odd case that there isn't a nobility
            limit = {
                country_has_estate = estate_type:nobles_estate
            }
            # Give nobility a share based on their power
            # Give 10% extra with elaborate_court_ceremonies privilege
            set_variable = {
                name = noble_diplomatic_spending_transfer
                value = {
                    value = 0
                    add = {
                        value = local_var:country_diplomatic_spending
                        multiply = "estate_power(estate_type:nobles_estate)"
                    }
                }
            }

            if = {
                limit = {
                    has_estate_privilege = estate_privilege:elaborate_court_ceremonies
                }
                change_variable = {
                    name = noble_diplomatic_spending_transfer
                    multiply = 1.1
                }
            }

            "estate(estate_type:nobles_estate)" = {
                estate_add_gold = {
                    value = root.var:noble_diplomatic_spending_transfer
                }
            }
        }

        # The other 50% of the spending goes to generate market demands
        capital.market.location = {
            if = {
                limit = {
                    not = { has_variable = total_diplomatic_spending}
                }
                set_variable = {
                    name = total_diplomatic_spending
                    value = 0
                }
            }
            if = {
                limit = {
                    root = {
                        country_has_estate = estate_type:nobles_estate
                    }
                }
                change_variable = {
                    name = total_diplomatic_spending
                    add = {
                        value = local_var:country_diplomatic_spending
                    }
                    subtract = {
                        value = root.var:noble_diplomatic_spending_transfer
                    }
                }
            }
            else = {
                change_variable = {
                    name = total_diplomatic_spending
                    add = {
                        value = local_var:country_diplomatic_spending
                        multiply = 0.5  #Balance here how much is spent in case Nobles don't exist. Extremely unlikely to be ever the case.
                    }
                }
            }
        }
    }
}

# Aggregate all diplomatic spending demands into temporary market demands.
# The per-market work rides the load balancer's market index, spread across
# up to 512 worker countries instead of one rank-1 great power. Fires one day
# after the monthly pulse, so every country's diplomatic_spending.1
# accumulation is already in.
diplomatic_spending.2 = {
    type = country_event
    title = hidden_event.t
    hidden = yes
    # No trigger: workers are threads sweeping the market index, so a worker's
    # own standing is irrelevant, and any entry condition the balancer's
    # reindex does not share drops that worker's slice.
    immediate = {
        mnt_wealth_load_balance_this_effect = {
            CYCLE = 1
            PASS_KEY = 5
            INDEX_ACCESSOR = mnt_wealth_markets_index_accessor
            INDEX_SIZE = mnt_wealth_market_count
            EFFECT = apply_diplomatic_spending_demand
        }
    }
}
