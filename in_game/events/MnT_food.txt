namespace = mnt_food

#fires once at startup
mnt_food.1 = {
    type = country_event
    title = hidden_event.t
    hidden = yes

    immediate = {
		every_owned_location = {
			limit = { can_build_building = building_type:farming_village }
			destroy_all_buildings_of_type = building_type:farming_village
			change_building_level_in_location = {
				building = building_type:farming_village
				value = {
					add = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)"
					floor = yes
					subtract = 1
				}
			}
		} 
    }
}

#frees a peasant where a location has run out of unemployed ones
mnt_food.2 = {
	type = country_event
	title = hidden_event.t
	hidden = yes

	immediate = {
		every_market_in_world = {
			random_location_in_market = {
				limit = {
					has_building_with_at_least_one_level = farming_village
					num_pop_type:peasants > 1
					"unemployed_pops_of_pop_type_in_location(pop_type:peasants)" <= 0
				}
				change_building_level_in_location = {
					building = building_type:farming_village
					value = -1
				}
			}
		}
	}
}

mnt_food.3 = {
	type = country_event
	title = hidden_event.t
	desc = hidden_event.d
	hidden = yes

	trigger = {
		country_exists_and_can_own_locations = yes
	}

	immediate = {
		every_owned_location = {
			limit = {
				OR = {
					can_build_building = building_type:farming_village
					has_building_with_at_least_one_level = farming_village
				}
			}
			change_building_level_in_location = {
				building = building_type:farming_village
				value = {
					add = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)"
					floor = yes
					subtract = 1
				}
			}
		}
	}
}
