namespace = mnt_food

#fires once at startup
mnt_food.1 = {
    type = country_event
    title = hidden_event.t
    hidden = yes

    immediate = {
		every_owned_location = {
			limit = { 
				can_build_building = building_type:farming_village
				num_pop_type:peasants > 1.5
			}
			destroy_all_buildings_of_type = building_type:farming_village
			change_building_level_in_location = {
				building = building_type:farming_village
				value = {
					add = num_pop_type:peasants
					divide = 2.5 #one per 2500 peasants
					max = farming_village_cap
					floor = yes	#round down to make sure we don't accidentally go over the cap
				}
			}
		} 
    }
}

mnt_food.2 = {
	type = country_event
	title = hidden_event.t

	trigger = {
		country_exists_and_can_own_locations = yes
		is_ai = yes
		any_market_present_in_country = {
			market_monthly_food_balance < 0
			market_food_percentage < 80 #a check to make sure this isn't just a monthly weird war spike.
		}
	}
	
	hidden = yes
	
	immediate = {
		if = {
			limit = { monthly_income_total > 15 }
			every_market_present_in_country = {
				limit  = { 
					market_monthly_food_balance < 0
					market_food_percentage < 80
				}
				random_location_in_market = {
					limit  = { 
						owner ?= root
						can_build_building = building_type:farming_village
						location_unemployed_population_for_building_type = {
							building_type = building_type:farming_village
							value >= 800 #it's ok if this partially employs, better than nothing
						}
						
					}
					construct_building = {
						building_type = building_type:farming_village
					}
				}
				if = {
					limit  = { market_monthly_food_balance < -50 }
					random_location_in_market = {
						limit  = { 
							owner ?= root
							can_build_building = building_type:farming_village
							location_unemployed_population_for_building_type = {
								building_type = building_type:farming_village
								value >= 800 #it's ok if this partially employs, better than nothing
							}
							
						}
						construct_building = {
							building_type = building_type:farming_village
						}
					}
				}
			}
		}
		else = {
			every_market_present_in_country = {
				limit  = { 
					market_monthly_food_balance < 0
					market_food_percentage < 80
				}
				random_location_in_market = {
					limit  = { 
						owner ?= root
						can_build_building = building_type:farming_village
						location_unemployed_population_for_building_type = {
							building_type = building_type:farming_village
							value >= 800 #it's ok if this partially employs, better than nothing
						}
					}
					construct_building = {
						building_type = building_type:farming_village
						cost_multiplier = 0
						cost_multiplier_reason = "ai_broke_so_farm_village_free"
					}
				}
			}
		}
	}
}

#This event looks for markets with large amounts of food and frees up peasants in a random location without a large peasant surplus. This leaves the peasants free to migrate, be employed in other places, etc.
mnt_food.3 = {
	type = country_event
	title = hidden_event.t
	desc = hidden_event.d
	
	trigger = {
		country_exists_and_can_own_locations = yes
		is_ai = yes
	}
	
	hidden = yes
	
	immediate = {
		every_market_present_in_country = {
			limit  = { market_monthly_food_balance > 500 } #much safer check, since there are never positive spikes of food, only negative.
			random_location_in_market = {
				limit = {
					owner ?= root
					has_building_with_at_least_one_level = farming_village
					location_unemployed_population_for_building_type = {
						building_type = building_type:farming_village
						value < 2000
					}
				}
				destroy_building = "building(building_type:farming_village|owner)"
			}
			random_location_in_market = {
				limit = {
					owner ?= root
					has_building_with_at_least_one_level = farming_village
					location_unemployed_population_for_building_type = {
						building_type = building_type:farming_village
						value < 2000
					}
				}
				destroy_building = "building(building_type:farming_village|owner)"
			}
		}	
	}
}
# Cache the local food balance once a month. Native starvation is checked live by the auto modifier.
mnt_food.4 = {
    type = country_event
    hidden = yes
    title = hidden_event.t

    immediate = {
        every_province = {
            set_local_variable = { name = mnt_food_production value = 0 }
            set_local_variable = { name = mnt_food_consumption value = 0 }
            every_location_in_province = {
                change_local_variable = { name = mnt_food_production add = food_production }
                change_local_variable = { name = mnt_food_consumption add = mnt_location_food_consumption_amount }
            }
            set_variable = {
                name = mnt_food_local_surplus
                value = {
                    value = local_var:mnt_food_production
                    subtract = local_var:mnt_food_consumption
                }
            }

            set_variable = { name = mnt_food_ai_weight value = 1 }
            set_variable = { name = mnt_food_starvation_severity value = 0 }
            if = {
                limit = { local_var:mnt_food_consumption > 0 }
                set_local_variable = {
                    name = mnt_food_surplus_share
                    value = {
                        value = var:mnt_food_local_surplus
                        divide = local_var:mnt_food_consumption
                    }
                }

                # Ease the weight from 10 at break-even to 1 at a 40% surplus.
                set_local_variable = {
                    name = mnt_food_weight_taper
                    value = {
                        value = 0.40
                        subtract = local_var:mnt_food_surplus_share
                        divide = 0.40
                        min = 0
                        max = 1
                    }
                }
                set_variable = {
                    name = mnt_food_ai_weight
                    value = {
                        value = 9
                        multiply = local_var:mnt_food_weight_taper
                        multiply = local_var:mnt_food_weight_taper
                        add = 1
                    }
                }

                # Keep severity ready if reserves run out after the monthly update.
                set_variable = {
                    name = mnt_food_starvation_severity
                    value = {
                        value = local_var:mnt_food_surplus_share
                        multiply = -1
                        min = 0
                        max = 1
                    }
                }
            }
        }
    }
}
