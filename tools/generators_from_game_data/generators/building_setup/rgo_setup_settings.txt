# RGO SETUP SETTINGS
#
# Starting RGO building levels. Other buildings: building_setup_settings.txt.
# Run after editing (py on Windows):
#     python3 tools/generators_from_game_data/generators/building_setup/building_setup.py
# New game data (after patches, location/pop/building changes): run with --setup-data-dump,
# start a new game in debug to the country screen, close, run the generator again.
#
# [all RGO buildings] sets defaults. A named section like [RGO_building_lumber] overrides what it
# sets; None drops a setting. Each location gets the RGO for its raw material; secondary RGOs are
# levelled the same way but without ideal_rgo_multiplier. fixed_level bypasses the normal
# base+multiplier calculation.
# Values: yes/no, number (25 = 25%), None, list [ "SWE" ], or map { "flatland": 1, "hills": 0.5 }.
# A map only needs what it changes; unlisted names count as 1. Multipliers compound.


[general]
# Many pops change type while the game sets up, through setup rules and promotion into jobs,
# so the count of one pop type at game start can be unreliable. all is the steadiest measure.
pop_types_for_levels = [ "peasants", "laborers", "slaves" ]   # The pops counted for levels. Choose from: burghers clergy laborers nobles peasants slaves soldiers tribesmen, or all for every pop type
respect_technology = yes                          # yes: no RGO building the owner cannot build yet, like an iron RGO before Iron Working.
place_secondary_rgos = yes                        # yes: locations also get secondary RGO buildings. max_secondary_rgos_per_location only matters then.
max_secondary_rgos_per_location = 2               # The most secondary RGO buildings one location can get.

# All multipliers compound. lake_multiplier = 1.25 and coastal_multiplier = 1.5 == 1.25 * 1.5
[all RGO buildings]
place = yes                                # yes: this RGO building can be placed at all.
place_as_secondary = yes                   # yes: this RGO building can be picked as a secondary building.
ideal_rgo_multiplier = 2                   # Multiplies the main building's level. 2 makes the other RGO buildings about half its size.
destroy_vanilla_building_levels_before_counting = no # yes: the levels this building starts the game with are removed, so the level here is the whole level.
pops_per_level = 20                        # Base level: thousands of counted pops per level. 10 means one level per 10,000.


final_level_multiplier = 1                 # Multiplies every level. 2 doubles them.
round_levels = up                          # How levels are rounded: up, down or nearest.
percentage_of_building_max_level = 33.3    # The highest level, as a percent of the building's max level. None means no limit.
fixed_level_floor = 1                      # The lowest level, a whole number. Overrides the percentage. None means no floor.
