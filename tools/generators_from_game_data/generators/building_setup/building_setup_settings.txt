# BUILDING SETUP SETTINGS
#
# Starting building levels for non-RGO buildings. RGO buildings: rgo_setup_settings.txt.
# Run after editing (py on Windows):
#     python3 tools/generators_from_game_data/generators/building_setup/building_setup.py
# New game data (after patches, location/pop/building changes): run with --setup-data-dump,
# start a new game in debug to the country screen, close, run the generator again.
#
# [all buildings] sets defaults. A named section like [victualling_house] overrides what it sets;
# None drops a setting. A distribution_group section shares one computed level round-robin among
# its building types (such as all valid villages getting 10 collective levels). fixed_level bypasses
# the normal base+multiplier calculation.
# Values: yes/no, number (25 = 25%), None, list [ "SWE" ], or map { "flatland": 1, "hills": 0.5 }.
# rgo_setup_settings.txt lists the names each multiplier map accepts.

# Many pops change type while the game sets up, through setup rules and promotion into jobs,
# so the count of one pop type at game start can be unreliable. all is the steadiest measure.
pop_types_for_levels = [ "all" ]   # The pops counted for levels. Choose from: burghers clergy laborers nobles peasants slaves soldiers tribesmen, or all for every pop type
respect_technology = yes                          # yes: no building the owner cannot build yet, like a gun smith before its advance.

[all buildings]
place = yes                                # yes: this building type is placed at all.
destroy_vanilla_building_levels_before_counting = no # yes: the levels this building type starts the game with are removed, so the level here is the whole level.
pops_per_level = None                      # Base level: thousands of counted pops per level. 10 means one level per 10,000.
development_per_level = None               # Base level: development per level. 10 means one level per 10 development.
base_level_by_location_rank = {            # Base level by the location's rank at the start. A rank left out adds nothing.
    "town": 1,
    "city": 2,
    "megalopolis": 3,
}
additive_multiplier_per_development = 0.02 # Multiplier scaling with development. 0.02 at 20 development applies a 1.4x multiplier. 1+(0.02*20)
additive_multiplier_per_population = 0.002 # Multiplier scaling with counted pops, per thousand. 0.01 at 20,000 pops applies a 1.2x multiplier. 1+(0.01*20)
lake_multiplier = None                     # Multiplier when the location borders a lake.
coastal_multiplier = None                  # Multiplier when the location is on the coast.
river_multiplier = None                    # Multiplier by the location's river size, or none.
location_rank_multiplier = None            # Multiplier by the location's rank at the start.

topography_multiplier = {                  # Multiplier by the location's topography. Clear land gets a little more, rough land loses more.
    "flatland": 1.1,
    "plateau": 1,
    "hills": 0.9,
    "wetlands": 0.7,
    "mountains": 0.6,
}

vegetation_multiplier = {                  # Multiplier by the location's vegetation. Clear land gets a little more, rough land loses more.
    "farmland": 1.1,
    "grasslands": 1.1,
    "sparse": 1,
    "woods": 0.9,
    "desert": 0.7,
    "forest": 0.7,
    "jungle": 0.6,
}

climate_multiplier = None                  # Multiplier by the location's climate.
final_level_multiplier = 1                 # Multiplies every level. 2 doubles them.
round_levels = down                        # How levels are rounded: up, down or nearest.
percentage_of_building_max_level = None    # The highest level, as a percent of the building's max level. None means no limit.
fixed_level = None                         # Instead of steps 1 to 3: a level for every location, or a list by location rank. Steps 1 to 3 must be None.
fixed_level_floor = None                   # The lowest level, a whole number. Overrides the percentage. None means no floor.
fixed_level_ceiling = None                 # The highest level, a whole number. Overrides the percentage and the floor. None means no ceiling.
