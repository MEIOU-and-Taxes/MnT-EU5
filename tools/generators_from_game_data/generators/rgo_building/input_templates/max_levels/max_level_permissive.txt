rgo_$GOOD$_max_level = {
	if = {
		limit = { has_owner = yes }
		if = {
			limit = { raw_material = goods:$GOOD$ }
			add = { desc = "MNT_RGO_BASE_IDEAL_$GOOD_UPPER$"
				value = modifier:mnt_rgo_$METHOD$_size
				add = modifier:mnt_rgo_$GOOD$_size
				save_temporary_value_as = base_before_ideal
				value = 1
				add = modifier:local_raw_material_output
				add = owner.modifier:global_raw_material_output
				multiply = scope:base_before_ideal
			}
			save_temporary_value_as = base_value
			add = mnt_rgo_size_modifiers
			min = 1
		}
		else = {
			add = { desc = "MNT_RGO_BASE_$GOOD_UPPER$"
				value = modifier:mnt_rgo_$METHOD$_size
				add = modifier:mnt_rgo_$GOOD$_size
			}
			save_temporary_value_as = base_value
			add = mnt_rgo_size_modifiers
$ELSE_EXTRA$		}
		min = 0
	}
}
